81   /**
82    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
83    * copied before codegen.
84    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.
85    */
86   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
87     val expressions = ShimUtils.copyStateful(dirtyExpressions)
88     import ctx._
89     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
90 
91     // Add each expression tree and compute the common subexpressions.
92     expressions.foreach(equivalentExpressions.addExprTree(_))
93     var subexprFunctions = ""
94     // Get all the expressions that appear at least twice and set up the state for subexpression
95     // elimination.
96     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
97     commonExprs.foreach { expr =>
98       val fnName = freshName("subExpr")
99       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
100       val value = addMutableState(javaType(expr.dataType), "subExprValue")
101 
102       // Generate the code for this expression tree and wrap it in a function.
103       val eval = expr.genCode(ctx)
104       val fn =
105         s"""
106            |private void $fnName(InternalRow $INPUT_ROW) {
107            |  ${eval.code}
108            |  $isNull = ${eval.isNull};
109            |  $value = ${eval.value};
110            |}
111            """.stripMargin
112 
113       // Add a state and a mapping of the common subexpressions that are associate with this
114       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
115       // when it is code generated. This decision should be a cost based one.
116       //
117       // The cost of doing subexpression elimination is:
118       //   1. Extra function call, although this is probably *good* as the JIT can decide to
119       //      inline or not.
120       // The benefit doing subexpression elimination is:
121       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
122       //      above.
123       //   2. Less code.
124       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
125       // at least two nodes) as the cost of doing it is expected to be low.
126 
127       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
128       subexprFunctions += subExprCode
129       val state = ShimExprUtils.state(
130         ExprCode(code"$subExprCode",
131           JavaCode.isNullGlobal(isNull),
132           JavaCode.global(value, expr.dataType)))
133       ShimExprUtils.addSubExpr(ctx, expr, state)
134     }
135 
136     subexprFunctions
137   }
