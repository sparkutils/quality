82   /**
83    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
84    * copied before codegen.
85    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.

87   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
88     val expressions = ShimUtils.copyStateful(dirtyExpressions)
89     import ctx._
90     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
91 
92     // Add each expression tree and compute the common subexpressions.
93     expressions.foreach(equivalentExpressions.addExprTree(_))
94     var subexprFunctions = ""
95     // Get all the expressions that appear at least twice and set up the state for subexpression
96     // elimination.
97     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
98     commonExprs.foreach { expr =>
99       val fnName = freshName("subExpr")
100       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
101       val value = addMutableState(javaType(expr.dataType), "subExprValue")
102 
103       // Generate the code for this expression tree and wrap it in a function.
104       val eval = expr.genCode(ctx)
105       val fn =
106         s"""
107            |private void $fnName(InternalRow $INPUT_ROW) {
108            |  ${eval.code}
109            |  $isNull = ${eval.isNull};
110            |  $value = ${eval.value};
111            |}
112            """.stripMargin
113 
114       // Add a state and a mapping of the common subexpressions that are associate with this
115       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
116       // when it is code generated. This decision should be a cost based one.
117       //
118       // The cost of doing subexpression elimination is:
119       //   1. Extra function call, although this is probably *good* as the JIT can decide to
120       //      inline or not.
121       // The benefit doing subexpression elimination is:
122       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
123       //      above.
124       //   2. Less code.
125       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
126       // at least two nodes) as the cost of doing it is expected to be low.
127 
128       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
129       subexprFunctions += subExprCode
130       val state = ShimExprUtils.state(
131         ExprCode(code"$subExprCode",
132           JavaCode.isNullGlobal(isNull),
133           JavaCode.global(value, expr.dataType)))
134       ShimExprUtils.addSubExpr(ctx, expr, state)
135     }
136 
137     subexprFunctions
138   }

155   /**
156    * Splits the generated code of expressions into multiple functions, because function has
157    * 64kb code size limit in JVM. If the class to which the function would be inlined would grow
158    * beyond 1000kb, we declare a private, inner sub-class, and the function is inlined to it
159    * instead, because classes have a constant pool limit of 65,536 named values.
160    *
161    * @param expressions the codes to evaluate expressions.
162    * @param funcName the split function name base.
163    * @param arguments the list of (type, name) of the arguments of the split function.
164    * @param returnType the return type of the split function.
165    * @param makeSplitFunction makes split function body, e.g. add preparation or cleanup.
166    * @param foldFunctions folds the split function calls.
167    */
168   def splitExpressions(ctx: CodegenContext,
169                        expressions: Seq[String],
170                        funcName: String,
171                        arguments: Seq[(String, String)],
172                        returnType: String = "void",
173                        makeSplitFunction: String => String = identity,
174                        foldFunctions: Seq[String] => String = _.mkString("", ";\n", ";")): String = {
175     val blocks = buildCodeBlocks(expressions)
176 
177     if (blocks.length == 1) {
178       // inline execution if only one block
179       blocks.head
180     } else {
181       val func = ctx.freshName(funcName)
182       val argString = arguments.map { case (t, name) => s"$t $name" }.mkString(", ")
183       val functions = blocks.zipWithIndex.map { case (body, i) =>
184         val name = s"${func}_$i"
185         val code = s"""
186                       |private $returnType $name($argString) {
187                       |  ${makeSplitFunction(body)}
188                       |}
189          """.stripMargin
190         addNewFunctionInternal.invoke(ctx, name, code, false).asInstanceOf[NewFunctionSpec]
191       }
192 
193       val (outerClassFunctions, innerClassFunctions) = functions.partition(_.innerClassName.isEmpty)
194 
195       val argsString = arguments.map(_._2).mkString(", ")
196       val outerClassFunctionCalls = outerClassFunctions.map(f => s"${f.functionName}($argsString)")
197 
198       val innerClassFunctionCalls = generateInnerClassesFunctionCall.invoke(ctx,
199         innerClassFunctions,
200         func,
201         arguments,
202         returnType,
203         makeSplitFunction,
204         foldFunctions).asInstanceOf[Iterable[String]]
205 
206       foldFunctions(outerClassFunctionCalls ++ innerClassFunctionCalls)
207     }
208   }

211   /**
212    * Splits the generated code of expressions into multiple sequences of String
213    * based on a threshold of length of a String
214    *
215    * @param expressions the codes to evaluate expressions.
216    */
217   private def buildCodeBlocks(expressions: Seq[String]): Seq[String] = {
218     val blocks = new ArrayBuffer[String]()
219     val blockBuilder = new StringBuilder()
220     var length = 0
221     val splitThreshold = SQLConf.get.methodSplitThreshold
222     for (code <- expressions) {
223       // We can't know how many bytecode will be generated, so use the length of source code
224       // as metric. A method should not go beyond 8K, otherwise it will not be JITted, should
225       // also not be too small, or it will have many function calls (for wide table), see the
226       // results in BenchmarkWideTable.
227       if (length > splitThreshold) {
228         blocks += blockBuilder.toString()
229         blockBuilder.clear()
230         length = 0
231       }
232       blockBuilder.append(code)
233       length += code.length//CodeFormatter.stripExtraNewLinesAndComments(code).length
234     }
235     blocks += blockBuilder.toString()
236     blocks.toSeq
237   }
