305   def stripBrackets(v: VariableValue): (String, String) = {
306     val openb = v.toString().indexOf("[")
307     if (openb == -1)
308       (v.variableName, "")
309     else
310       (v.variableName.dropRight(v.length - openb), v.variableName.drop(openb))
311   }

313   def formatParams(ctx: CodegenContext, a: Seq[ExprValue], additional: Seq[ExprValue] = Seq.empty, callsKeepArrays: Boolean = false): ParameterInformation = {
314     def filterOutArrays(use: Seq[ExprValue]) = use.flatMap {
315       case a: VariableValue => Some(a)
316       case _ => None
317     }
318 
319     val filteredA = filterOutArrays(a)
320     val filteredAdditional = filterOutArrays(additional)
321 
322     val size = filteredA.size + filteredAdditional.size
323     val use =
324       if (size <= 22)
325         filteredA ++ filteredAdditional
326       else
327         filteredA // additional are then handled via class level
328 
329     // filter out any top level arrays, the input is a set, so params need the same order
330     val ordered = use
331 
332     def prepFields(ordered: Seq[VariableValue]) =
333       ordered.map { v =>
334         val (stripped, arrayInName) = stripBrackets(v)
335 
336         val (typ, array) =
337           if (v.javaType.isArray)
338             (s"${v.javaType.getComponentType.getName}", "[]")
339           else if (v.javaType.isPrimitive)
340             (v.javaType.toString, arrayInName.replaceAll("[^\\[\\]]",""))
341           else
342             (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
343 
344         (s"$typ$array", stripped, v.javaType)
345       }
346 
347     val pairs = prepFields(ordered)
348 
349     val combined =
350       if (size <= 22)
351         pairs
352       else
353         pairs ++ prepFields(filteredAdditional)
354 
355     val paramsCall =
356       ordered.map(v =>
357         if (v.javaType.isArray && callsKeepArrays)
358           v.variableName
359         else
360           stripBrackets(v)._1
361       ).mkString(", ")
362 
363     ParameterInformation(pairs.map {
364       case (typ, stripped, _) =>
365 
366         s"$typ $stripped"
367       }.mkString(", ")
368       , paramsCall, size, combined, outerCallParams = paramsCall, nonCombinedParams = pairs)
369   }
