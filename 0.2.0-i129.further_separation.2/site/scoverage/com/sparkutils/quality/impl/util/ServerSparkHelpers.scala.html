12 object Arrays {
13   /**
14    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
15    *
16    * @param array
17    * @param dataType
18    * @param f
19    * @return
20    */
21   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
22     array match {
23       case _: UnsafeArrayData =>
24         val res = Array.ofDim[T](array.numElements() + by)
25         array.foreach(dataType, (i, v) => res.update(i, f(v)))
26         res
27       case _ => {
28         val r = array.array.map(f)
29         if (by == 0)
30           r
31         else {
32           val res = Array.ofDim[T](array.numElements() + by)
33           r.copyToArray(res)
34           res
35         }
36 
37       }
38     }
39 
40   /**
41    * gets an array out of UnsafeArrayData or others
42    * @param array
43    * @param dataType
44    * @return
45    */
46   def toArray(array: ArrayData, dataType: DataType): Array[Any] =
47     array match {
48       case _: UnsafeArrayData =>
49         mapArray(array, dataType, identity)
50       case _ => array.array
51     }

142    *
143    * @param mapData the source mapdata to search through.
144    * @param cachedPositions already known positions to check for what
145    * @param what compared against the keyArray to obtain the position
146    * @param getValue obtain the value from the mapData, with any necessary conversions, against the index parameter
147    * @tparam T

150   def getMapEntry[T <: AnyRef](mapData: MapData, cachedPositions: ArrayBuffer[Int], keyTest: (ArrayData, Int) => Boolean )(getValue: Int => T): (T, ArrayBuffer[Int]) = {
151     val keys = mapData.keyArray()
152     val n = keys.numElements()
153 
154     def withCached() =
155       cachedPositions.find { i =>
156         if (i < n)
157           keyTest(keys, i)
158         else
159           false
160       }
161 
162     withCached().map(i => (getValue(i), cachedPositions))
163       .getOrElse {

229   def getKeyIndex(map: MapData, equal: (ArrayData, Int) => Boolean): Int = {
230     var index = -1
231     var i = 0
232     val keys = map.keyArray()
233     while(index == -1 && i < map.numElements()) {
234       if (equal(keys, i)) {
235         index = i
236       }
237       i += 1
238     }
