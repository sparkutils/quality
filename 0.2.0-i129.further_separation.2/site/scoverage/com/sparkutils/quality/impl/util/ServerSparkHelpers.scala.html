13 object Arrays {
14   /**
15    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
16    *
17    * @param array
18    * @param dataType
19    * @param f
20    * @return
21    */
22   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
23     array match {
24       case _: UnsafeArrayData =>
25         val res = Array.ofDim[T](array.numElements() + by)
26         array.foreach(dataType, (i, v) => res.update(i, f(v)))
27         res
28       case _ => {
29         val r = array.array.map(f)
30         if (by == 0)
31           r
32         else {
33           val res = Array.ofDim[T](array.numElements() + by)
34           r.copyToArray(res)
35           res
36         }
37 
38       }
39     }
40 
41   /**
42    * gets an array out of UnsafeArrayData or others
43    * @param array
44    * @param dataType
45    * @return
46    */
47   def toArray(array: ArrayData, dataType: DataType): Array[Any] =
48     array match {
49       case _: UnsafeArrayData =>
50         mapArray(array, dataType, identity)
51       case _ => array.array
52     }
