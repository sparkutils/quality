12 object Arrays {
13   /**
14    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
15    *
16    * @param array
17    * @param dataType
18    * @param f
19    * @return
20    */
21   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
22     array match {
23       case _: UnsafeArrayData =>
24         val res = Array.ofDim[T](array.numElements() + by)
25         array.foreach(dataType, (i, v) => res.update(i, f(v)))
26         res
27       case _ => {
28         val r = array.array.map(f)
29         if (by == 0)
30           r
31         else {
32           val res = Array.ofDim[T](array.numElements() + by)
33           r.copyToArray(res)
34           res
35         }
36 
37       }
38     }
39 
40   /**
41    * gets an array out of UnsafeArrayData or others
42    * @param array
43    * @param dataType
44    * @return
45    */
46   def toArray(array: ArrayData, dataType: DataType): Array[Any] =
47     array match {
48       case _: UnsafeArrayData =>
49         mapArray(array, dataType, identity)
50       case _ => array.array
51     }

137    * Assuming MapData entries are stable, as they are array backed, over a dataset.  When the position of 'what' is not in the cachedPositions a full search is performed with the new location being returned in addition.
138    *
139    * @param mapData the source mapdata to search through.
140    * @param cachedPositions already known positions to check for what
141    * @param what compared against the keyArray to obtain the position
142    * @param getValue obtain the value from the mapData, with any necessary conversions, against the index parameter
143    * @tparam T
144    * @return (result from getValue or null when not found, either cachedPositions or cachedPositions and a newly found position)
145    */
146   def getMapEntry[T <: AnyRef](mapData: MapData, cachedPositions: ArrayBuffer[Int], keyTest: (ArrayData, Int) => Boolean )(getValue: Int => T): (T, ArrayBuffer[Int]) = {
147     val keys = mapData.keyArray()
148     val n = keys.numElements()
149 
150     def withCached() =
151       cachedPositions.find { i =>
152         if (i < n)
153           keyTest(keys, i)
154         else
155           false
156       }
157 
158     withCached().map(i => (getValue(i), cachedPositions))
159       .getOrElse {

161         if (r == -1)

163         else

165       }

234     }
