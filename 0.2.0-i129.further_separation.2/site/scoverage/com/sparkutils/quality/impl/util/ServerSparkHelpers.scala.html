12 object Arrays {
13   /**
14    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
15    *
16    * @param array
17    * @param dataType
18    * @param f
19    * @return
20    */
21   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
22     array match {
23       case _: UnsafeArrayData =>
24         val res = Array.ofDim[T](array.numElements() + by)
25         array.foreach(dataType, (i, v) => res.update(i, f(v)))
26         res
27       case _ => {
28         val r = array.array.map(f)
29         if (by == 0)
30           r
31         else {
32           val res = Array.ofDim[T](array.numElements() + by)
33           r.copyToArray(res)
34           res
35         }
36 
37       }
38     }
39 
40   /**
41    * gets an array out of UnsafeArrayData or others
42    * @param array
43    * @param dataType
44    * @return
45    */
46   def toArray(array: ArrayData, dataType: DataType): Array[Any] =
47     array match {
48       case _: UnsafeArrayData =>
49         mapArray(array, dataType, identity)
50       case _ => array.array
51     }

135    *
136    * @param mapData the source mapdata to search through.
137    * @param cachedPositions already known positions to check for what
138    * @param what compared against the keyArray to obtain the position
139    * @param getValue obtain the value from the mapData, with any necessary conversions, against the index parameter
140    * @tparam T

142    */
143   def getMapEntry[T <: AnyRef](mapData: MapData, cachedPositions: ArrayBuffer[Int], keyTest: (ArrayData, Int) => Boolean )(getValue: Int => T): (T, ArrayBuffer[Int]) = {
144     val keys = mapData.keyArray()
145     val n = keys.numElements()
146 
147     def withCached() =
148       cachedPositions.find { i =>
149         if (i < n)
150           keyTest(keys, i)
151         else
152           false
153       }
154

156       .getOrElse {

162       }

223     var index = -1
224     var i = 0
225     val keys = map.keyArray()
226     while(index == -1 && i < map.numElements()) {

229       }

231     }
