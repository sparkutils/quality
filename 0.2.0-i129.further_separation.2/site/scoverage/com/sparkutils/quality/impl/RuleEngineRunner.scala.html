63       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals

65         expressions.map(PassThroughCompileEvals)

68 
69     // clean out expressions, UnresolvedRelations etc. from subquery usage forceRunnerEval,
70     val runner =
71       if (forceRunnerEval || resolveWith.isDefined)
72         new RuleEngineRunnerEval(cleaned, exprs, resultDataType, compileEvals,
73           debugMode, variablesPerFunc, variableFuncGroup, expressionOffsets = indexes,
74           forceTriggerEval, triggerCount = triggerCount, extraConfig)

77           debugMode, variablesPerFunc, variableFuncGroup, expressionOffsets = indexes,
78           forceTriggerEval, triggerCount = triggerCount, extraConfig)

82         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)

85           // replace the expr
