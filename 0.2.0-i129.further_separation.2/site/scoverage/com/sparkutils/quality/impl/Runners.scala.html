84       if (extraConfig.boolean(useEmptyRuleSetResults, false))
85         EmptyMap  // copy is expensive
86       else {
87 
88         val ids = new LongArray( ruleSuite.ruleSets.map{ ruleSet => packTheId(ruleSet.id) }.toArray )
89         val rules = ruleSuite.ruleSets.map{
90           ruleSet =>
91             InternalRow(defaultOverallResult, {
92               val ids = ruleSet.rules.map(r => packTheId(r.id)).toArray
93               val defaults = ruleSet.rules.map( _ => defaultRuleResult).toArray

95               new RuleSetMap(
96                 new LongArray(ids),
97                 new IntegerArray(defaults)
98               )
99             })
100         }

102         new ArrayBasedMapData(ids, new GenericArrayData(rules.toArray))
103       }
104     )

106   /**

108    * used by codegen
109    */
110   def applyResult(level1: Int, level2: Int, result: InternalRow, ruleResult: Int): Unit = {
111     val sar = result.getMap(2).asInstanceOf[ArrayBasedMapData]
112     // update result directly
113     val sv = sar.valueArray.asInstanceOf[GenericArrayData]
114     val struct = sv.getStruct(level1, 2)
115     struct.getMap(1).valueArray().asInstanceOf[IntegerArray].update(level2, ruleResult) // comment out to squeeze extra, spark still more expensive by 20s, nothing in quality code
116 
117     // processOverall
118     val cur = struct.getInt(0)
119     val nr = defaultOverallProcessor(ruleResult, cur)
120     struct.update(0, nr)
121     result.update(1, defaultOverallProcessor(nr, result.getInt(1)))
122   }

224       generatorClazz_ = generatorClassSource.map(CodeGenerator.compile(_)._1)

226       val end = System.nanoTime()

228       if (extraConfig.boolean(showSplitCompilationTime, false)){
