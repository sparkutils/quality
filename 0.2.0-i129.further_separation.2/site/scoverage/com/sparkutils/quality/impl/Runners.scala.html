98   def applyResult(level1: Int, level2: Int, result: InternalRow, ruleResult: Int): Unit = {
99     val sar = result.getMap(2).asInstanceOf[ArrayBasedMapData]
100     // update result directly
101     val sv = sar.valueArray.asInstanceOf[GenericArrayData]
102     val struct = sv.getStruct(level1, 2)
103     struct.getMap(1).valueArray().asInstanceOf[GenericArrayData].update(level2, ruleResult) // comment out to squeeze extra, spark still more expensive by 20s, nothing in quality code
104 
105     // processOverall
106     val cur = struct.getInt(0)
107     val nr = defaultOverallProcessor(ruleResult, cur)
108     struct.update(0, nr)
109     result.update(1, defaultOverallProcessor(nr, result.getInt(1)))
110   }
