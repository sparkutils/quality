82   def createDefaultRuleResult(): InternalRow =
83     InternalRow(packTheId(ruleSuite.id), defaultOverallResult,
84       if (extraConfig.boolean(useEmptyRuleSetResults, false))
85         EmptyMap  // copy is expensive
86       else
87         ArrayBasedMapData(
88           ruleSuite.ruleSets.map{
89             ruleSet =>
90               packTheId(ruleSet.id) -> InternalRow(defaultOverallResult, {
91                 val ids = ruleSet.rules.map(r => packTheId(r.id)).toArray
92                 val defaults = ruleSet.rules.map( _ => defaultRuleResult).toArray
93                 // use optimised copys, by default generic array does scanning

95                   new LongArray(ids),

98               })

103   /**
104    * rolls the overalls up in place - must be genericarraydata / arraybasedmap data with a copy from createDefaultRuleResult
105    * used by codegen
106    */
107   def applyResult(level1: Int, level2: Int, result: InternalRow, ruleResult: Int): Unit = {
108     val sar = result.getMap(2).asInstanceOf[ArrayBasedMapData]
109     // update result directly
110     val sv = sar.valueArray.asInstanceOf[GenericArrayData]
111     val struct = sv.getStruct(level1, 2)
112     struct.getMap(1).valueArray().asInstanceOf[IntegerArray].update(level2, ruleResult) // comment out to squeeze extra, spark still more expensive by 20s, nothing in quality code
113 
114     // processOverall
115     val cur = struct.getInt(0)
116     val nr = defaultOverallProcessor(ruleResult, cur)
117     struct.update(0, nr)
118     result.update(1, defaultOverallProcessor(nr, result.getInt(1)))
119   }
120 
121   /**
122    * only applies overallResult to the top level result and ignores any rule or ruleset level information
123    * used by codegen
124    */
125   def applyEmptyResult(level1: Int, level2: Int, result: InternalRow, ruleResult: Int): Unit = {
126     result.update(1, defaultOverallProcessor(ruleResult, result.getInt(1)))
127   }

129   def inPlaceArrayOffsets(ctx: CodegenContext, resultRow: String, ruleRunnerExpressionIdx: Int): InPlaceOffsets = {
130     val className = this.getClass.getName

147                $runner.apply${empty}Result($level1, $level2, $resultRow, $result);
148                 """
149         }

151   }

216   def generatorClazz(i: Int): GeneratedClass = {
217     // allow it to be replaced
218     if (generatorClazz_ == null) {
219       val start = System.nanoTime()
220 
221       generatorClazz_ = generatorClassSource.map(CodeGenerator.compile(_)._1)
222 
223       val end = System.nanoTime()

225       if (extraConfig.boolean(showSplitCompilationTime, false)){
226         println(s"${this.getClass.getSimpleName} RuleSuite ${ruleSuite.id} - took ${compileTime.toMinutes}m${compileTime.toSeconds % 60}s to compile")
227       }
228     }
