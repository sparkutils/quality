114 
115     val groupExprs = groups.map(_.groupFilter)
116

118

122         for (exprGroup <- grouped) yield {

144                       genParamsForNested(ctx, allGroupExprs, additionalParams).copy(paramsDef = "", paramsCall = "")
145 
146                     val resCode = ExprCode(VariableValue(ctx.freshName("groupResultNull"), java.lang.Boolean.TYPE),
147                       VariableValue(ctx.freshName("groupResult"), classOf[GenericInternalRow]))
148                     val (body, expr) =

150                         SubCompilation(id, s"Trigger group $groupIndex"),
151                         // we need to pipe the row in
152                         extraParams = additionalParams,
153                         useParams = preCalcParams

155                         // group the group, params holds any subexprs used/generated for this sub compilation
156 
157                         val grpResult = ctx.freshName("groupResult")
158 
159                         val sgr = simpleGrouper(ctx, runner, grpResult, additionalParams,

164                           ExprCode(VariableValue(ctx.freshName("groupResultNull"), java.lang.Boolean.TYPE),

173                               ${funNames.map { f => s"$f(${params.paramsCall});" }.mkString("\n")}

175                                 new Object[]{
176                                 ${additionalParams.filterNot(_.javaType.isArray).map(_.variableName).mkString(",\n")}
177                                 }
178                               );
179                               """
180                         ), sgr.extraClasses, sgr.ignoreTopLevelSubExpressions)
181                       }

193                         ${eval.code}

195                           ${expr.code}

206               (ctx.addNewFunction(exprFuncName,
207                 code"""
208                  private void $exprFuncName(${params.paramsDef}) {
209                    $body
210                  }
211                 """.code
212               ), groupCalls.map(_._2))
213             }
214 
215           (ctx.addNewFunction(groupName,
216 
217             code"""
218              private void $groupName(${params.paramsDef}) {
219                ${exprFunEnd()}
220                ${subClasses.map { f => s"${f._1}(${params.paramsCall});" }.mkString(s"${exprFunEnd()}\n")}
221              }

229     }

233

238     } else {

242       )
243       val subExpressionCode = ShimExprUtils.evaluateSubExprEliminationState(ctx, subExprs)
244 
245       val (funNames, extraClasses) =

247           builder
248         }
249       TriggerResult(funNames.iterator, SeparateCompilation.splitGlobalSubExprs(ctx, subExpressionCode), extraClasses, true)
250     }
251   }
