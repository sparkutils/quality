104   protected def performGrouping(ctx: CodegenContext, runner: Runner, resultRow: String,
105                                 additionalParams: Seq[VariableValue],
106                                 expressions: Seq[(Trigger, (CodegenContext, ParameterInformation, Expression) => Block)],
107                                 params: ParameterInformation, prefix: String, exprEnd: () => Block,
108                                 exprFunEnd: () => Block, groupSalienceCheck: String => Block,
109                                 groups: Seq[Group]): TriggerResult = {
110     val map = expressions.map(p => p._1.index -> p._2).toMap
111 
112     val simpleGrouper = DefaultTriggerGrouper()
113 
114     //System.out.println(s"the groups had ${groups.size} entries")
115 
116     val groupExprs = groups.map(_.groupFilter)
117 
118     def builder = {
119 
120       val grouped = groups.zipWithIndex.grouped(runner.variablesPerFunc).grouped(runner.variableFuncGroup).toSeq
121 
122       val funPairs =
123         for (exprGroup <- grouped) yield {
124           val groupName = ctx.freshName(prefix + "GEGroup")
125 
126           val subClasses =
127             for {
128               exprFunc <- exprGroup
129             } yield {
130               val exprFuncName = ctx.freshName(prefix + "GEFuncGroup")
131 
132               val groupCalls =
133                 exprFunc.map {
134                   case (group, groupIndex) =>
135 
136                     val id = s"Group$groupIndex"
137 
138                     val allGroupExprs = group.triggers.flatMap {
139                       trigger =>
140                         Seq(trigger.expression) ++ trigger.outputExpression.map(Seq(_)).getOrElse(Seq.empty)
141                     }
142 
143                     // remove the params usage, everything is in the object variables, this is top level only
144                     val preCalcParams = (ctx: CodegenContext) =>
145                       genParamsForNested(ctx, allGroupExprs, additionalParams).copy(paramsDef = "", paramsCall = "")
146 
147                     val resCode = ExprCode(VariableValue(ctx.freshName("groupResultNull"), java.lang.Boolean.TYPE),
148                       VariableValue(ctx.freshName("groupResult"), classOf[GenericInternalRow]))
149                     val (body, expr) =
150                       SeparateCompilation.withSubExpressions(runner, allGroupExprs, ctx, resCode,
151                         SubCompilation(id, s"Trigger group $groupIndex"),
152                         // we need to pipe the row in
153                         extraParams = additionalParams,
154                         useParams = preCalcParams
155                       ) { (ctx, index, params) =>
156                         // group the group, params holds any subexprs used/generated for this sub compilation
157 
158                         val grpResult = ctx.freshName("groupResult")
159 
160                         val sgr = simpleGrouper(ctx, runner, grpResult, additionalParams,
161                           group.triggers.map(t => (t, map(t.index))), params, prefix, exprEnd, exprFunEnd, groupSalienceCheck)
162 
163                         val funNames = sgr.groupCalls
164                         val exprRunner =
165                           ExprCode(VariableValue(ctx.freshName("groupResultNull"), java.lang.Boolean.TYPE),
166                             VariableValue(grpResult, classOf[GenericInternalRow])
167                           )
168 
169                         // the top level is 0 arrays are filtered out from the row as they don't need explicit returning
170                         GenerateResult(SeparateClassGenerator(runner.getClass.getName, additionalParams, groupIndex + 1), exprRunner.copy(
171                           code =
172                             code"""
173                               boolean ${exprRunner.isNull} = false;
174                               ${funNames.map { f => s"$f(${params.paramsCall});" }.mkString("\n")}
175                               GenericInternalRow ${exprRunner.value} = new org.apache.spark.sql.catalyst.expressions.GenericInternalRow(
176                                 new Object[]{
177                                 ${additionalParams.filterNot(_.javaType.isArray).map(_.variableName).mkString(",\n")}
178                                 }
179                               );
180                               """
181                         ), sgr.extraClasses, sgr.ignoreTopLevelSubExpressions)
182                       }
183 
184                     val eval = group.groupFilter.genCode(ctx)
185                     /*
186                     if (groupIndex > 40 && groupIndex < 45) {
187                       System.out.println(s"""${s"$id - Size ${group.triggers.size} filter is ${group.groupFilter.toString.replaceAll("\n","")}"}""")
188                     }
189                     */
190                     // if ruleEngine is used salience may need comparison, if it's expression or dq any comparison is meaningless
191                     (
192                       code"""
193                         ${exprEnd()}\n
194                         ${eval.code}
195                         if ((!${eval.isNull}) && ${eval.value} ${groupSalienceCheck(group.lowestSalience.toString)} ) {
196                           ${expr.code}
197                         }
198                       """, body)
199                 }
200 
201               val argPairs = params.nonCombinedParams.map(t => t._1 -> t._2)
202               val body =
203                 QualityCodeGenUtils.splitExpressions(ctx, groupCalls.map(_._1.code + s"${exprEnd()}\n"), exprFuncName, argPairs,
204                   foldFunctions = _.mkString(s"${exprEnd()}\n", s";\n${exprEnd()}\n", ";")
205                 )
206               //${groupCalls.map(_._1).mkString(s"\n")}
207               (ctx.addNewFunction(exprFuncName,
208                 code"""
209                  private void $exprFuncName(${params.paramsDef}) {
210                    $body
211                  }
212                 """.code
213               ), groupCalls.map(_._2))
214             }
215 
216           (ctx.addNewFunction(groupName,
217 
218             code"""
219              private void $groupName(${params.paramsDef}) {
220                ${exprFunEnd()}
221                ${subClasses.map { f => s"${f._1}(${params.paramsCall});" }.mkString(s"${exprFunEnd()}\n")}
222              }
223              """.code
224 
225           ), subClasses.flatMap(_._2))
226         }
227       funPairs
228     }.foldLeft((Seq.empty[String], Seq.empty[CodeAndComment])) {
229       case ((ns, cs), (n, c)) => (ns :+ n, cs ++ c.flatten)
230     }
231 
232     if (ctx.currentVars eq null) {
233       // only fails on "via ProcessFactory with Avro inputs" RowToRowTest shows it doesn't always work for projections
234 
235       val subExpressionCode = QualityCodeGenUtils.nonWholeStageSubexpressionElimination(ctx, groupExprs)
236 
237       val (funNames, extraClasses) = builder
238       TriggerResult(funNames.iterator, subExpressionCode, extraClasses, true)
239     } else {
240       // will generate again, the sub exprs will be present on the projection unless ZeroCodeGen is enabled
241       val subExprs = SubExprCodeGen.subexpressionEliminationForWholeStageCodegen(ctx, groupExprs ++
242         ShimExprUtils.currentSubExprState(ctx).map(s => ShimExprUtils.fromState(s._1))
243       )
244       val subExpressionCode = ShimExprUtils.evaluateSubExprEliminationState(ctx, subExprs)
245 
246       val (funNames, extraClasses) =
247         QualityCodeGenUtils.withSubExprEliminationExprs(ctx, subExprs.states) {
248           builder
249         }
250       TriggerResult(funNames.iterator, SeparateCompilation.splitGlobalSubExprs(ctx, subExpressionCode), extraClasses, true)
251     }
252   }
253 
254 }
