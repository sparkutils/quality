121         for (exprGroup <- grouped) yield {

178                               """

208                    $body
209                  }

246           builder
247         }
248       TriggerResult(funNames.iterator, SeparateCompilation.splitGlobalSubExprs(ctx, subExpressionCode), extraClasses, true)
249     }
250   }
