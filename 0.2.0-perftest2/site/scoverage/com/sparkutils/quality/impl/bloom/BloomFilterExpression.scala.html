47   /**
48    * Identifies blooms from either a resolved or unresolved expression
49    *
50    * @param expression a single Expression, can be a complex tree or simple direct probabilityIn or rowid
51    * @return The bloom id's used, for unresolved expression trees this may contain blooms which are not present in the bloom map
52    */
53   @ClassicOnly
54   def getBlooms(expression: Expression): Seq[String] = {
55     val res = Seq.empty
56 
57     def bloom(expression: Expression, acc: Seq[String]): Seq[String] = {
58 
59       // case one
60       expression match {
61         case UnresolvedFunction4(name, Seq(left, Literal(id: UTF8String, StringType)), _, _) if name.replaceAll("_", "").toLowerCase == "probabilityin" => bloom(left, acc :+ id.toString)
62         case BloomFilterLookupExpression(left, Literal(id: UTF8String, StringType), _) => bloom(left, acc :+ id.toString)
63         case e: Expression => e.children.foldLeft(acc) { (acc, e) => bloom(e, acc) }
64         case _ => acc
65       }
66     }
67 
68     bloom(expression, res)
69   }
