30 /**
31  * Registers Quality sql functions using the defaults for registerQualityFunctions, these can be overridden without having to subclass DriverPlugin.  Functions are registered via FunctionRegistry.builtIn making them available for view creation, if they should be registered via extensions only then use the quality_force_inject_function = true configuration.
32  *
33  * It also registers plan optimiser rule's such as AsUUIDFilter, which rewrites filters with variables backed by as_uuid.
34  *
35  * Optimiser rules can be disabled via the quality_disable_optimiser_rules system environment variable. "*" disables all rules, otherwise a comma separated list of fqn class names may be used.
36  */
37 class QualitySparkExtension extends ((SparkSessionExtensions) => Unit) with Logging {
38 
39   def parseTypes: String => Option[DataType] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultParseTypes _
40   def zero: DataType => Option[Any] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultZero _
41   def add: DataType => Option[(Expression, Expression) => Expression] = (dataType: DataType) => com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultAdd(dataType)
42   def mapCompare: DataType => Option[(Any, Any) => Int] = (dataType: DataType) => utils.defaultMapCompare(dataType)
43   def writer: String => Unit = println(_)

45   /**

47    * @param str
48    * @param prefix defaults to "Quality SparkExtensions"

59   }

61   /**
62    * Adds AsymmetricFilterExpressions for AsUUID
63    * Derived implementations should also call super if they wish to register the default rules (or specify both as extensions).
64    * These are registered after resolution is done.
65    * The first String should be the fqn classname / rule name for the rule
66    * @return
67    */
68   def optimiserRules: Seq[(String, SparkSession => Rule[LogicalPlan])] =
69     Seq((AsUUIDFilter.getClass.getName, _ => AsUUIDFilter), (IDBase64Filter.getClass.getName, _ => IDBase64Filter),
70       (FunNRewrite.getClass.getName, _ => FunNRewrite))
71 
72   override def apply(extensions: SparkSessionExtensions): Unit = {
73     val func =
74       if (com.sparkutils.quality.getConfig(forceInjectFunction, "false").toBoolean) {
75         attemptLogInfo("registering quality functions via injection - they are classed as temporary functions")
76         ShimUtils.registerFunctionViaExtension(extensions) _
77       } else {
78         attemptLogInfo("registering quality functions via builtin function registry - whilst you can use these in global views the extension must always be present")
79         ShimUtils.registerFunctionViaBuiltin _
80       }
81     com.sparkutils.quality.registerQualityFunctions(parseTypes, zero, add, mapCompare, writer,
82       registerFunction = func
83     )
84 
85     if (Testing.testing) {
86       ExtensionTesting.disableRuleResult = ""
87     }
88     val (all, disabledRules) = disabledOptimiserRules()
89     if (!all) {
90       val filteredRules = optimiserRules.filterNot(p => disabledRules.contains(p._1.trim))
91       val str = s"$disableRulesConf = $disabledRules leaving ${filteredRules.map(_._1)} remaining"
92       attemptLogInfo(str)
93       if (Testing.testing) {
94         ExtensionTesting.disableRuleResult = str
95       }
96       filteredRules.map(_._2).foreach(extensions.injectOptimizerRule _)
97     } else {
98       val str = s"All optimiser rules are disabled via $disableRulesConf"
99       attemptLogInfo(str)
100     }
101   }
102 
103 }
