202 trait ResultStatistics[T <: ResultStatistics[_]] extends Serializable {
203   def failed: Long
204   def passed: Long
205   def softFailed: Long
206   def disabled: Long
207   def ignored: Long
208   def defaulted: Long
209   def probabilityPassed: Long
210   def probabilityFailed: Long
211 
212   def update(failed: Long = failed, passed: Long = passed, softFailed: Long = softFailed, disabled: Long = disabled,
213              ignored: Long = ignored, defaulted: Long = defaulted, probabilityPassed: Long = probabilityPassed,
214              probabilityFailed: Long = probabilityFailed): T
215 
216   def combineResults(other: T): T =
217     update(failed = failed + other.failed, passed = passed + other.passed, softFailed = softFailed + other.softFailed,
218       disabled = disabled + other.disabled, ignored = ignored + other.ignored, defaulted = defaulted + other.defaulted,
219       probabilityPassed = probabilityPassed + other.probabilityPassed,
220       probabilityFailed = probabilityFailed + other.probabilityFailed)
221 
222   def combine(other: T): T
223 
224   @tailrec
225   final def processResult(ruleResult: RuleResult, probabilityPass: Double = 0.8d): T =
226     ruleResult match {
227       case Failed => update(failed = failed + 1)
228       case Passed => update(passed = passed + 1)
229       case SoftFailed => update(softFailed = softFailed + 1)
230       case DisabledRule => update(disabled = disabled + 1)
231       case IgnoredRule => update(ignored = ignored + 1)
232       case DefaultRule => update(defaulted = defaulted + 1)
233       case Probability(percentage) if (percentage >= probabilityPass) => update(probabilityPassed = probabilityPassed + 1)
234       case Probability(_) => update(probabilityFailed = probabilityFailed + 1)
235       case RuleResultWithProcessor(ruleResult: RuleResult, _) =>
236         processResult(ruleResult, probabilityPass)
237     }

239 }
