246   protected[quality] case class SameOrNextVersionLower[T](map: Map[Id, Seq[T]]) extends Function1[Id, Option[Seq[T]]] {
247     implicit val ordering: Ordering[Id] =
248       (x: Id, y: Id) => {
249         if (x == y)
250           0
251         else
252           if (x.id != y.id)
253             x.id - y.id
254           else
255             x.version - y.version
256       }
257 
258     val sorted = TreeMap[Id, Seq[T]]() ++ map
259 
260     override def apply(id: Id): Option[Seq[T]] =
261       sorted.to(id).lastOption.map(_._2)
262   }
