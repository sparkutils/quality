129   /**
130    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
131    * copied before codegen.
132    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.
133    */
134   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
135     val expressions = ShimUtils.copyStateful(dirtyExpressions)
136     import ctx._
137     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
138 
139     // Add each expression tree and compute the common subexpressions.
140     expressions.foreach(equivalentExpressions.addExprTree(_))
141     var subexprFunctions = ""
142     // Get all the expressions that appear at least twice and set up the state for subexpression
143     // elimination.
144     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
145     commonExprs.foreach { expr =>
146       val fnName = freshName("subExpr")
147       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
148       val value = addMutableState(javaType(expr.dataType), "subExprValue")
149 
150       // Generate the code for this expression tree and wrap it in a function.
151       val eval = expr.genCode(ctx)
152       val fn =
153         s"""
154            |private void $fnName(InternalRow $INPUT_ROW) {
155            |  ${eval.code}
156            |  $isNull = ${eval.isNull};
157            |  $value = ${eval.value};
158            |}
159            """.stripMargin
160 
161       // Add a state and a mapping of the common subexpressions that are associate with this
162       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
163       // when it is code generated. This decision should be a cost based one.
164       //
165       // The cost of doing subexpression elimination is:
166       //   1. Extra function call, although this is probably *good* as the JIT can decide to
167       //      inline or not.
168       // The benefit doing subexpression elimination is:
169       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
170       //      above.
171       //   2. Less code.
172       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
173       // at least two nodes) as the cost of doing it is expected to be low.
174 
175       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
176       subexprFunctions += subExprCode
177       val state = ShimExprUtils.state(
178         ExprCode(code"$subExprCode",
179           JavaCode.isNullGlobal(isNull),
180           JavaCode.global(value, expr.dataType)))
181       ShimExprUtils.addSubExpr(ctx, expr, state)
182     }
183 
184     subexprFunctions
185   }

202   /**
203    * Splits the generated code of expressions into multiple functions, because function has
204    * 64kb code size limit in JVM. If the class to which the function would be inlined would grow
205    * beyond 1000kb, we declare a private, inner sub-class, and the function is inlined to it
206    * instead, because classes have a constant pool limit of 65,536 named values.
207    *
208    * @param expressions the codes to evaluate expressions.
209    * @param funcName the split function name base.
210    * @param arguments the list of (type, name) of the arguments of the split function.
211    * @param returnType the return type of the split function.
212    * @param makeSplitFunction makes split function body, e.g. add preparation or cleanup.
213    * @param foldFunctions folds the split function calls.
214    */
215   def splitExpressions(ctx: CodegenContext,
216                        expressions: Seq[String],
217                        groupingSize: Int,
218                        funcName: String,
219                        arguments: Seq[(String, String)],
220                        returnType: String = "void",
221                        makeSplitFunction: String => String = identity,
222                        foldFunctions: Seq[String] => String = _.mkString("", ";\n", ";")): String = {
223     val blocks = buildCodeBlocks(expressions.grouped(groupingSize).map(_.mkString("\n")).toSeq)
224 
225     if (blocks.length == 1) {
226       // inline execution if only one block
227       blocks.head
228     } else {
229       val func = ctx.freshName(funcName)
230       val argString = arguments.map { case (t, name) => s"$t $name" }.mkString(", ")
231       val functions = blocks.zipWithIndex.map { case (body, i) =>
232         val name = s"${func}_$i"
233         val code = s"""
234                       |private $returnType $name($argString) {
235                       |  ${makeSplitFunction(body)}
236                       |}
237          """.stripMargin
238         addNewFunctionInternal.invoke(ctx, name, code, java.lang.Boolean.FALSE).asInstanceOf[NewFunctionSpec]
239       }
240 
241       val (outerClassFunctions, innerClassFunctions) = functions.partition(_.innerClassName.isEmpty)
242 
243       val argsString = arguments.map(_._2).mkString(", ")
244       val outerClassFunctionCalls = outerClassFunctions.map(f => s"${f.functionName}($argsString)")
245 
246       val innerClassFunctionCalls = generateInnerClassesFunctionCall.invoke(ctx,
247         innerClassFunctions,
248         func,
249         arguments,
250         returnType,
251         makeSplitFunction,
252         foldFunctions).asInstanceOf[Iterable[String]]
253 
254       foldFunctions(outerClassFunctionCalls ++ innerClassFunctionCalls)
255     }
256   }

259   /**
260    * Splits the generated code of expressions into multiple sequences of String
261    * based on a threshold of length of a String
262    *
263    * @param expressions the codes to evaluate expressions.
264    */
265   private def buildCodeBlocks(expressions: Seq[String]): Seq[String] = {
266     val blocks = new ArrayBuffer[String]()
267     val blockBuilder = new StringBuilder()
268     var length = 0
269     val splitThreshold = SQLConf.get.methodSplitThreshold
270     for (code <- expressions) {
271       // We can't know how many bytecode will be generated, so use the length of source code
272       // as metric. A method should not go beyond 8K, otherwise it will not be JITted, should
273       // also not be too small, or it will have many function calls (for wide table), see the
274       // results in BenchmarkWideTable.
275       if (length > splitThreshold) {
276         blocks += blockBuilder.toString()
277         blockBuilder.clear()
278         length = 0
279       }
280       blockBuilder.append(code)
281       length += code.length//CodeFormatter.stripExtraNewLinesAndComments(code).length
282     }
283     blocks += blockBuilder.toString()
284     blocks.toSeq
285   }
