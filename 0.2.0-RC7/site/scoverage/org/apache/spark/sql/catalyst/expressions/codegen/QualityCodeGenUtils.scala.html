94    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are

98   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
99     val expressions = ShimUtils.copyStateful(dirtyExpressions)
100     import ctx._
101     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
102 
103     // Add each expression tree and compute the common subexpressions.
104     expressions.foreach(equivalentExpressions.addExprTree(_))
105     var subexprFunctions = ""
106     // Get all the expressions that appear at least twice and set up the state for subexpression
107     // elimination.
108     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
109     commonExprs.foreach { expr =>
110       val fnName = freshName("subExpr")
111       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
112       val value = addMutableState(javaType(expr.dataType), "subExprValue")
113 
114       // Generate the code for this expression tree and wrap it in a function.
115       val eval = expr.genCode(ctx)
116       val fn =
117         s"""
118            |private void $fnName(InternalRow $INPUT_ROW) {
119            |  ${eval.code}
120            |  $isNull = ${eval.isNull};
121            |  $value = ${eval.value};
122            |}
123            """.stripMargin
124 
125       // Add a state and a mapping of the common subexpressions that are associate with this
126       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
127       // when it is code generated. This decision should be a cost based one.
128       //
129       // The cost of doing subexpression elimination is:
130       //   1. Extra function call, although this is probably *good* as the JIT can decide to
131       //      inline or not.
132       // The benefit doing subexpression elimination is:
133       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
134       //      above.
135       //   2. Less code.
136       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
137       // at least two nodes) as the cost of doing it is expected to be low.
138 
139       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
140       subexprFunctions += subExprCode
141       val state = ShimExprUtils.state(
142         ExprCode(code"$subExprCode",
143           JavaCode.isNullGlobal(isNull),
144           JavaCode.global(value, expr.dataType)))
145       ShimExprUtils.addSubExpr(ctx, expr, state)
146     }
147 
148     subexprFunctions
149   }

166   /**
167    * Splits the generated code of expressions into multiple functions, because function has
168    * 64kb code size limit in JVM. If the class to which the function would be inlined would grow
169    * beyond 1000kb, we declare a private, inner sub-class, and the function is inlined to it
170    * instead, because classes have a constant pool limit of 65,536 named values.
171    *
172    * @param expressions the codes to evaluate expressions.
173    * @param funcName the split function name base.
174    * @param arguments the list of (type, name) of the arguments of the split function.
175    * @param returnType the return type of the split function.
176    * @param makeSplitFunction makes split function body, e.g. add preparation or cleanup.
177    * @param foldFunctions folds the split function calls.
178    */
179   def splitExpressions(ctx: CodegenContext,
180                        expressions: Seq[String],
181                        groupingSize: Int,
182                        funcName: String,
183                        arguments: Seq[(String, String)],
184                        returnType: String = "void",
185                        makeSplitFunction: String => String = identity,
186                        foldFunctions: Seq[String] => String = _.mkString("", ";\n", ";")): String = {
187     val blocks = buildCodeBlocks(expressions.grouped(groupingSize).map(_.mkString("\n")).toSeq)
188 
189     if (blocks.length == 1) {
190       // inline execution if only one block
191       blocks.head
192     } else {
193       val func = ctx.freshName(funcName)
194       val argString = arguments.map { case (t, name) => s"$t $name" }.mkString(", ")
195       val functions = blocks.zipWithIndex.map { case (body, i) =>
196         val name = s"${func}_$i"
197         val code = s"""
198                       |private $returnType $name($argString) {
199                       |  ${makeSplitFunction(body)}
200                       |}
201          """.stripMargin
202         addNewFunctionInternal.invoke(ctx, name, code, java.lang.Boolean.FALSE).asInstanceOf[NewFunctionSpec]
203       }
204 
205       val (outerClassFunctions, innerClassFunctions) = functions.partition(_.innerClassName.isEmpty)
206 
207       val argsString = arguments.map(_._2).mkString(", ")
208       val outerClassFunctionCalls = outerClassFunctions.map(f => s"${f.functionName}($argsString)")
209 
210       val innerClassFunctionCalls = generateInnerClassesFunctionCall.invoke(ctx,
211         innerClassFunctions,
212         func,
213         arguments,
214         returnType,
215         makeSplitFunction,
216         foldFunctions).asInstanceOf[Iterable[String]]
217 
218       foldFunctions(outerClassFunctionCalls ++ innerClassFunctionCalls)
219     }
220   }

223   /**
224    * Splits the generated code of expressions into multiple sequences of String
225    * based on a threshold of length of a String
226    *
227    * @param expressions the codes to evaluate expressions.
228    */
229   private def buildCodeBlocks(expressions: Seq[String]): Seq[String] = {
230     val blocks = new ArrayBuffer[String]()
231     val blockBuilder = new StringBuilder()
232     var length = 0
233     val splitThreshold = SQLConf.get.methodSplitThreshold
234     for (code <- expressions) {
235       // We can't know how many bytecode will be generated, so use the length of source code
236       // as metric. A method should not go beyond 8K, otherwise it will not be JITted, should
237       // also not be too small, or it will have many function calls (for wide table), see the
238       // results in BenchmarkWideTable.
239       if (length > splitThreshold) {
240         blocks += blockBuilder.toString()
241         blockBuilder.clear()
242         length = 0
243       }
244       blockBuilder.append(code)
245       length += code.length//CodeFormatter.stripExtraNewLinesAndComments(code).length
246     }
247     blocks += blockBuilder.toString()
248     blocks.toSeq
249   }
