85   private def generateSwitch(ctx: CodegenContext, prefix: String, groupDepth: Int, params: ParameterInformation,
86                              groupCalls: Seq[(String, Block)]) = {

95           code"""

98            }

110     def buildSwitch(switchVal: String, cases: Seq[(String,Block)], default: String, single: Boolean): String = {
111       val folded = cases.foldLeft(code"") {
112         case (cur, n) =>
113           code"""
114           $cur
115           ${n._2}
116         """
117       }
118 
119       val switchIf: String => String =
120         if (single)
121           identity
122         else
123         (cases.headOption.map( v => greaterThanOrEqual(switchVal, v._1)),
124           cases.lastOption.map( v => lessThanOrEqual(switchVal, v._1))) match {
125           case (Some(min), Some(max)) =>
126             s => s"""
127                 if (($min) && ($max)) {
128                   $s
129                 } else {
130                   $default
131                 }
132               """
133           case (Some(min), None) =>
134             s => s"""
135                 if ($min) {
136                   $s
137                 } else {
138                   $default
139                 }
140               """
141           case (None, Some(max)) =>
142             s => s"""
143                 if ($max) {
144                   $s
145                 } else {
146                   $default
147                 }
148               """
149         }
150 
151 
152       switchIf(s"""
153         switch($switchVal) {
154           $folded
155         }
156          """)
157     }

159     def buildSwitches(switchVal: String, chunked: Seq[Seq[(String,Block)]]): String = {
160       if (chunked.size == 1) {
161         buildSwitch(switchVal, chunked.head, "", true)
162       } else {
163         // we have more chunks
164         val head = chunked.head
165         val (restLeft, restRight) = chunked.tail.splitAt((chunked.size / 2) - 1)
166 
167         def headVal(chunked: Seq[Seq[(String,Block)]]): String =
168           (for {
169             first <- chunked.headOption
170             second <- first.headOption
171           } yield second._1).getOrElse(zero)
172 
173         val (restLeftStartBucket, restRightStartBucket) = (headVal(restLeft), headVal(restRight))
174 
175         def switches(chunked: Seq[Seq[(String,Block)]]): Option[String] =
176           if (chunked.isEmpty)
177             None
178           else
179             Some(buildSwitches(switchVal, chunked))
180 
181         val (switchLeft, switchRight) = (switches(restLeft), switches(restRight))
182 
183         def switch(switch: String, lOrR: String) = {
184           val exprFuncName = ctx.freshName(prefix + s"GEFuncSwitchGroupsNested${lOrR}_" + groupDepth)
185           val callFun = ctx.addNewFunction(exprFuncName,
186             code"""
187              private void $exprFuncName($typ $switchVal, ${params.paramsDef}) {
188                $switch
189              }
190             """.code
191           )
192           callFun
193         }
194 
195         val (leftFun, rightFun) = (switchLeft.map(switch(_, "L")), switchRight.map(switch(_, "R")))
196 
197         val (geLeft, lessThanRight) =
198           (greaterThanOrEqual(switchVal, restLeftStartBucket), lessThan(switchVal, restRightStartBucket))
199 
200         val defalt =
201           (leftFun, rightFun) match {
202             case (Some(leftFun), Some(rightFun)) =>
203               s"""
204                 if (($geLeft) && ($lessThanRight)) {
205                   $leftFun($switchVal, ${params.paramsCall});
206                 } else {
207                   $rightFun($switchVal, ${params.paramsCall});
208                 }
209               """
210             case (Some(leftFun), None) =>
211               s"""
212                 if ($geLeft) {
213                   $leftFun($switchVal, ${params.paramsCall});
214                 }
215               """
216             case (None, Some(rightFun)) =>
217               s"""
218                 if (${greaterThanOrEqual(switchVal, restRightStartBucket)}) {
219                   $rightFun($switchVal, ${params.paramsCall});
220                 }
221               """
222           }
223 
224         val r = buildSwitch(switchVal, head, defalt, false)
225 
226         r
227       }
228     }
229 
230     val minSpark = ctx.addMutableState(sparkType, "minSpark", initFunc = initSpark(_,min))
231     val maxSpark = ctx.addMutableState(sparkType, "maxSpark", initFunc = initSpark(_,max))
232 
233     val converted = conversion(expr.value)
234     val (converting, switchName) =
235       if (converted.isEmpty || converted == expr.value.code)
236         ("", expr.value.code)
237       else {
238         val fresh = ctx.freshName("converted")
239         (s"$typ $fresh = $converted;", fresh)
240       }
241 
242     val groupSize = 200
243     val rangeCheck =
244       if (cases.size > groupSize)
245         s"|| (${lessThanSpark(expr.value, minSpark)}) || (${greaterThanSpark(expr.value, maxSpark)})"
246       else
247         ""
248 
249     ctx.addNewFunction(exprFuncName,
250       code"""
251        private void $exprFuncName(${params.paramsDef}) {
252          ${expr.code}
253          if (${expr.isNull} $rangeCheck) {
254          } else {
255            $converting
256 
257            ${
258             buildSwitches(switchName, /// TODO uses Spark mechanism to group, this is for PoC
259               cases.grouped(groupSize).toSeq)
260           }
261          }
262        }
263       """.code
264     )
265   }
