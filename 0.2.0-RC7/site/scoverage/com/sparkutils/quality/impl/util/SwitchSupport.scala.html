72  * Produces grouping via a switch, unlike normal grouping the group itself is the separate compilation and
73  * each sub SwitchGroup is directly inlined if there is only one trigger per group

103   private def generateSwitch(ctx: CodegenContext, prefix: String, groupDepth: Int, params: ParameterInformation,
104                              groupCalls: Seq[(String, Block)]) = {
105     val exprFuncName = ctx.freshName(prefix + "GEFuncSwitchGroups" + groupDepth)
106 
107     val (min, max) = (groupCalls.head._1, groupCalls.last._1)
108 
109     val cases = groupCalls.map {
110       case (bucket, codeToRun) =>
111         val exprFuncName = ctx.freshName(prefix + "callGEFuncSwitchGroup" + groupDepth)
112         val callFun = ctx.addNewFunction(exprFuncName,
113           code"""
114            private void $exprFuncName(${params.paramsDef}) {
115              $codeToRun
116            }
117           """.code
118         )
119         bucket -> code"""
120           case $bucket:
121             $callFun(${params.paramsCall});
122             break;
123         """
124     }
125 
126     val expr = groupingExpression.genCode(ctx)
127 
128     def buildSwitch(switchVal: String, cases: Seq[(String,Block)], default: String, single: Boolean): String = {
129       val folded = cases.foldLeft(code"") {
130         case (cur, n) =>
131           code"""
132           $cur
133           ${n._2}
134         """
135       }
136 
137       val switchIf: String => String =
138         if (single)
139           identity
140         else
141         (cases.headOption.map( v => greaterThanOrEqual(switchVal, v._1)),
142           cases.lastOption.map( v => lessThanOrEqual(switchVal, v._1))) match {
143           case (Some(min), Some(max)) =>
144             s => s"""
145                 if (($min) && ($max)) {
146                   $s
147                 } else {
148                   $default
149                 }
150               """
151           case (Some(min), None) =>
152             s => s"""
153                 if ($min) {
154                   $s
155                 } else {
156                   $default
157                 }
158               """
159           case (None, Some(max)) =>
160             s => s"""
161                 if ($max) {
162                   $s
163                 } else {
164                   $default
165                 }
166               """
167         }
168 
169 
170       switchIf(s"""
171         switch($switchVal) {
172           $folded
173         }
174          """)
175     }
176 
177     def buildSwitches(switchVal: String, chunked: Seq[Seq[(String,Block)]]): String = {
178       if (chunked.size == 1) {
179         buildSwitch(switchVal, chunked.head, "", true)
180       } else {
181         // we have more chunks
182         val head = chunked.head
183         val (restLeft, restRight) = chunked.tail.splitAt((chunked.size / 2) - 1)
184 
185         def headVal(chunked: Seq[Seq[(String,Block)]]): String =
186           (for {
187             first <- chunked.headOption
188             second <- first.headOption
189           } yield second._1).getOrElse(zero)
190 
191         val (restLeftStartBucket, restRightStartBucket) = (headVal(restLeft), headVal(restRight))
192 
193         def switches(chunked: Seq[Seq[(String,Block)]]): Option[String] =
194           if (chunked.isEmpty)
195             None
196           else
197             Some(buildSwitches(switchVal, chunked))
198 
199         val (switchLeft, switchRight) = (switches(restLeft), switches(restRight))
200 
201         val (paramsDef, paramsCall) = {
202           val switchTypeVal = s"$typ $switchVal"
203           if (params.paramsDef.contains(switchTypeVal))
204             (params.paramsDef, params.paramsCall) // for example if it's directly an inputadapter_value e.g. primitive (booleangroupertest)
205           else
206             (s"$switchTypeVal, ${params.paramsDef}", s"$switchVal, ${params.paramsCall}")
207         }
208 
209         def switch(switch: String, lOrR: String) = {
210           val exprFuncName = ctx.freshName(prefix + s"GEFuncSwitchGroupsNested${lOrR}_" + groupDepth)
211 
212           val callFun = ctx.addNewFunction(exprFuncName,
213             code"""
214              private void $exprFuncName($paramsDef) {
215                $switch
216              }
217             """.code
218           )
219           callFun
220         }
221 
222         val (leftFun, rightFun) = (switchLeft.map(switch(_, "L")), switchRight.map(switch(_, "R")))
223 
224         val (geLeft, lessThanRight) =
225           (greaterThanOrEqual(switchVal, restLeftStartBucket), lessThan(switchVal, restRightStartBucket))
226 
227         val defalt =
228           (leftFun, rightFun) match {
229             case (Some(leftFun), Some(rightFun)) =>
230               s"""
231                 if (($geLeft) && ($lessThanRight)) {
232                   $leftFun($paramsCall);
233                 } else {
234                   $rightFun($paramsCall);
235                 }
236               """
237             case (Some(leftFun), None) =>
238               s"""
239                 if ($geLeft) {
240                   $leftFun($paramsCall);
241                 }
242               """
243             case (None, Some(rightFun)) =>
244               s"""
245                 if (${greaterThanOrEqual(switchVal, restRightStartBucket)}) {
246                   $rightFun($paramsCall);
247                 }
248               """
249           }
250 
251         val r = buildSwitch(switchVal, head, defalt, false)
252 
253         r
254       }
255     }
256 
257     val minSpark = ctx.addMutableState(sparkType, "minSpark", initFunc = initSpark(_,min))
258     val maxSpark = ctx.addMutableState(sparkType, "maxSpark", initFunc = initSpark(_,max))
259 
260     val converted = conversion(expr.value)
261     val (converting, switchName) =
262       if (converted.isEmpty || converted == expr.value.code)
263         ("", expr.value.code)
264       else {
265         val fresh = ctx.freshName("converted")
266         (s"$typ $fresh = $converted;", fresh)
267       }
268 
269     val groupSize = 200
270     val rangeCheck =
271       if (cases.size > groupSize)
272         s"|| (${lessThanSpark(expr.value, minSpark)}) || (${greaterThanSpark(expr.value, maxSpark)})"
273       else
274         ""
275 
276     ctx.addNewFunction(exprFuncName,
277       code"""
278        private void $exprFuncName(${params.paramsDef}) {
279          ${expr.code}
280          if (${expr.isNull} $rangeCheck) {
281          } else {
282            $converting
283 
284            ${
285             buildSwitches(switchName, /// TODO uses Spark mechanism to group, this is for PoC
286               cases.grouped(groupSize).toSeq)
287           }
288          }
289        }
290       """.code
291     )
292   }
