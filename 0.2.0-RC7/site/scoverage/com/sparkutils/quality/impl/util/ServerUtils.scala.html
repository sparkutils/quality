347   def stripBrackets(v: VariableValue): (String, String) = {
348     val openb = v.toString().indexOf("[")
349     if (openb == -1)
350       (v.variableName, "")
351     else
352       (v.variableName.dropRight(v.length - openb), v.variableName.drop(openb))
353   }

355   def formatParams(ctx: CodegenContext, a: Seq[ExprValue], additional: Seq[VariableValue] = Seq.empty, callsKeepArrays: Boolean = false): ParameterInformation = {
356 
357     def filterOutArrays(use: Seq[ExprValue]) = use.flatMap {
358       case a: VariableValue => Some(a)
359       case _ => None
360     }
361 
362     val filteredA = filterOutArrays(a)
363     val filteredAdditional = filterOutArrays(additional)
364 
365     val size = filteredA.size + filteredAdditional.size
366     val use =
367       if (size <= 22)
368         filteredA ++ filteredAdditional
369       else
370         filteredA // additional are then handled via class level
371 
372     // filter out any top level arrays, the input is a set, so params need the same order
373     val ordered = use
374 
375     def prepFields(ordered: Seq[VariableValue]) =
376       ordered.map { v =>
377         val (stripped, arrayInName) = stripBrackets(v)
378 
379         val (typ, array) =
380           if (v.javaType.isArray)
381             (s"${v.javaType.getComponentType.getName}", "[]")
382           else if (v.javaType.isPrimitive)
383             (v.javaType.toString, arrayInName.replaceAll("[^\\[\\]]",""))
384           else
385             (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
386 
387         (s"$typ$array", stripped, v.javaType, array.nonEmpty) // if it's not empty we want to pass through
388       }.distinct
389 
390     val pairs = prepFields(ordered)
391 
392     val combined =
393       if (size <= 22)
394         pairs
395       else
396         pairs ++ prepFields(filteredAdditional)
397 
398     val paramsCall =
399       ordered.map(v =>
400         if (v.javaType.isArray && callsKeepArrays)
401           v.variableName
402         else
403           stripBrackets(v)._1
404       ).distinct.mkString(", ")
405 
406     ParameterInformation(pairs.map {
407       case (typ, stripped, _, _) =>
408 
409         s"$typ $stripped"
410       }.distinct.mkString(", ")
411       , paramsCall, combined.size, combined, outerCallParams = paramsCall, nonCombinedParams = pairs)
