133   /**
134    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
135    * copied before codegen.
136    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.
137    */
138   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
139     val expressions = ShimUtils.copyStateful(dirtyExpressions)
140     import ctx._
141     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
142 
143     // Add each expression tree and compute the common subexpressions.
144     expressions.foreach(equivalentExpressions.addExprTree(_))
145     var subexprFunctions = ""
146     // Get all the expressions that appear at least twice and set up the state for subexpression
147     // elimination.
148     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
149     commonExprs.foreach { expr =>
150       val fnName = freshName("subExpr")
151       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
152       val value = addMutableState(javaType(expr.dataType), "subExprValue")
153 
154       // Generate the code for this expression tree and wrap it in a function.
155       val eval = expr.genCode(ctx)
156       val fn =
157         s"""
158            |private void $fnName(InternalRow $INPUT_ROW) {
159            |  ${eval.code}
160            |  $isNull = ${eval.isNull};
161            |  $value = ${eval.value};
162            |}
163            """.stripMargin
164 
165       // Add a state and a mapping of the common subexpressions that are associate with this
166       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
167       // when it is code generated. This decision should be a cost based one.
168       //
169       // The cost of doing subexpression elimination is:
170       //   1. Extra function call, although this is probably *good* as the JIT can decide to
171       //      inline or not.
172       // The benefit doing subexpression elimination is:
173       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
174       //      above.
175       //   2. Less code.
176       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
177       // at least two nodes) as the cost of doing it is expected to be low.
178 
179       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
180       subexprFunctions += subExprCode
181       val state = ShimExprUtils.state(
182         ExprCode(code"$subExprCode",
183           JavaCode.isNullGlobal(isNull),
184           JavaCode.global(value, expr.dataType)))
185       ShimExprUtils.addSubExpr(ctx, expr, state)
186     }
187 
188     subexprFunctions
189   }

206   /**
207    * Splits the generated code of expressions into multiple functions, because function has
208    * 64kb code size limit in JVM. If the class to which the function would be inlined would grow
209    * beyond 1000kb, we declare a private, inner sub-class, and the function is inlined to it
210    * instead, because classes have a constant pool limit of 65,536 named values.
211    *
212    * @param expressions the codes to evaluate expressions.
213    * @param funcName the split function name base.
214    * @param arguments the list of (type, name) of the arguments of the split function.
215    * @param returnType the return type of the split function.
216    * @param makeSplitFunction makes split function body, e.g. add preparation or cleanup.
217    * @param foldFunctions folds the split function calls.
218    */
219   def splitExpressions(ctx: CodegenContext,
220                        expressions: Seq[String],
221                        groupingSize: Int,
222                        funcName: String,
223                        arguments: Seq[(String, String)],
224                        returnType: String = "void",
225                        makeSplitFunction: String => String = identity,
226                        foldFunctions: Seq[String] => String = _.mkString("", ";\n", ";")): String = {
227     val blocks = buildCodeBlocks(expressions.grouped(groupingSize).map(_.mkString("\n")).toSeq)
228 
229     if (blocks.length == 1) {
230       // inline execution if only one block
231       blocks.head
232     } else {
233       val func = ctx.freshName(funcName)
234       val argString = arguments.map { case (t, name) => s"$t $name" }.mkString(", ")
235       val functions = blocks.zipWithIndex.map { case (body, i) =>
236         val name = s"${func}_$i"
237         val code = s"""
238                       |private $returnType $name($argString) {
239                       |  ${makeSplitFunction(body)}
240                       |}
241          """.stripMargin
242         addNewFunctionInternal.invoke(ctx, name, code, java.lang.Boolean.FALSE).asInstanceOf[NewFunctionSpec]
243       }
244 
245       val (outerClassFunctions, innerClassFunctions) = functions.partition(_.innerClassName.isEmpty)
246 
247       val argsString = arguments.map(_._2).mkString(", ")
248       val outerClassFunctionCalls = outerClassFunctions.map(f => s"${f.functionName}($argsString)")
249 
250       val innerClassFunctionCalls = generateInnerClassesFunctionCall.invoke(ctx,
251         innerClassFunctions,
252         func,
253         arguments,
254         returnType,
255         makeSplitFunction,
256         foldFunctions).asInstanceOf[Iterable[String]]
257 
258       foldFunctions(outerClassFunctionCalls ++ innerClassFunctionCalls)
259     }
260   }

263   /**
264    * Splits the generated code of expressions into multiple sequences of String
265    * based on a threshold of length of a String
266    *
267    * @param expressions the codes to evaluate expressions.
268    */
269   private def buildCodeBlocks(expressions: Seq[String]): Seq[String] = {
270     val blocks = new ArrayBuffer[String]()
271     val blockBuilder = new StringBuilder()
272     var length = 0
273     val splitThreshold = SQLConf.get.methodSplitThreshold
274     for (code <- expressions) {
275       // We can't know how many bytecode will be generated, so use the length of source code
276       // as metric. A method should not go beyond 8K, otherwise it will not be JITted, should
277       // also not be too small, or it will have many function calls (for wide table), see the
278       // results in BenchmarkWideTable.
279       if (length > splitThreshold) {
280         blocks += blockBuilder.toString()
281         blockBuilder.clear()
282         length = 0
283       }
284       blockBuilder.append(code)
285       length += code.length//CodeFormatter.stripExtraNewLinesAndComments(code).length
286     }
287     blocks += blockBuilder.toString()
288     blocks.toSeq
289   }

291   /**
292    * Manage project vs wholestage exec, lambdas and subexpr, the function f is responsible for using the results
293    *
294    * @param ctx
295    * @param fullChildren
296    * @param f children that have had FunNLambda.swap then swapBack applied, 2nd param is the subexpr 'pre' code, third
297    *          signifies currentVars present rather than itr
298    * @tparam R1
299    * @tparam R2
300    * @return
301    */
302   def produceCode[R1,R2](ctx: CodegenContext, fullChildren: Seq[Expression])
303                         (f: (Seq[Expression], String, Boolean) => (R1, R2)): (R1, R2) = {
304 
305     // replace all usedAsLambda FunNs to make sure they cannot be turned into subexprs
306     val lambdaSafeChildren = fullChildren.map(FunNLambda.swap)
307 
308     if (ctx.currentVars eq null) {
309       val subExpressionCode = QualityCodeGenUtils.nonWholeStageSubexpressionElimination(ctx, lambdaSafeChildren)
310 
311       // replace the original non-subExpr FunNs
312       val children = lambdaSafeChildren.map(FunNLambda.swapBack)
313 
314       f(children, subExpressionCode, false)
315     } else {
316       val subExprs = SubExprCodeGen.subexpressionEliminationForWholeStageCodegen(ctx, lambdaSafeChildren)
317       val subExpressionCode = ShimExprUtils.evaluateSubExprEliminationState(ctx, subExprs)
318 
319       // replace the original non-subExpr FunNs
320       val children = lambdaSafeChildren.map(FunNLambda.swapBack)
321 
322       QualityCodeGenUtils.withSubExprEliminationExprs(ctx, subExprs.states) {
323         f(children, subExpressionCode, true)
324       }
325     }
326   }
