215    * Supplying a ddlType triggers the output type for the expression to be that ddl type, rather than using yaml conversion.
216    * @param renderOptions provides rendering options to the underlying snake yaml implementation
217    * @param ddlType optional DDL string, when present yaml output is disabled and the output expressions must all have the same type
218    * @param name the default column name "expressionResults"
219    * @param stripDDL the default of false leaves the DDL present, true strips ddl from the output using strip_result_ddl
220    */
221   def addExpressionRunner(dataFrame: DataFrame, ruleSuite: RuleSuite, name: String = "expressionResults",
222                        renderOptions: Map[String, String] = Map.empty, ddlType: String = "",
223                        stripDDL: Boolean = false, postProcess: DataFrame => DataFrame = identity): DataFrame = {
224     import org.apache.spark.sql.functions.expr
225     val runner =
226       if (ddlType.isEmpty)
227         expressionRunner(ruleSuite, name = name, renderOptions = renderOptions)
228       else
229         typedExpressionRunner(ruleSuite, name = name, ddlType = ddlType)
230 
231     postProcess(
232       dataFrame.select(expr("*"),
233         if (stripDDL)
234           strip_result_ddl(runner).as(name)
235         else
236           runner
237       )
238     )
239   }
