119   private def generateFunctions(ctx: CodegenContext, expr: Expression, i: Int, eval: ExprCode, argList: Seq[String],
120                                 callParameters: String) = {
121 
122     // forceInline so it's type is not corrupted, must be handled in the SplitCompilation
123     def isInitialised(i: Int) = s"$initialisationArray[$i]"
124 
125     def setInitialised(i: Int) = s"$initialisationArray[$i] = true;"
126 
127     def isNull(i: Int) = s"$isNullArray[$i]"
128 
129     // Generate the code for this expression tree and wrap it in a function.
130     val fnName = ctx.freshName("subExpr")
131 
132     val realValue = ctx.addMutableState(javaType(expr.dataType), "subExprValue")
133 
134     val isNullFnName = s"${fnName}ForIsNull"
135 
136     val isNullFn =
137       s"""
138          |private boolean $isNullFnName(${argList.mkString(", ")}) {
139          |  if (!${isInitialised(i)}) {
140          |    ${eval.code}
141          |    ${isNull(i)} = ${eval.isNull};
142          |    $realValue = ${eval.value};
143          |    ${setInitialised(i)}
144          |  }
145          |  return ${isNull(i)};
146          |}
147            """.stripMargin
148 
149     val nullCallName = ctx.addNewFunction(isNullFnName, isNullFn)
150     val nullCall = s"$nullCallName($callParameters)"
151 
152     val fn =
153       s"""
154          |private ${javaType(expr.dataType)} $fnName(${argList.mkString(", ")}) {
155          |  $nullCall;
156          |  return $realValue;
157          |}
158        """.stripMargin
159 
160     val fnCallName = ctx.addNewFunction(fnName, fn)
161     val initValueCall = s"$fnCallName($callParameters)"
162 
163     (initValueCall, nullCall)
164   }

171   def lazyNonWholeStageSubexpressionElimination(ctx: CodegenContext,
172                                                 dirtyExpressions: Seq[Expression]): String = {
173     val expressions = ShimUtils.copyStateful(dirtyExpressions)
174     import ctx._
175     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
176 
177     // Add each expression tree and compute the common subexpressions.
178     expressions.foreach(equivalentExpressions.addExprTree(_))
179     // there are no functions to call in lazy mode
180     val subexprFunctions = ""
181     // Get all the expressions that appear at least twice and set up the state for subexpression
182     // elimination.
183     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
184     commonExprs.zipWithIndex.foreach { case (expr, i) =>
185       // Generate the code for this expression tree and wrap it in a function.
186       val eval = expr.genCode(ctx)
187 
188       val (value: String, isNullF: String) =
189         generateFunctions(ctx,expr, i, eval, Seq(s"InternalRow $INPUT_ROW"), INPUT_ROW)
190 
191       // Add a state and a mapping of the common subexpressions that are associate with this
192       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
193       // when it is code generated. This decision should be a cost based one.
194       //
195       // The cost of doing subexpression elimination is:
196       //   1. Extra function call, although this is probably *good* as the JIT can decide to
197       //      inline or not.
198       // The benefit doing subexpression elimination is:
199       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
200       //      above.
201       //   2. Less code.
202       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
203       // at least two nodes) as the cost of doing it is expected to be low.
204 
205       val state = ShimExprUtils.state(
206         ExprCode(code"", SimpleExprValue(isNullF, java.lang.Boolean.TYPE), JavaCode.global(value, expr.dataType))
207       )
208       ShimExprUtils.addSubExpr(ctx, expr, state)
209     }
210 
211     subexprFunctions
212   }
