123   private def generateFunctions(ctx: CodegenContext, expr: Expression, i: Int, eval: ExprCode, argList: Seq[String],
124                                 callParameters: String) = {
125 
126     // forceInline so it's type is not corrupted, must be handled in the SplitCompilation
127     def isInitialised(i: Int) = s"$initialisationArray[$i]"
128 
129     def setInitialised(i: Int) = s"$initialisationArray[$i] = true;"
130 
131     def isNull(i: Int) = s"$isNullArray[$i]"
132 
133     // Generate the code for this expression tree and wrap it in a function.
134     val fnName = ctx.freshName("subExpr")
135 
136     val realValue = ctx.addMutableState(javaType(expr.dataType), "subExprValue")
137 
138     val isNullFnName = s"${fnName}ForIsNull"
139 
140     val isNullFn =
141       s"""
142          |private boolean $isNullFnName(${argList.mkString(", ")}) {
143          |  if (!${isInitialised(i)}) {
144          |    ${eval.code}
145          |    ${isNull(i)} = ${eval.isNull};
146          |    $realValue = ${eval.value};
147          |    ${setInitialised(i)}
148          |  }
149          |  return ${isNull(i)};
150          |}
151            """.stripMargin
152 
153     val nullCallName = ctx.addNewFunction(isNullFnName, isNullFn)
154     val nullCall = s"$nullCallName($callParameters)"
155 
156     val fn =
157       s"""
158          |private ${javaType(expr.dataType)} $fnName(${argList.mkString(", ")}) {
159          |  $nullCall;
160          |  return $realValue;
161          |}
162        """.stripMargin
163 
164     val fnCallName = ctx.addNewFunction(fnName, fn)
165     val initValueCall = s"$fnCallName($callParameters)"
166 
167     (initValueCall, nullCall)
168   }

175   def lazyNonWholeStageSubexpressionElimination(ctx: CodegenContext,
176                                                 dirtyExpressions: Seq[Expression]): String = {
177     val expressions = ShimUtils.copyStateful(dirtyExpressions)
178     import ctx._
179     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
180 
181     // Add each expression tree and compute the common subexpressions.
182     expressions.foreach(equivalentExpressions.addExprTree(_))
183     // there are no functions to call in lazy mode
184     val subexprFunctions = ""
185     // Get all the expressions that appear at least twice and set up the state for subexpression
186     // elimination.
187     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
188     commonExprs.zipWithIndex.foreach { case (expr, i) =>
189       // Generate the code for this expression tree and wrap it in a function.
190       val eval = expr.genCode(ctx)
191 
192       val (value: String, isNullF: String) =
193         generateFunctions(ctx,expr, i, eval, Seq(s"InternalRow $INPUT_ROW"), INPUT_ROW)
194 
195       // Add a state and a mapping of the common subexpressions that are associate with this
196       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
197       // when it is code generated. This decision should be a cost based one.
198       //
199       // The cost of doing subexpression elimination is:
200       //   1. Extra function call, although this is probably *good* as the JIT can decide to
201       //      inline or not.
202       // The benefit doing subexpression elimination is:
203       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
204       //      above.
205       //   2. Less code.
206       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
207       // at least two nodes) as the cost of doing it is expected to be low.
208 
209       val state = ShimExprUtils.state(
210         ExprCode(code"", SimpleExprValue(isNullF, java.lang.Boolean.TYPE), JavaCode.global(value, expr.dataType))
211       )
212       ShimExprUtils.addSubExpr(ctx, expr, state)
213     }
214 
215     subexprFunctions
216   }
