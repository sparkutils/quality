34   /**
35    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
36    *
37    * @param ruleSuite The Qualty RuleSuite to evaluate
38    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
39    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
40    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
41    * @param variableFuncGroup Defaulting to 20
42    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
43    * @return A Column representing the Quality DQ expression built from this ruleSuite
44    */
45   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
46                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
47                      extraConfig: Map[String, String] = Map.empty): Column = {
48     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
49     val flattened = flattenExpressions(ruleSuite)
50     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
51     val input =
52       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
53       if (compileEvals)
54         flattened.map(PassThroughCompileEvals)
55       else
56         flattened
57 
58     val runner =
59       if (forceRunnerEval || resolveWith.isDefined)
60         new RuleRunnerEval(cleaned, input,
61           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
62       else
63         new RuleRunner(cleaned, input,
64           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
65 
66     column(
67       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
68         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
69 
70         resolved.withNewChildren(resolved.children.map{
71           // replace the expr
72           case PassThroughCompileEvals(child) => NonPassThrough(child)
73           case child => NonPassThrough(child)
74         })
75       } getOrElse ZeroCodeGenWrap.wrap(runner)
76     )
77   }
