113     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
114       newParser()
115     }
116     /*
117     attempt for a simple expression, then try a plan with exactly one output row.
118     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
119      */
120     val rawExpr =
121       try {
122         parser.parseExpression(rule)
123       } catch {
124         case ot: Throwable => // e.g. suitable for output expressions
125           try {
126             ScalarSubquery(parser.parsePlan(rule))
127           } catch {
128             case _: Throwable =>
129               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
130               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
131               val r = rule.split("->")
132               val wrapped =
133                 if (r.size == 2)
134                   s"${r(0)} -> ( ${r(1)} )"
135                 else
136                   s"( ${r(0)} )"
137 
138               try {
139                 parser.parseExpression(wrapped)
140               } catch {
141                 case _: Throwable => throw ot // if this didn't work return the original error
142               }
143           }
144       }
145 
146     val res =
147       rawExpr match {
148         case l: SparkLambdaFunction if hasSubQuery(l) =>
149           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
