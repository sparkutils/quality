14   def collectFirstZeroCodeGenSplitCompilations(ctx: CodegenContext, expr: Seq[Expression]): Seq[SplitCompilation] = {
15     def firstZeroWith(expression: Expression): Seq[SplitCompilation] =
16       expression match {
17         /*
18         via #145 OSS uses the deterministic property to remove nesting and enforce projections, DBR ignores this
19         and behaves like pre #145
20         */
21         // $COVERAGE-OFF$
22         case ZeroCodeGen(_, s: SplitCompilation, _, _) =>
23           // ensure it has been compiled
24           expression.genCode(ctx)
25           // do not go further
26           Seq(s)
27         case p: ExpressionProxy =>
28           firstZeroWith(p.child)
29         // $COVERAGE-ON$
30         case _ => expression.children.flatMap(firstZeroWith)
31       }
32     expr.flatMap(firstZeroWith)
33   }
