136    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are

138    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.

140   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
141     val expressions = ShimUtils.copyStateful(dirtyExpressions)
142     import ctx._
143     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
144 
145     // Add each expression tree and compute the common subexpressions.
146     expressions.foreach(equivalentExpressions.addExprTree(_))
147     var subexprFunctions = ""
148     // Get all the expressions that appear at least twice and set up the state for subexpression
149     // elimination.
150     val commonExprs = ShimExprUtils.getAllEquivalentExprs(equivalentExpressions)
151     commonExprs.foreach { expr =>
152       val fnName = freshName("subExpr")
153       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
154       val value = addMutableState(javaType(expr.dataType), "subExprValue")
155 
156       // Generate the code for this expression tree and wrap it in a function.
157       val eval = expr.genCode(ctx)
158       val fn =
159         s"""
160            |private void $fnName(InternalRow $INPUT_ROW) {
161            |  ${eval.code}
162            |  $isNull = ${eval.isNull};
163            |  $value = ${eval.value};
164            |}
165            """.stripMargin
166 
167       // Add a state and a mapping of the common subexpressions that are associate with this
168       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
169       // when it is code generated. This decision should be a cost based one.
170       //
171       // The cost of doing subexpression elimination is:
172       //   1. Extra function call, although this is probably *good* as the JIT can decide to
173       //      inline or not.
174       // The benefit doing subexpression elimination is:
175       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
176       //      above.
177       //   2. Less code.
178       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
179       // at least two nodes) as the cost of doing it is expected to be low.
180 
181       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
182       subexprFunctions += subExprCode
183       val state = ShimExprUtils.state(
184         ExprCode(code"$subExprCode",
185           JavaCode.isNullGlobal(isNull),
186           JavaCode.global(value, expr.dataType)))
187       ShimExprUtils.addSubExpr(ctx, expr, state)
188     }
189 
190     subexprFunctions
191   }

208   /**
209    * Splits the generated code of expressions into multiple functions, because function has
210    * 64kb code size limit in JVM. If the class to which the function would be inlined would grow
211    * beyond 1000kb, we declare a private, inner sub-class, and the function is inlined to it
212    * instead, because classes have a constant pool limit of 65,536 named values.
213    *
214    * @param expressions the codes to evaluate expressions.
215    * @param funcName the split function name base.
216    * @param arguments the list of (type, name) of the arguments of the split function.
217    * @param returnType the return type of the split function.
218    * @param makeSplitFunction makes split function body, e.g. add preparation or cleanup.
219    * @param foldFunctions folds the split function calls.
220    */
221   def splitExpressions(ctx: CodegenContext,
222                        expressions: Seq[String],
223                        groupingSize: Int,
224                        funcName: String,
225                        arguments: Seq[(String, String)],
226                        returnType: String = "void",
227                        makeSplitFunction: String => String = identity,
228                        foldFunctions: Seq[String] => String = _.mkString("", ";\n", ";")): String = {
229     val blocks = buildCodeBlocks(expressions.grouped(groupingSize).map(_.mkString("\n")).toSeq)
230 
231     if (blocks.length == 1) {
232       // inline execution if only one block
233       blocks.head
234     } else {
235       val func = ctx.freshName(funcName)
236       val argString = arguments.map { case (t, name) => s"$t $name" }.mkString(", ")
237       val functions = blocks.zipWithIndex.map { case (body, i) =>
238         val name = s"${func}_$i"
239         val code = s"""
240                       |private $returnType $name($argString) {
241                       |  ${makeSplitFunction(body)}
242                       |}
243          """.stripMargin
244         addNewFunctionInternal.invoke(ctx, name, code, java.lang.Boolean.FALSE).asInstanceOf[NewFunctionSpec]
245       }
246 
247       val (outerClassFunctions, innerClassFunctions) = functions.partition(_.innerClassName.isEmpty)
248 
249       val argsString = arguments.map(_._2).mkString(", ")
250       val outerClassFunctionCalls = outerClassFunctions.map(f => s"${f.functionName}($argsString)")
251 
252       val innerClassFunctionCalls = generateInnerClassesFunctionCall.invoke(ctx,
253         innerClassFunctions,
254         func,
255         arguments,
256         returnType,
257         makeSplitFunction,
258         foldFunctions).asInstanceOf[Iterable[String]]
259 
260       foldFunctions(outerClassFunctionCalls ++ innerClassFunctionCalls)
261     }
262   }

265   /**
266    * Splits the generated code of expressions into multiple sequences of String
267    * based on a threshold of length of a String
268    *
269    * @param expressions the codes to evaluate expressions.
270    */
271   private def buildCodeBlocks(expressions: Seq[String]): Seq[String] = {
272     val blocks = new ArrayBuffer[String]()
273     val blockBuilder = new StringBuilder()
274     var length = 0
275     val splitThreshold = SQLConf.get.methodSplitThreshold
276     for (code <- expressions) {
277       // We can't know how many bytecode will be generated, so use the length of source code
278       // as metric. A method should not go beyond 8K, otherwise it will not be JITted, should
279       // also not be too small, or it will have many function calls (for wide table), see the
280       // results in BenchmarkWideTable.
281       if (length > splitThreshold) {
282         blocks += blockBuilder.toString()
283         blockBuilder.clear()
284         length = 0
285       }
286       blockBuilder.append(code)
287       length += code.length//CodeFormatter.stripExtraNewLinesAndComments(code).length
288     }
289     blocks += blockBuilder.toString()
290     blocks.toSeq
291   }

293   /**
294    * Manage project vs wholestage exec, lambdas and subexpr, the function f is responsible for using the results
295    *
296    * @param ctx
297    * @param fullChildren
298    * @param lazyGen when true sub expr code generation is lazy, suitable for large rulesuites
299    * @param f children that have had FunNLambda.swap then swapBack applied, 2nd param is the subexpr 'pre' code, third
300    *          signifies currentVars present rather than itr
301    * @tparam R1
302    * @tparam R2
303    * @return
304    */
305   def produceCode[R1,R2](ctx: CodegenContext, fullChildren: Seq[Expression], lazyGen: Boolean)
306                         (f: (Seq[Expression], String, Boolean) => (R1, R2)): (R1, R2) = {
307 
308     // replace all usedAsLambda FunNs to make sure they cannot be turned into subexprs
309     val lambdaSafeChildren = fullChildren.map(FunNLambda.swap)
310 
311     if (ctx.currentVars eq null) {
312       val subExpressionCode = QualityCodeGenUtils.nonWholeStageSubexpressionElimination(ctx, lambdaSafeChildren)
313 
314       // replace the original non-subExpr FunNs
315       val children = lambdaSafeChildren.map(FunNLambda.swapBack)
316 
317       f(children, subExpressionCode, false)
318     } else {
319       val subExprs =
320         if (lazyGen)
321           LazySubExprCodeGen.lazySubexpressionEliminationForWholeStageCodegen(ctx, lambdaSafeChildren)
322         else
323           SubExprCodeGen.subexpressionEliminationForWholeStageCodegen(ctx, lambdaSafeChildren)
324 
325       val subExpressionCodeRaw = ShimExprUtils.evaluateSubExprEliminationState(ctx, subExprs)
326 
327       val subExpressionCode =
328         if (lazyGen) {
329           ctx.addMutableState("boolean[]", initialisationArray, forceInline = true, useFreshName = false)
330           ctx.addMutableState("boolean[]", isNullArray, forceInline = true, useFreshName = false)
331 
332           // reset the state for the values
333           s"""
334             $initialisationArray = new boolean[${subExprs.states.size}];
335             $isNullArray = new boolean[${subExprs.states.size}];
336             $subExpressionCodeRaw
337           """
338 
339         } else
340           subExpressionCodeRaw
341 
342       // replace the original non-subExpr FunNs
343       val children = lambdaSafeChildren.map(FunNLambda.swapBack)
344 
345       QualityCodeGenUtils.withSubExprEliminationExprs(ctx, subExprs.states) {
346         f(children, subExpressionCode, true)
347       }
348     }
349   }
