281   def stripBrackets(v: VariableValue): (String, String) = {
282     val openb = v.toString().indexOf("[")
283     if (openb == -1)
284       (v.variableName, "")
285     else
286       (v.variableName.dropRight(v.length - openb), v.variableName.drop(openb))

289   def formatParams(ctx: CodegenContext, a: Seq[ExprValue], callsKeepArrays: Boolean = false): ParameterInformation = {
290     // filter out any top level arrays, the input is a set, so params need the same order
291     val ordered = a.flatMap {
292       case a: VariableValue => Some(a)
293       case _ => None
294     }
295 
296     val pairs = ordered.map { v =>
297       val (stripped, arrayInName) = stripBrackets(v)
298 
299       val (typ, array) =
300         if (v.javaType.isArray)
301           (s"${v.javaType.getComponentType.getName}", "[]")
302         else if (v.javaType.isPrimitive)
303           (v.javaType.toString, arrayInName.replaceAll("[^\\[\\]]",""))
304         else
305           (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
306 
307       (s"$typ$array", stripped, v.javaType)
308     }
309 
310     ParameterInformation(pairs.map {
311       case (typ, stripped, _) =>
312 
313         s"$typ $stripped"
314       }.mkString(", ")
315       , ordered.map(v =>
316         if (v.javaType.isArray && callsKeepArrays)
317           v.variableName
318         else
319           stripBrackets(v)._1
320       ).mkString(", "), ordered.size, pairs)
321   }
