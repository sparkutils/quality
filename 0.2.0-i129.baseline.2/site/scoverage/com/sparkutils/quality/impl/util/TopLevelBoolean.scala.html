226   def from(expression: Expression, subExprs: Map[Expression, Int], filterOut: Double): Expression = {
227 
228     // anything that hits more than x % is not useful to group with, any top bool that only applies to one trigger is

233       case _ if res.isEmpty  => expression
234       case _ => res.reduce(And(_,_))
