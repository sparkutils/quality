12 object Arrays {
13   /**
14    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
15    *
16    * @param array
17    * @param dataType
18    * @param f
19    * @return
20    */
21   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
22     array match {
23       case _: UnsafeArrayData =>
24         val res = Array.ofDim[T](array.numElements() + by)
25         array.foreach(dataType, (i, v) => res.update(i, f(v)))
26         res
27       case _ => {
28         val r = array.array.map(f)
29         if (by == 0)
30           r
31         else {
32           val res = Array.ofDim[T](array.numElements() + by)
33           r.copyToArray(res)
34           res
35         }
36 
37       }
38     }
39 
40   /**
41    * gets an array out of UnsafeArrayData or others
42    * @param array
43    * @param dataType
44    * @return
45    */
46   def toArray(array: ArrayData, dataType: DataType): Array[Any] =
47     array match {
48       case _: UnsafeArrayData =>
49         mapArray(array, dataType, identity)
50       case _ => array.array

92     val keys = mapData.keyArray()

121     val ns = map.numElements() + withPairs.length
122     val nka = Arrays.mapArray(map.keyArray(), kt, identity, withPairs.length)
123     for(i <- (map.numElements() until ns).zipWithIndex) {

131     }

164   /**
165    * finds matching entries in the keyArray
166    * @param map
167    * @param equal
168    * @return -1 when not found, otherwise the index of the matching kez
169    */
170   def getKeyIndex(map: MapData, equal: (ArrayData, Int) => Boolean): Int = {
171     var index = -1
172     var i = 0
173     val keys = map.keyArray()
174     while(index == -1 && i < map.numElements()) {
175       if (equal(keys, i)) {
176         index = i
177       }
178       i += 1
179     }
180     index
181   }
