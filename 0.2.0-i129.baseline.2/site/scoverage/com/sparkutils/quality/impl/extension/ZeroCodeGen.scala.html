29     if (newChild.collect{ case c => c.isInstanceOf[Unevaluable] }.nonEmpty)
30       copy(child = newChild, realChild = newChild)
31     else // hopefully late enough to not be an issue to swap a nullable, the below match is a safeguard

34           logTrace("ZeroCodeGen doesn't expect another expression optimisation after Unevaluable's are removed but got "+newChild)

36         case _ =>
37           copy(child = Literal(null, newChild.dataType), realChild = newChild)
38       }
39 }
