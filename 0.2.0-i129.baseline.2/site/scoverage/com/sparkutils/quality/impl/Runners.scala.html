98     val sar = result.getMap(2).asInstanceOf[ArrayBasedMapData]
99     // update result directly
100     val sv = sar.valueArray.asInstanceOf[GenericArrayData]
101     val struct = sv.getStruct(level1, 2)

105     val cur = struct.getInt(0)
106     val nr = defaultOverallProcessor(ruleResult, cur)
107     struct.update(0, nr)
