100 
101         val alreadyProcessed = mutable.Map.empty[Id, RunOnPassProcessor]
102 
103         override def processor(t: Rule): RunOnPassProcessor = t.runOnPassProcessor
104 
105         override def updateWith(t: Rule, outputExpression: OutputExpression): Rule = {
106           val existing = t.runOnPassProcessor
107           val processor =
108             alreadyProcessed.getOrElse(existing.id, {
