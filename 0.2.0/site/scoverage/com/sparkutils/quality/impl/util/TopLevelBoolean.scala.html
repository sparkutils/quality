352   def fromParts(expression: Expression): Set[Expression] = expression match {
353     // TODO: intentionally excluded from subexpression elimination
354     case _: Or => Set.empty
355     case And(left, right) => fromParts(left) ++ fromParts(right)
356     // if_relevant is IntegerType, so the BooleanType case below never sees it.  Its filter is
357     // only groupable because Group carries filterFalseResults, see filterFalseResultFor.
358     // A nullable filter cannot be grouped: if_relevant answers Failed for a null filter but the
359     // generated branch is `if ((!isNull) && value)`, which cannot tell null from false.
360     case IfRelevantExpr(filter, _) if filter.dataType == BooleanType && !filter.nullable =>
361       fromParts(filter)
362     case If(predicate, trueValue, falseValue)
363       if predicate.dataType == BooleanType &&
364          trueValue.dataType == BooleanType &&
365          falseValue.dataType == BooleanType =>
366       fromParts(predicate)
367     case e: Expression if e.dataType == BooleanType =>
368       Set(e)
369     case _ => Set.empty
370   }

374    *
375    * fromParts recurses into a composite filter and the frequency filter in `from` then keeps only
376    * the shared conjuncts, so the group filter is regularly a strict subset of the condition it was

381     condition match {
382       case _ if condition.semanticEquals(groupFilter) => true
383       case And(left, right) => coversCondition(groupFilter, left) || coversCondition(groupFilter, right)
384       case _ => false
