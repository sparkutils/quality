448               ${funNames.map { f => s"$f(${params.paramsCall});" }.mkString("\n")}

471            """
472

478         ${eval.code}
479         if ((!${eval.isNull}) && ${eval.value} ) {
480           ${expr.code}
481         }
482         $filterFalseBranch
483         """
484       ), body, compilationParams)
