11   /**
12    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite.
13    *
14    * NOTE if using this in a Connect session resolveWith, compileEvals, forceRunnerEval and forceTriggerEval are ignored and use defaults
15    *
16    * @param ruleSuite The ruleSuite with runOnPassProcessors
17    * @param resultDataType The type of the results from runOnPassProcessors - must be the same for all result types,
18    *                       by default most fields will be nullable and encoding must follow the fields when not specified.   *
19    * @param compileEvals Should the rules be compiled out to interim objects - by default false
20    * @param debugMode When debugMode is enabled the resultDataType is wrapped in Array of (salience, result)
21    *                  pairs to ease debugging
22    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve
23    *                    and optimise the sql expressions, see the documentation for details on when to and not to use this.
24    * @param variablesPerFunc Defaulting to 40 allows, in combination with variableFuncGroup allows customisation of
25    *                         handling the 64k jvm method size limitation when performing WholeStageCodeGen
26    * @param variableFuncGroup Defaulting to 20
27    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation
28    *                        (compileEvals must be false to get fully interpreted)
29    * @param forceTriggerEval Defaulting to false
30    * @return A Column representing the QualityRules expression built from this ruleSuite
31    */
32   def ruleEngineRunner(ruleSuite: RuleSuite, resultDataType: Option[DataType] = None, compileEvals: Boolean = false,
33                        debugMode: Boolean = false, resolveWith: Option[DataFrame] = None, variablesPerFunc: Int = 40,
34                        variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, forceTriggerEval: Boolean = false,
35                        extraConfig: Map[String, String] = Map.empty): Column =
36     if (ResolveUtil.checkResolveMakesSenseOrClassic(resolveWith))
37       RuleEngineRunnerImpl.ruleEngineRunnerImpl(ruleSuite, resultDataType, compileEvals, debugMode, resolveWith, variablesPerFunc,
38         variableFuncGroup, forceRunnerEval, forceTriggerEval, extraConfig)
39     else
40       ShimUtils.callFunction("rule_engine_runner", lit(RuleSuiteHelpers.serialize(ruleSuite)),
41         lit(resultDataType.map(_.sql).getOrElse("")), lit(debugMode), lit(variablesPerFunc),
42         lit(variableFuncGroup), typedLit(extraConfig)
43       )
