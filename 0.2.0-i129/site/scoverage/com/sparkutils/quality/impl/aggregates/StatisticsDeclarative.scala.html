146   def allMergedIn(base: MapData, newRows: Seq[(InternalRow, Boolean)]): Boolean = {
147     val inplace = base.valueArray().asInstanceOf[GenericArrayData].array
148 
149     val bN = base.numElements()
150     val bK = base.keyArray()
151 
152     var notIn = false
153     var i = 0
154     while( i < newRows.length && !notIn) {
155       val id = newRows(i)._1.getLong(0)
156       var j = 0
157       var in = false
158       while( j < bN && !in) {
159         if (bK.getLong(j) == id) {
160           in = true

162         }
163         j += 1
164       }

166       i += 1
167     }

207               if (nextInputRowSize > 0)

209               else
210                 // it's the rule level result

212               , rs, stats, depth + 1) // this should be the correct row, so we should be inside already...

214       // if any new rows were added below then we'll have to re-create all above
215       val createdSubMap = newRows.exists(_._2)
216       def newRow: InternalRow = {
217         // if rs is GenericArrayData map based we may be able to in-place swap entries, scanning will be quicker than map changes

229           }

232           } {

236           val values = new GenericArrayData(newMap.values.toArray)

238         }
239       }
240

242         rs

244         newRow,
245         createdNewMap || createdSubMap)
246     }

253       // either we need to re-integrate it or it was brand new
