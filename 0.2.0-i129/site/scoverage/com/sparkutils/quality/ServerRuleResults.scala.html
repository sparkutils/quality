55   @tailrec
56   final def processResult(t: T)(ruleResult: RuleResult, probabilityPass: Double = defaultProbablePass): T =
57     ruleResult match {
58       case Failed => update(t)(failed = t.failed + 1)
59       case Passed => update(t)(passed = t.passed + 1)
60       case SoftFailed => update(t)(softFailed = t.softFailed + 1)
61       case DisabledRule => update(t)(disabled = t.disabled + 1)
62       case IgnoredRule => update(t)(ignored = t.ignored + 1)
63       case DefaultRule => update(t)(defaulted = t.defaulted + 1)
64       case Probability(percentage) if (percentage >= probabilityPass) => update(t)(probabilityPassed = t.probabilityPassed + 1)
65       case Probability(_) => update(t)(probabilityFailed = t.probabilityFailed + 1)
66       case RuleResultWithProcessor(ruleResult: RuleResult, _) =>
67         processResult(t)(ruleResult, probabilityPass)
68       case UnevaluatedRule => update(t)(unevaluated = t.unevaluated + 1)
69     }
