254   protected def produceGroupTriggers(ctx: CodegenContext, runner: Runner, additionalParams: Seq[VariableValue], prefix: String,
255                                      exprEnd: () => Block, exprFunEnd: () => Block, groupSalienceCheck: String => Block,

261

288         val exprRunner =

292 
293         // the top level is 0 arrays are filtered out from the row as they don't need explicit returning
294         GenerateResult(SeparateClassGenerator(runner.getClass.getName, additionalParams, groupIndex + 1), exprRunner.copy(
295           code =
296             code"""
297                               boolean ${exprRunner.isNull} = false;
298                               ${funNames.map { f => s"$f(${params.paramsCall});" }.mkString("\n")}
299                               GenericInternalRow ${exprRunner.value} = new org.apache.spark.sql.catalyst.expressions.GenericInternalRow(

304                               """

306       }
307 
308     val eval = group.groupFilter.genCode(ctx)
309 
310     // if ruleEngine is used salience may need comparison, if it's expression or dq any comparison is meaningless
311     (
312       code"""
313         ${exprEnd()}\n
314         ${eval.code}
315         if ((!${eval.isNull}) && ${eval.value} ${groupSalienceCheck(group.lowestSalience.toString)} ) {
316           ${expr.code}
317         }
318       """, body)
