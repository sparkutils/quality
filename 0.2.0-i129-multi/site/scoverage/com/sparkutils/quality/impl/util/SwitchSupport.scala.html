93   private def generateSwitch(ctx: CodegenContext, prefix: String, groupDepth: Int, params: ParameterInformation,
94                              groupCalls: Seq[(String, Block)]) = {

96 
97     val (min, max) = (groupCalls.head._1, groupCalls.last._1)
98 
99     val cases = groupCalls.map {
100       case (bucket, codeToRun) =>
101         val exprFuncName = ctx.freshName(prefix + "callGEFuncSwitchGroup" + groupDepth)
102         val callFun = ctx.addNewFunction(exprFuncName,
103           code"""
104            private void $exprFuncName(${params.paramsDef}) {
105              $codeToRun
106            }
107           """.code
108         )

113         """

118     def buildSwitch(switchVal: String, cases: Seq[(String,Block)], default: String, single: Boolean): String = {
119       val folded = cases.foldLeft(code"") {
120         case (cur, n) =>
121           code"""
122           $cur
123           ${n._2}
124         """
125       }
126 
127       val switchIf: String => String =
128         if (single)
129           identity
130         else
131         (cases.headOption.map( v => greaterThanOrEqual(switchVal, v._1)),
132           cases.lastOption.map( v => lessThanOrEqual(switchVal, v._1))) match {
133           case (Some(min), Some(max)) =>
134             s => s"""
135                 if (($min) && ($max)) {
136                   $s
137                 } else {
138                   $default
139                 }
140               """
141           case (Some(min), None) =>
142             s => s"""
143                 if ($min) {
144                   $s
145                 } else {
146                   $default
147                 }
148               """
149           case (None, Some(max)) =>
150             s => s"""
151                 if ($max) {
152                   $s
153                 } else {
154                   $default
155                 }
156               """
157         }
158 
159 
160       switchIf(s"""
161         switch($switchVal) {
162           $folded
163         }
164          """)
165     }
166 
167     def buildSwitches(switchVal: String, chunked: Seq[Seq[(String,Block)]]): String = {
168       if (chunked.size == 1) {
169         buildSwitch(switchVal, chunked.head, "", true)
170       } else {
171         // we have more chunks
172         val head = chunked.head
173         val (restLeft, restRight) = chunked.tail.splitAt((chunked.size / 2) - 1)
174 
175         def headVal(chunked: Seq[Seq[(String,Block)]]): String =
176           (for {
177             first <- chunked.headOption
178             second <- first.headOption
179           } yield second._1).getOrElse(zero)
180 
181         val (restLeftStartBucket, restRightStartBucket) = (headVal(restLeft), headVal(restRight))
182 
183         def switches(chunked: Seq[Seq[(String,Block)]]): Option[String] =
184           if (chunked.isEmpty)
185             None
186           else
187             Some(buildSwitches(switchVal, chunked))
188 
189         val (switchLeft, switchRight) = (switches(restLeft), switches(restRight))
190 
191         def switch(switch: String, lOrR: String) = {
192           val exprFuncName = ctx.freshName(prefix + s"GEFuncSwitchGroupsNested${lOrR}_" + groupDepth)
193           val callFun = ctx.addNewFunction(exprFuncName,
194             code"""
195              private void $exprFuncName($typ $switchVal, ${params.paramsDef}) {
196                $switch
197              }
198             """.code
199           )
200           callFun
201         }
202 
203         val (leftFun, rightFun) = (switchLeft.map(switch(_, "L")), switchRight.map(switch(_, "R")))
204 
205         val (geLeft, lessThanRight) =
206           (greaterThanOrEqual(switchVal, restLeftStartBucket), lessThan(switchVal, restRightStartBucket))
207 
208         val defalt =
209           (leftFun, rightFun) match {
210             case (Some(leftFun), Some(rightFun)) =>
211               s"""
212                 if (($geLeft) && ($lessThanRight)) {
213                   $leftFun($switchVal, ${params.paramsCall});
214                 } else {
215                   $rightFun($switchVal, ${params.paramsCall});
216                 }
217               """
218             case (Some(leftFun), None) =>
219               s"""
220                 if ($geLeft) {
221                   $leftFun($switchVal, ${params.paramsCall});
222                 }
223               """
224             case (None, Some(rightFun)) =>
225               s"""
226                 if (${greaterThanOrEqual(switchVal, restRightStartBucket)}) {
227                   $rightFun($switchVal, ${params.paramsCall});
228                 }
229               """
230           }
231 
232         val r = buildSwitch(switchVal, head, defalt, false)
233 
234         r
235       }
236     }
237 
238     val minSpark = ctx.addMutableState(sparkType, "minSpark", initFunc = initSpark(_,min))
239     val maxSpark = ctx.addMutableState(sparkType, "maxSpark", initFunc = initSpark(_,max))
240 
241     val converted = conversion(expr.value)
242     val (converting, switchName) =
243       if (converted.isEmpty || converted == expr.value.code)
244         ("", expr.value.code)
245       else {
246         val fresh = ctx.freshName("converted")
247         (s"$typ $fresh = $converted;", fresh)
248       }
249

252       if (cases.size > groupSize)
253         s"|| (${lessThanSpark(expr.value, minSpark)}) || (${greaterThanSpark(expr.value, maxSpark)})"
254       else
255         ""
256

258       code"""
259        private void $exprFuncName(${params.paramsDef}) {
260          ${expr.code}
261          if (${expr.isNull} $rangeCheck) {
262          } else {
263            $converting
264 
265            ${
266             buildSwitches(switchName, /// TODO uses Spark mechanism to group, this is for PoC
267               cases.grouped(groupSize).toSeq)
268           }
269          }
270        }
271       """.code
272     )
273   }
