39   def genArrayCompare(elementType: DataType, elComp: (Any, Any) => Int): (Any, Any) => Int = (oleft, oright) => {
40     val left = oleft.asInstanceOf[ArrayData]
41     val right = oright.asInstanceOf[ArrayData]
42     if (left.numElements() > right.numElements())
43       1
44     else if (left.numElements() < right.numElements())
45       -1
46     else {
47       // same
48       var i = 0
49       var res = 0
50       val al = (idx: Int) => left.get(idx, elementType)// Arrays.toArray(left, elementType)
51       val ar = (idx: Int) => right.get(idx, elementType)// Arrays.toArray(right, elementType)
52       while (i < left.numElements() && res == 0) {
53         res = elComp(al(i), ar(i))
54         i += 1
55       }
56       res
57     }
58   }

60   def defaultMapCompare(dataType: DataType, extension: DataType => Option[(Any, Any) => Int] = noopCompare): Option[(Any, Any) => Int] =
61     dataType match {
62       case mt: MapType => {
63         val kt = defaultMapCompare(mt.keyType, extension).
64           getOrElse(sys.error(s"Could not find compare function for map key type ${mt.keyType}"))
65 
66         val kr = BoundReference(0, mt.keyType, nullable = true)
67 
68         // actual field is in a key val struct
69         Some(
70           genArrayCompare(keyValueType(mt.keyType, mt.valueType),
71             (left, right) => {
72               val lkv = kr.eval(left.asInstanceOf[InternalRow])
73               val rkv = kr.eval(right.asInstanceOf[InternalRow])
74 
75               val r = kt(lkv, rkv)
76               if (r != 0)
77                 r
78               else {
79                 /* shouldn't happen unless MapData is created directly and it's not
80                    actually a map checking on key uniqueness, here for completeness */
81                 val vt = defaultMapCompare(mt.valueType, extension).
82                   getOrElse(sys.error(s"Could not find compare function for map value type ${mt.valueType}"))
83                 val vr = BoundReference(1, mt.valueType, nullable = true)
84 
85                 val lvv = vr.eval(left.asInstanceOf[InternalRow])
86                 val rvv = vr.eval(right.asInstanceOf[InternalRow])
87 
88                 vt(lvv, rvv)
89               }
90             })
91         )
92       }
93       case dt: AtomicType => Some(compareToOrdering(ShimUtils.sparkOrdering(dt)))
94       case arrayType: ArrayType =>
95         val oelcomp = defaultMapCompare(arrayType.elementType, extension)
96         oelcomp.map{
97           elcomp =>
98             genArrayCompare(arrayType.elementType, elcomp)
99         }
100       case structType: StructType =>
101         val allFields = structType.fields.zipWithIndex.map{
102           case (f, index) =>
103             defaultMapCompare( f.dataType, extension).map(cf => (BoundReference(index, f.dataType, nullable = true), cf)).
104               getOrElse(sys.error(s"Could not find compare function for ${f.dataType} in struct $structType"))
105         }
106         Some(
107           // fail early for any non 0
108           (left, right) => {
109             var i = 0
110             var res = 0
111             while (i < allFields.size && res == 0) {
112               val leftV = allFields(i)._1.eval(left.asInstanceOf[InternalRow])
113               val rightV = allFields(i)._1.eval(right.asInstanceOf[InternalRow])
114               res = allFields(i)._2(leftV, rightV)
115               i += 1
116             }
117             res
118           }
119         )
120       case _ => extension(dataType).orElse(
121         sys.error(s"Could not find compare function for ${dataType}")
122       )
123     }

126   def compress(obj: Object with Serializable): Array[Byte] = {
127     val bos = new ByteArrayOutputStream()
128     val cos = CompressionCodec.createCodec(SparkContext.getActive.get.conf).compressedOutputStream(bos)
129     val os = new ObjectOutputStream(cos)
130     // get rid of List's Vectors are serializable
131     os.writeObject(obj)
132     val res = bos.toByteArray
133     os.close()

137   def decompress[T: ClassTag](arr: Array[Byte]) = {
138     val bis = new ByteArrayInputStream(arr)
139     val cis = CompressionCodec.createCodec(SparkContext.getActive.get.conf).compressedInputStream(bis)
140     val os = new ObjectInputStream(cis) {
141       override def resolveClass(desc: ObjectStreamClass): Class[_] =
142         Class.forName(desc.getName, false, getContextOrSparkClassLoader)
143     }
144     val suite = os.readObject()
145     os.close()
146     suite.asInstanceOf[T]
