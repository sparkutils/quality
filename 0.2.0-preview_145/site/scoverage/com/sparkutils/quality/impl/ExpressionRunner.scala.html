24 object ExpressionRunner {
25   /**
26    * Runs the ruleSuite expressions saving results as a tuple of (ruleResult: yaml, resultType: String)
27    * Supplying a ddlType triggers the output type for the expression to be that ddl type, rather than using yaml conversion.
28    * @param renderOptions provides rendering options to the underlying snake yaml implementation
29    * @param ddlType optional DDL string, when present yaml output is disabled and the output expressions must all have the same type
30    * @param name the default column name "expressionResults"
31    */
32   def apply(ruleSuite: RuleSuite, name: String = "expressionResults", renderOptions: Map[String, String] = Map.empty,
33             ddlType: String = "", variablesPerFunc: Int = 40, variableFuncGroup: Int = 20,
34             forceRunnerEval: Boolean = false, compileEvals: Boolean = false,
35             extraConfig: Map[String, String] = Map.empty): Column = {
36     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
37     val expressions = flattenExpressions(ruleSuite)
38     val collectExpressions =
39       if (ddlType.isEmpty)
40         expressions.map( i => YamlEncoderExpr(i, renderOptions))
41       else
42         expressions
43     val exprs =
44       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
45       if (compileEvals)
46         collectExpressions.map(PassThroughCompileEvals)
47       else
48         collectExpressions
49 
50     val ddl_type =
51       if (ddlType.isEmpty)
52         expressionResultTypeYaml
53       else
54         DataType.fromDDL(ddlType)
55 
56     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
57 
58     ShimUtils.column(ZeroCodeGenWrap.wrap(
59       if (forceRunnerEval)
60         new ExpressionRunnerEval(cleaned, exprs,
61           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
62           compileEvals = compileEvals, extraConfig = extraConfig)
63       else
64         new ExpressionRunnerCompiled(cleaned, exprs,
65           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
66           compileEvals = compileEvals, extraConfig = extraConfig)
67     )).as(name)
68   }
69 }

73   protected[quality] def expressionsResultToRow[R](ruleSuiteResult: GeneralExpressionsResult[R]): InternalRow =
74     InternalRow(
75       packId(ruleSuiteResult.id),
76       ArrayBasedMapData(
77         ruleSuiteResult.ruleSetResults, packId _, (a: Any) => {
78           val v = a.asInstanceOf[Map[VersionedId, GeneralExpressionResult]]
79           ArrayBasedMapData(
80             v, packId _, (a: Any) => a match {
81               case r: GeneralExpressionResult =>
82                 InternalRow(UTF8String.fromString( r.ruleResult ), UTF8String.fromString( r.resultDDL) )
83               case s: String =>  UTF8String.fromString( s )
84               case _ => a // handle nulls *and* R's
85             }
86           )
87         }
88       )
89     )
