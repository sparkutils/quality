20    * Creates a column that runs the folding RuleSuite.  This also forces registering the lambda functions used by that RuleSuite.
21    *
22    * FolderRunner runs all output expressions for matching rules in order of salience, the startingStruct is passed ot the first
23    * matching, the result passed to the second etc.  In contrast to ruleEngineRunner OutputExpressions should be lambdas with one parameter, that of the structure
24    *
25    * NOTE if using this in a Connect session resolveWith, compileEvals, forceRunnerEval and forceTriggerEval are ignored and use defaults
26    *
27    * @param ruleSuite The ruleSuite with runOnPassProcessors
28    * @param startingStruct This struct is passed to the first matching rule, ideally you would use the spark dsl struct function to refer to existing columns
29    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
30    * @param debugMode When debugMode is enabled the resultDataType is wrapped in Array of (salience, result) pairs to ease debugging
31    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this.
32    * @param variablesPerFunc Defaulting to 40 allows, in combination with variableFuncGroup allows customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen
33    * @param variableFuncGroup Defaulting to 20
34    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
35    * @param forceTriggerEval Defaulting to false, passing true forces each trigger expression to be compiled (compileEvals) and used in place, false instead expands the trigger in-line giving possible performance boosts based on JIT
36    * @param useType In the case you must use select and can't use withColumn you may provide a type directly to stop the NPE
37    * @return A Column representing the QualityRules expression built from this ruleSuite
38    */

66    * @param forceTriggerEval Defaulting to false, passing true forces each trigger expression to be compiled (compileEvals) and used in place, false instead expands the trigger in-line giving possible performance boosts based on JIT
67    * @param useType In the case you must use select and can't use withColumn you may provide a type directly to stop the NPE
68    * @return A Column representing the QualityRules expression built from this ruleSuite
69    */
70   def ruleFolderRunnerClassic(ruleSuite: RuleSuite, startingStruct: Column, compileEvals: Boolean = false,
71                        debugMode: Boolean = false, resolveWith: Option[DataFrame] = None, variablesPerFunc: Int = 40,
72                        variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, useType: Option[StructType] = None,
73                        forceTriggerEval: Boolean = false, extraConfig: Map[String, String] = Map.empty): Column = {
74     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
75 
76     // needed to resolve variables -- this changes between invocation and stops the type checks.  In the test case it's subcode that is on one type nullable and the other not
77     val dataRef = new AtomicReference[DataType]()
78 
79     val realType = () => {
80       val starter =
81         useType.getOrElse(dataRef.get())
82       if (debugMode)
83       // wrap it in an array with the priority result
84         ArrayType(StructType(Seq(StructField("salience", IntegerType), StructField("result", starter))))
85       else
86         starter
87     }
88 
89     val lazyRef = RefExpressionLazyType(dataRef, true)
90 
91     val liftLambda = (e: Expression) => FunN(Seq(lazyRef), e, usedAsLambda = true)
92 
93     val (oexpressions, indexes, triggerCount) = flattenExpressions(ruleSuite, liftLambda)
94 
95     val starter = expression(startingStruct)
96 
97     val attributes = starter.collect{
98       case a: Attribute => a // UnresolvedAttribute for classic and Attribute for connect, so it needs converting to Unresolved
99     }
100     val firstTriggerAttributes = oexpressions.head.collect {
101       case a: Attribute => a
102     }
103 
104     // Connect stops attribute resolution for some reason when the very first trigger rule is the string "true"
105     // supplying a binder to the outer scope works, starter itself doesn't and a resolved doesn't either.
106     val expressions =
107       if (firstTriggerAttributes.isEmpty)
108         Seq( InputWrapper( UnresolvedAttribute(attributes.head.name), oexpressions.head ) ) ++ oexpressions.tail
109       else
110         oexpressions
111 
112     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
113     val exprs =
114       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
115       if (compileEvals)
116         expressions.map(PassThroughCompileEvals(_))
117       else
118         expressions
119 
120     val runner =
121       if (forceRunnerEval || resolveWith.isDefined)
122         new RuleFolderRunnerEval(cleaned, starter +: exprs,
123           realType, compileEvals = compileEvals,
124           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
125           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount, extraConfig)
126       else
127         new RuleFolderRunner(cleaned, starter +: exprs,
128           realType, compileEvals = compileEvals,
129           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
130           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount, extraConfig)
131 
132     column(
133       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
134         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
135 
136         resolved.withNewChildren(resolved.children.map{
137           // replace the expr
138           case PassThroughCompileEvals(child) => NonPassThrough(child)
139           case child => NonPassThrough(child)
140         })
141       } getOrElse ZeroCodeGenWrap.wrap(runner)
142     )
143   }
