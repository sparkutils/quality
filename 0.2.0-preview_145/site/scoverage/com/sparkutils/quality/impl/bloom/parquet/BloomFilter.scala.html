203 private[bloom] trait BloomHash extends Serializable {
204   def hashStrategy: BloomFilter.HashStrategy
205 
206   val hashFunction =
207     hashStrategy match {
208       case BloomFilter.XXH64 =>
209         new XxHash()
210       case _ =>
211         throw new RuntimeException("Unsupported hash strategy")
212     }
213 
214   def cacheBuffer: ByteBuffer
215   def mask: Array[Int]
216 
217   def setMask(key: Int) = { // The following three loops are written separately so that they could be
218     // optimized for vectorization.
219     for (i <- 0 until BlockSplitBloomFilterImpl.BITS_SET_PER_BLOCK) {
220       mask(i) = key * BlockSplitBloomFilterImpl.SALT(i)
221     }
222     for (i <- 0 until BlockSplitBloomFilterImpl.BITS_SET_PER_BLOCK) {
223       mask(i) = mask(i) >>> 27
224     }
225     for (i <- 0 until BlockSplitBloomFilterImpl.BITS_SET_PER_BLOCK) {
226       mask(i) = 0x1 << mask(i)
227     }
228     mask
229   }
230 
231   def hash(value: Any): Long = {
232     value match {
233       case b: BloomFilter.Binary => return hashFunction.hashBytes(b)
234       case wa: mutable.WrappedArray[_] =>
235         val ar = wa.asInstanceOf[mutable.WrappedArray[Long]]
236         return hashFunction.hashLongs(ar.toArray)
237       case la: Array[Long] => return hashFunction.hashLongs(la)
238       case i: Integer => cacheBuffer.putInt(i)
239       case l: Long => cacheBuffer.putLong(l)
240       case f: Float => cacheBuffer.putFloat(f)
241       case d: Double => cacheBuffer.putDouble(d)
242       case s: String => return hashFunction.hashChars(s)
243       case _ => throw new RuntimeException("Parquet Bloom filter: Not supported type")
244     }
245     doHash
246   }
247 
248   def doHash = {
249     cacheBuffer.flip
250     val hashResult = hashFunction.hashByteBuffer(cacheBuffer)
251     cacheBuffer.clear
252     hashResult
253   }
254 
255 }

268 private[bloom] trait BloomLookupImpl extends Serializable with com.sparkutils.quality.BloomLookup with BloomHash {
269   def intBuffer: IntBuffer
270 
271   def findHash(hash: Long): Boolean = {
272     val numBlocks = (intBuffer.limit() * 4) / BlockSplitBloomFilterImpl.BYTES_PER_BLOCK
273     val lowHash = hash >>> 32
274     val blockIndex = ((lowHash * numBlocks) >> 32).toInt
275     val key = hash.toInt
276     // Calculate mask for the tiny Bloom filter.
277     val mask = setMask(key)
278     for (i <- 0 until BlockSplitBloomFilterImpl.BITS_SET_PER_BLOCK) {
279       if (0 == (intBuffer.get(blockIndex * (BlockSplitBloomFilterImpl.BYTES_PER_BLOCK / 4) + i) & mask(i))) return false
280     }
281     true
282   }
283 
284   def mightContain( value : Any) = {
285     val h = hash(value)
286     findHash(h)
287   }
