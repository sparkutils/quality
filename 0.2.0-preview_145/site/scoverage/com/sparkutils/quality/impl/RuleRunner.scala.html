42    * @param variableFuncGroup Defaulting to 20

44    * @return A Column representing the Quality DQ expression built from this ruleSuite

47                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
48                      extraConfig: Map[String, String] = Map.empty): Column = {
49     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
50     val flattened = flattenExpressions(ruleSuite)
51     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)

55         flattened.map(PassThroughCompileEvals)
56       else
57         flattened
58 
59     val runner =
60       if (forceRunnerEval || resolveWith.isDefined)
61         new RuleRunnerEval(cleaned, input,
62           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
63       else
64         new RuleRunner(cleaned, input,
65           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
66

69         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
70 
71         resolved.withNewChildren(resolved.children.map{
72           // replace the expr
73           case PassThroughCompileEvals(child) => NonPassThrough(child)
74           case child => NonPassThrough(child)
75         })
76       } getOrElse ZeroCodeGenWrap.wrap(runner)

213 
214     val funNames: Iterator[String] = groups.groupCalls

216     val exp = ExprCode(VariableValue(resName, ev.value.javaType), isNullVariable(resNull))

220       // copy row
221       $resultRowCopy
222       ${funNames.map { f => s"$f($paramsCall);" }.mkString("\n")}
223

290

294         topLevelCompilationUnit = true) {

296

298         val params = genParams(ctx, this)

300         // bind the rules
301         val utilsName = "com.sparkutils.quality.impl.RuleRunnerUtils"
302

306               anyToRuleResultIntGen(code, isNull),

308           )

311     }
312     setClazzSource(clazz)

314   }
