22 @ExpressionDescription(
23   usage = "if_relevant(filter, cond) - Returns passed if filter and cond are true, failed if filter is true and cond is false or null, ignoredRule if filter is false, failed if filter is null.",

25     Examples:
26       > SELECT if_relevant(id > 0, id % 2 = 0);

30   """

41   override def eval(input: InternalRow): Any = {
42     val leftValue = left.eval(input)
43     if (leftValue == null)

45     else if (RuleLogicUtils.anyToRuleResultInt(leftValue) != PassedInt)

55     }
56   }
