24 object ExpressionRunner {
25   /**
26    * Runs the ruleSuite expressions saving results as a tuple of (ruleResult: yaml, resultType: String)
27    * Supplying a ddlType triggers the output type for the expression to be that ddl type, rather than using yaml conversion.
28    * @param renderOptions provides rendering options to the underlying snake yaml implementation
29    * @param ddlType optional DDL string, when present yaml output is disabled and the output expressions must all have the same type
30    * @param name the default column name "expressionResults"
31    */
32   def apply(ruleSuite: RuleSuite, name: String = "expressionResults", renderOptions: Map[String, String] = Map.empty, ddlType: String = "",
33             variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, compileEvals: Boolean = false): Column = {
34     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
35     val expressions = flattenExpressions(ruleSuite)
36     val collectExpressions =
37       if (ddlType.isEmpty)
38         expressions.map( i => YamlEncoderExpr(i, renderOptions))
39       else
40         expressions
41     val exprs =
42       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
43       if (compileEvals)
44         collectExpressions.map(PassThroughCompileEvals)
45       else
46         collectExpressions
47 
48     val ddl_type =
49       if (ddlType.isEmpty)
50         expressionResultTypeYaml
51       else
52         DataType.fromDDL(ddlType)
53 
54     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
55 
56     ShimUtils.column(
57       if (forceRunnerEval)
58         new ExpressionRunnerEval(cleaned, exprs,
59           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
60           compileEvals = compileEvals)
61       else
62         new ExpressionRunner(cleaned, exprs,
63           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
64           compileEvals = compileEvals)
65     ).as(name)
66   }
67 }
