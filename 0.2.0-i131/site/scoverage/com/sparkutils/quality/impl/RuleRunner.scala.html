43   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
44                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false): Column = {
45     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
46     val flattened = flattenExpressions(ruleSuite)
47     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
48     val input =
49       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
50       if (compileEvals)
51         flattened.map(PassThroughCompileEvals)
52       else
53         flattened
54 
55     val runner =
56       if (forceRunnerEval || resolveWith.isDefined)
57         new RuleRunnerEval(cleaned, input,
58           compileEvals, variablesPerFunc, variableFuncGroup)
59       else
60         new RuleRunner(cleaned, input,
61           compileEvals, variablesPerFunc, variableFuncGroup)
62 
63     column(
64       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
65         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
66 
67         resolved.withNewChildren(resolved.children.map{
68           // replace the expr
69           case PassThroughCompileEvals(child) => NonPassThrough(child)
70           case child => NonPassThrough(child)
71         })
72       } getOrElse runner

294 /**
295  * Children will be rewritten by the plan, either by resolveWith, it's then re-incorporated into ruleSuite
296  *
297  * The variablePerFunc (40) and variableFuncGroup (20) parameters are chosen due to 64k method limitation,
298  * is null for all of a 1600 column row blasts the 64k jvm limit.  You may customise these values if a given rulesuite
299  * requires it.
300  *
301  * @param variablesPerFunc How many variables are in a function
302  * @param variableFuncGroup How many functions are then grouped into a new function
303  */

332   /**
333    * Instead of evaluating through the RuleSuite structure the expressions are evaluated back to back, then
334    * reincorporated via just the results afterwards before working out the overall results
335    * @param ctx
336    * @param ev
337    * @return

339   protected def doGenCodeI(outerCtx: CodegenContext, ev: ExprCode): ExprCode = {
340 
341     outerCtx.references += this
342     val ctx = QualityCodeGenUtils.clone(outerCtx)
343     // must be called before the rule gen runs
344     val params = genParams(ctx, this)
345 
346     // bind the rules
347     val ruleSuitTerm = genRuleSuiteTerm[T](ctx)._1
348     val utilsName = "com.sparkutils.quality.impl.RuleRunnerUtils"
349 
350     val res =
351       nonOutputRuleGen(ctx, this, ev, ruleSuitTerm, utilsName, realChildren, variablesPerFunc, variableFuncGroup,
352         (code: ExprValue, idx: Int) => s"com.sparkutils.quality.impl.RuleLogicUtils.anyToRuleResultInt($code)"
353       )
354 
355     val (clazz, fres) = runnerCompilation(outerCtx, params,
356       classOf[RuleRunnerBase[T]].getName, ctx, res, ev, ruleSuite.id)
357     generatorClassSource = clazz
358     fres
359   }
