32 /**
33  * Registers Quality sql functions using the defaults for registerQualityFunctions, these can be overridden without having to subclass DriverPlugin.  Functions are registered via FunctionRegistry.builtIn making them available for view creation, if they should be registered via extensions only then use the quality_force_inject_function = true configuration.
34  *
35  * It also registers plan optimiser rule's such as AsUUIDFilter, which rewrites filters with variables backed by as_uuid.
36  *
37  * Optimiser rules can be disabled via the quality_disable_optimiser_rules system environment variable. "*" disables all rules, otherwise a comma separated list of fqn class names may be used.
38  */
39 class QualitySparkExtension extends ((SparkSessionExtensions) => Unit) with Logging {
40 
41   def parseTypes: String => Option[DataType] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultParseTypes
42   def zero: DataType => Option[Any] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultZero
43   def add: DataType => Option[(Expression, Expression) => Expression] = (dataType: DataType) => com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultAdd(dataType)
44   def mapCompare: DataType => Option[(Any, Any) => Int] = (dataType: DataType) => utils.defaultMapCompare(dataType)
45   def writer: String => Unit = println
46 
47   /**
48    * uses writer to write prefix: str
49    * @param str
50    * @param prefix defaults to "Quality SparkExtensions"

52   def dump(str: String, prefix: String = "Quality SparkExtensions") = writer(s"$prefix: $str")
53 
54   /**
55    * attempts to logInfo, typically doesn't work, but also then dumps the str.  Doesn't log when testing
56    * @param str
57    */
58   def attemptLogInfo(str: String) =
59     if (!Option(System.getProperty(testingConf)).contains("testing")) {
60       logInfo(str)
61       dump(str)
62     }

64   /**
65    * Adds AsymmetricFilterExpressions for AsUUID
66    * Derived implementations should also call super if they wish to register the default rules (or specify both as extensions).
67    * These are registered after resolution is done.
68    * The first String should be the fqn classname / rule name for the rule
69    * @return
70    */
71   def optimiserRules: Seq[(String, SparkSession => Rule[LogicalPlan])] =
72     Seq((AsUUIDFilter.getClass.getName, _ => AsUUIDFilter), (IDBase64Filter.getClass.getName, _ => IDBase64Filter),
73       (FunNRewrite.getClass.getName, _ => FunNRewrite))
74 
75   override def apply(extensions: SparkSessionExtensions): Unit = {
76     val func =
77       if (com.sparkutils.quality.getConfig(forceInjectFunction, "false").toBoolean) {
78         attemptLogInfo("registering quality functions via injection - they are classed as temporary functions")
79         ShimUtils.registerFunctionViaExtension(extensions) _
80       } else {
81         attemptLogInfo("registering quality functions via builtin function registry - whilst you can use these in global views the extension must always be present")
82         ShimUtils.registerFunctionViaBuiltin _
83       }
84     com.sparkutils.quality.classicFunctions.registerQualityFunctions(parseTypes, zero, add, mapCompare, writer,
85       registerFunction = func, fromExtension = true
86     )
87     extensions.injectParser(QualityFunctionParser(_,_))
88     extensions.injectParser(ConnectCommandParsers(_,_))
89 
90     if (Testing.testing) {
91       ExtensionTesting.disableRuleResult = ""
92     }
93     val (all, disabledRules) = disabledOptimiserRules()
94     if (!all) {
95       val filteredRules = optimiserRules.filterNot(p => disabledRules.contains(p._1.trim))
96       val str = s"$disableRulesConf = $disabledRules leaving ${filteredRules.map(_._1)} remaining"
97       attemptLogInfo(str)
98       if (Testing.testing) {
99         ExtensionTesting.disableRuleResult = str
100       }
101       filteredRules.map(_._2).foreach(extensions.injectOptimizerRule _)
102     } else {
103       val str = s"All optimiser rules are disabled via $disableRulesConf"
104       attemptLogInfo(str)
105     }
106   }
107 
108 }
