14   def getContextOrSparkClassLoader: ClassLoader =

17   /**
18    * The small-int encoding for Passed that anyToRuleResultInt(Gen) accepts.
19    *
20    * anyToRuleResultInt(Gen) only treats values in [UnevaluatedRuleInt, TRUE_INT]
21    * as rule results, mapping TRUE_INT to PassedInt. Expressions whose result is
22    * re-mapped by the runners must therefore emit TRUE_INT for Passed rather than
23    * PassedInt (100000), which would fall outside the range and coerce to Failed.
24    * See SoftFailedUtils.softFail, which returns 1/-1 for the same reason.
25    */
26   protected[sparkutils] val TRUE_INT = 1
27 
28   protected[quality] def deserializeImpl[T](in: Array[Byte]): T = {
29     val os = new ObjectInputStream(new ByteArrayInputStream(in)) {
30       override def resolveClass(desc: ObjectStreamClass): Class[_] =
31         Class.forName(desc.getName, false, getContextOrSparkClassLoader)
32     }
33     val suite = os.readObject()
34     os.close()
35     suite.asInstanceOf[T]
36   }
37 
38   protected[quality] def deserialize(in: Array[Byte]): RuleSuite = deserializeImpl[RuleSuite](in)

40 
41   protected[quality] def serializeImpl[T](in: T)(f: T => T): Array[Byte] = {
42     val bos = new ByteArrayOutputStream()
43     val os = new ObjectOutputStream(bos)
44     // get rid of List's Vectors are serializable
45     os.writeObject(f(in))
46     val res = bos.toByteArray
47     os.close()
48     res
49   }

51   // sparkutils to allow tests to call
52   protected[sparkutils] def serialize(ruleSuite: RuleSuite): Array[Byte] = serializeImpl(ruleSuite)(toSeq)

56   protected[quality] def serializeGroup(ruleSuiteGroup: RuleSuiteGroup): Array[Byte] =

58 
59   def ruleResultToInt(ruleResult: RuleResult): Int =
60     ruleResult match {
61       case Failed => FailedInt
62       case SoftFailed => SoftFailedInt
63       case DisabledRule => DisabledRuleInt
64       case IgnoredRule => IgnoredRuleInt
65       case DefaultRule => DefaultRuleInt
66       case Passed => PassedInt
67       case Probability(percentage) => (percentage * PassedInt).toInt
68       case RuleResultWithProcessor(res, _) => ruleResultToInt(res)
69       case UnevaluatedRule => UnevaluatedRuleInt
70     }
