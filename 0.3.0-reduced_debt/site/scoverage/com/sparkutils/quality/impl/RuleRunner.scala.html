35   /**
36    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
37    *
38    * @param ruleSuite The Qualty RuleSuite to evaluate
39    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
40    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
41    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
42    * @param variableFuncGroup Defaulting to 20
43    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
44    * @return A Column representing the Quality DQ expression built from this ruleSuite
45    */
46   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
47                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
48                      extraConfig: Map[String, String] = Map.empty): Column = {
49     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
50     val flattened = flattenExpressions(ruleSuite)
51     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
52     val input =
53       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
54       if (compileEvals)
55         flattened.map(PassThroughCompileEvals)
56       else
57         flattened
58 
59     val runner =
60       if (forceRunnerEval || resolveWith.isDefined)
61         new RuleRunnerEval(cleaned, input,
62           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
63       else
64         new RuleRunner(cleaned, input,
65           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
66 
67     column(
68       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
69         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
70 
71         resolved.withNewChildren(resolved.children.map{
72           // replace the expr
73           case PassThroughCompileEvals(child) => NonPassThrough(child)
74           case child => NonPassThrough(child)
75         })
76       } getOrElse ZeroCodeGenWrap.wrap(runner)
77     )
78   }

193

196     val exp = ExprCode(VariableValue(resName, ev.value.javaType), isNullVariable(resNull))

199       code"""
200       // copy row
201       $resultRowCopy
202       ${funNames.map { f => s"$f($paramsCall);" }.mkString("\n")}
203

205       boolean ${exp.isNull} = false;

274         topLevelCompilationUnit = true) {

276 
277         // must be called before the rule gen runs
278         val params = genParams(ctx, this)

280         // bind the rules

286               anyToRuleResultIntGenValues(code, isNull),

288           )

291     }
