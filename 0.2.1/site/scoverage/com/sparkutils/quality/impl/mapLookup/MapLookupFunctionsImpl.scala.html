93   private def registerMap(v: VariableReference, st: StructType, col: (StructField, Int), extract: Boolean = true) = {
94     col match {
95       case (f, i) =>

102         } else {
103

105 
106           def newNameMap(name: String) = {
107             val m = new mutable.WeakHashMap[SparkSession, WeakMapEntry]()
108             hm.put(name, m)
109             m
110           }
111 
112           val nameMap = hm.get(name).fold(newNameMap(name))(identity)
113 
114           val dt = f.dataType.asInstanceOf[MapType]
115           val r = v.eval().asInstanceOf[InternalRow]
116 
117           def createEntry: Broadcast[MapData] = {
118             // copy needed to keep the map isolated and serializable
119             val m = r.getMap(i).copy()
120 
121             val b = SparkSession.active.sparkContext.broadcast(m)
122             nameMap.put(SparkSession.active, WeakMapEntry(r, b, dt.valueType))
123             b
124           }
125 
126           val b =
127             nameMap.get(SparkSession.active).fold(createEntry) {
128               cur =>
129                 if (cur.exists(r))
130                   cur.broadcast
131                 else
132                   // it has been updated as the internal row is no longer the same
133                   createEntry
134             }
135 
136           // it's a map ..
137           (b, dt.valueType)
138         }
139     }
