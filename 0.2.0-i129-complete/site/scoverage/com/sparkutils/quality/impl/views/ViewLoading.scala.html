52   /**
53    * Attempts to load all the views present in the config.  If a view is already registered in that name it will be replaced.
54    * @param viewConfigs
55    * @return the names of views which have been replaced
56    */
57   def loadViews(viewConfigs: Seq[ViewConfig]): ViewLoadResults = {
58 
59     // assume some will not load, attempt count is used to stop cycles, 2x should be enough
60     var attemptCount = 0
61     var done = false
62     var leftToProcess = viewConfigs.map(t => t.name -> t).toMap
63 
64     val replaced = mutable.Set.empty[String]
65     // if we are missing a relation and the name is declared try to load that first
66     var mapOf = leftToProcess
67     var processed = mutable.Set.empty[String]
68 
69     def processView(viewPair: ViewConfig): Unit =
70       if (attemptCount < (viewConfigs.size * 2)) {
71         try {
72           val (name, config) = (viewPair.name, viewPair)
73           if (defaultViewLookup(name)) {
74             replaced += name
75           }
76 
77           config.source.fold(identity, SparkSession.active.sql)
78            .toDF().createOrReplaceTempView(name)
79 
80           // it worked, remove it
81           leftToProcess = leftToProcess - name
82           processed = processed + name
83         } catch {
84           // 3.5 introduced two different exceptions, ParserException and ExtendedAnalysisException
85           // AnalysisException no longer has the plan
86           case e: Exception =>
87             val res = ShimUtils.tableOrViewNotFound(e).getOrElse(throw e)
88             val sql = viewPair.source.right.getOrElse("")
89 
90             res.fold(a => throw ViewLoaderAnalysisException(a, s"AnalysisException for view ${viewPair.name}: $sql", viewPair.name, sql),
91               views => {
92                 val missingNames =
93                   views.flatMap { name =>
94                     val lookupName =
95                       if (mapOf.contains(name))
96                         name
97                       else
98                         s"`$name`"
99 
100                     if (mapOf.contains(lookupName)) { // quoted must also be in the view name if it's got minus' etc., on Spark < 3.2 this will be incorrectly unquoted
101                       attemptCount += 1
102                       processView(mapOf(lookupName))
103                       None
104                     } else // not one we can actually do anything about
105                       Some(name)
106                   }
107                 if (!missingNames.isEmpty) {
108                   throw MissingViewAnalysisException(e, s"Missing relations for view ${viewPair.name}: $missingNames used in sql $sql", viewPair.name, sql, missingNames)
109                 }
110               }
111             )
112         }
113       }
114 
115     while((attemptCount < (viewConfigs.size * 2)) && !done){
116       attemptCount += 1
117       leftToProcess.headOption.fold {
118         done = true;
119       } { p =>
120         processView(p._2)
121       }
122     }
123     ViewLoadResults(replaced.toSet, !done, leftToProcess.keySet)
124   }
