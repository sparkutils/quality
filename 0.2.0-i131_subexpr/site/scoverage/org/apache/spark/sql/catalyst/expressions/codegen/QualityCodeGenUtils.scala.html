20   def clone(ctx: CodegenContext): CodegenContext = {
21 
22     val thisCtx = new CodegenContext()
23     thisCtx.INPUT_ROW = ctx.INPUT_ROW
24     thisCtx.currentVars = if (ctx.currentVars ne null) { ctx.currentVars.map(_.copy()) } else null
25     thisCtx.references.++=(ctx.references)

60   /**
61    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
62    * copied before codegen.
63    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.
64    */
65   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
66     val expressions = ShimUtils.copyStateful(dirtyExpressions)
67     import ctx._
68     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
69 
70     // Add each expression tree and compute the common subexpressions.
71     expressions.foreach(equivalentExpressions.addExprTree(_))
72     var subexprFunctions = ""
73     // Get all the expressions that appear at least twice and set up the state for subexpression
74     // elimination.
75     val commonExprs = equivalentExpressions.getCommonSubexpressions
76     commonExprs.foreach { expr =>
77       val fnName = freshName("subExpr")
78       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
79       val value = addMutableState(javaType(expr.dataType), "subExprValue")
80 
81       // Generate the code for this expression tree and wrap it in a function.
82       val eval = expr.genCode(ctx)
83       val fn =
84         s"""
85            |private void $fnName(InternalRow $INPUT_ROW) {
86            |  ${eval.code}
87            |  $isNull = ${eval.isNull};
88            |  $value = ${eval.value};
89            |}
90            """.stripMargin
91 
92       // Add a state and a mapping of the common subexpressions that are associate with this
93       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
94       // when it is code generated. This decision should be a cost based one.
95       //
96       // The cost of doing subexpression elimination is:
97       //   1. Extra function call, although this is probably *good* as the JIT can decide to
98       //      inline or not.
99       // The benefit doing subexpression elimination is:
100       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
101       //      above.
102       //   2. Less code.
103       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
104       // at least two nodes) as the cost of doing it is expected to be low.
105 
106       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
107       subexprFunctions += subExprCode
108       val state = SubExprEliminationState(
109         ExprCode(code"$subExprCode",
110           JavaCode.isNullGlobal(isNull),
111           JavaCode.global(value, expr.dataType)))
112       subExprEliminationExprs += ExpressionEquals(expr) -> state
113     }
114 
115     subexprFunctions
116   }
