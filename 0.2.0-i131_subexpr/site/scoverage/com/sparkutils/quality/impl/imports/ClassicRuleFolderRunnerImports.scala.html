20    *

25    *
26    * @param ruleSuite The ruleSuite with runOnPassProcessors
27    * @param startingStruct This struct is passed to the first matching rule, ideally you would use the spark dsl struct function to refer to existing columns
28    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
29    * @param debugMode When debugMode is enabled the resultDataType is wrapped in Array of (salience, result) pairs to ease debugging
30    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this.

34    * @param forceTriggerEval Defaulting to false, passing true forces each trigger expression to be compiled (compileEvals) and used in place, false instead expands the trigger in-line giving possible performance boosts based on JIT

66    * @param useType In the case you must use select and can't use withColumn you may provide a type directly to stop the NPE
67    * @return A Column representing the QualityRules expression built from this ruleSuite
68    */
69   def ruleFolderRunnerClassic(ruleSuite: RuleSuite, startingStruct: Column, compileEvals: Boolean = false,
70                        debugMode: Boolean = false, resolveWith: Option[DataFrame] = None, variablesPerFunc: Int = 40,
71                        variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, useType: Option[StructType] = None,
72                        forceTriggerEval: Boolean = false): Column = {
73     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
74 
75     // needed to resolve variables -- this changes between invocation and stops the type checks.  In the test case it's subcode that is on one type nullable and the other not
76     val dataRef = new AtomicReference[DataType]()
77 
78     val realType = () => {
79       val starter =
80         useType.getOrElse(dataRef.get())
81       if (debugMode)
82       // wrap it in an array with the priority result
83         ArrayType(StructType(Seq(StructField("salience", IntegerType), StructField("result", starter))))
84       else
85         starter
86     }
87 
88     val lazyRef = RefExpressionLazyType(dataRef, true)
89 
90     val liftLambda = (e: Expression) => FunN(Seq(lazyRef), e, usedAsLambda = true)
91 
92     val (oexpressions, indexes, triggerCount) = flattenExpressions(ruleSuite, liftLambda)
93 
94     val starter = expression(startingStruct)
95 
96     val attributes = starter.collect{
97       case a: Attribute => a // UnresolvedAttribute for classic and Attribute for connect, so it needs converting to Unresolved
98     }
99     val firstTriggerAttributes = oexpressions.head.collect {
100       case a: Attribute => a
101     }
102 
103     // Connect stops attribute resolution for some reason when the very first trigger rule is the string "true"
104     // supplying a binder to the outer scope works, starter itself doesn't and a resolved doesn't either.
105     val expressions =
106       if (firstTriggerAttributes.isEmpty)
107         Seq( InputWrapper( UnresolvedAttribute(attributes.head.name), oexpressions.head ) ) ++ oexpressions.tail
108       else
109         oexpressions
110 
111     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
112     val exprs =
113       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
114       if (compileEvals)
115         expressions.map(PassThroughCompileEvals(_))
116       else
117         expressions
118 
119     val runner =
120       if (forceRunnerEval || resolveWith.isDefined)
121         new RuleFolderRunnerEval(cleaned, starter +: exprs,
122           realType, compileEvals = compileEvals,
123           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
124           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount)
125       else
126         new RuleFolderRunner(cleaned, starter +: exprs,
127           realType, compileEvals = compileEvals,
128           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
129           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount)
130 
131     column(
132       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
133         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
134 
135         resolved.withNewChildren(resolved.children.map{
136           // replace the expr
137           case PassThroughCompileEvals(child) => NonPassThrough(child)
138           case child => NonPassThrough(child)
139         })
140       } getOrElse runner
141     )
142   }
