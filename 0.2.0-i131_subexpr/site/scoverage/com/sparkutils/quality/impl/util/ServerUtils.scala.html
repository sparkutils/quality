302   def stripBrackets(v: VariableValue): (String, String) = {
303     val openb = v.toString().indexOf("[")
304     if (openb == -1)
305       (v.variableName, "")
306     else
307       (v.variableName.dropRight(v.length - openb), v.variableName.drop(openb))
308   }
309 
310   def formatParams(ctx: CodegenContext, a: Seq[ExprValue], callsKeepArrays: Boolean = false): ParameterInformation = {
311     // filter out any top level arrays, the input is a set, so params need the same order
312     val ordered = a.flatMap {
313       case a: VariableValue => Some(a)
314       case _ => None
315     }
316 
317     val pairs = ordered.map { v =>
318       val (stripped, arrayInName) = stripBrackets(v)
319 
320       val (typ, array) =
321         if (v.javaType.isArray)
322           (s"${v.javaType.getComponentType.getName}", "[]")
323         else if (v.javaType.isPrimitive)
324           (v.javaType.toString, arrayInName.replaceAll("[^\\[\\]]",""))
325         else
326           (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
327 
328       (s"$typ$array", stripped, v.javaType)
329     }
330 
331     ParameterInformation(pairs.map {
332       case (typ, stripped, _) =>
333 
334         s"$typ $stripped"
335       }.mkString(", ")
336       , ordered.map(v =>
337         if (v.javaType.isArray && callsKeepArrays)
338           v.variableName
339         else
340           stripBrackets(v)._1
341       ).mkString(", "), ordered.size, pairs)
342   }
