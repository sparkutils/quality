31   protected[quality] def inplace(ruleResult: RuleResult, currentResult: RuleResult, probablePass: Double): RuleResult =
32     ruleResult match {

34       case RuleResultWithProcessor(ruleResult, _) => inplace(ruleResult, currentResult, probablePass)

41     }

54     }
