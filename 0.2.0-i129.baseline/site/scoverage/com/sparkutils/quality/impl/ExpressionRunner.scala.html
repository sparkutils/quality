23 object ExpressionRunner {
24   /**
25    * Runs the ruleSuite expressions saving results as a tuple of (ruleResult: yaml, resultType: String)
26    * Supplying a ddlType triggers the output type for the expression to be that ddl type, rather than using yaml conversion.
27    * @param renderOptions provides rendering options to the underlying snake yaml implementation
28    * @param ddlType optional DDL string, when present yaml output is disabled and the output expressions must all have the same type
29    * @param name the default column name "expressionResults"
30    */
31   def apply(ruleSuite: RuleSuite, name: String = "expressionResults", renderOptions: Map[String, String] = Map.empty,
32             ddlType: String = "", variablesPerFunc: Int = 40, variableFuncGroup: Int = 20,
33             forceRunnerEval: Boolean = false, compileEvals: Boolean = false,
34             extraConfig: Map[String, String] = Map.empty): Column = {
35     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
36     val expressions = flattenExpressions(ruleSuite)
37     val collectExpressions =
38       if (ddlType.isEmpty)
39         expressions.map( i => YamlEncoderExpr(i, renderOptions))
40       else
41         expressions
42     val exprs =
43       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
44       if (compileEvals)
45         collectExpressions.map(PassThroughCompileEvals)
46       else
47         collectExpressions
48 
49     val ddl_type =
50       if (ddlType.isEmpty)
51         expressionResultTypeYaml
52       else
53         DataType.fromDDL(ddlType)
54 
55     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
56 
57     ShimUtils.column(
58       if (forceRunnerEval)
59         new ExpressionRunnerEval(cleaned, exprs,
60           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
61           compileEvals = compileEvals, extraConfig = extraConfig)
62       else
63         new ExpressionRunner(cleaned, exprs,
64           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
65           compileEvals = compileEvals, extraConfig = extraConfig)
66     ).as(name)
67   }
68 }

72   protected[quality] def expressionsResultToRow[R](ruleSuiteResult: GeneralExpressionsResult[R]): InternalRow =
73     InternalRow(
74       packId(ruleSuiteResult.id),
75       ArrayBasedMapData(
76         ruleSuiteResult.ruleSetResults, packId _, (a: Any) => {
77           val v = a.asInstanceOf[Map[VersionedId, GeneralExpressionResult]]
78           ArrayBasedMapData(
79             v, packId _, (a: Any) => a match {
80               case r: GeneralExpressionResult =>
81                 InternalRow(UTF8String.fromString( r.ruleResult ), UTF8String.fromString( r.resultDDL) )
82               case s: String =>  UTF8String.fromString( s )
83               case _ => a // handle nulls *and* R's
84             }
85           )
86         }
87       )
88     )
