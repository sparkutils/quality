34    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
35    *
36    * @param ruleSuite The Qualty RuleSuite to evaluate
37    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
38    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
39    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
40    * @param variableFuncGroup Defaulting to 20
41    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
42    * @return A Column representing the Quality DQ expression built from this ruleSuite
43    */
44   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
45                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
46                      extraConfig: Map[String, String] = Map.empty): Column = {
47     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
48     val flattened = flattenExpressions(ruleSuite)
49     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
50     val input =
51       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
52       if (compileEvals)
53         flattened.map(PassThroughCompileEvals)
54       else
55         flattened
56 
57     val runner =
58       if (forceRunnerEval || resolveWith.isDefined)
59         new RuleRunnerEval(cleaned, input,
60           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
61       else
62         new RuleRunner(cleaned, input,
63           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
64 
65     column(
66       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
67         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
68 
69         resolved.withNewChildren(resolved.children.map{
70           // replace the expr
71           case PassThroughCompileEvals(child) => NonPassThrough(child)
72           case child => NonPassThrough(child)
73         })
74       } getOrElse runner
75     )
76   }

179   def nonOutputRuleGen[T: ClassTag](ctx: CodegenContext, runner: Runner, ev: ExprCode, utilsName: String,
180                        realChildren: Seq[Expression], variablesPerFunc: Int, variableFuncGroup: Int,
181                        resultF: (ExprValue, Int) => String, extraConfig: Map[String, String],
182                        ruleRunnerExpressionIdx: Int, applyResult: String = "applyResult"
183                       ): ExprCode = {
184     val paramInfo = genParams(ctx, runner)
185     import paramInfo._
186 
187     val resTerms = resultRowTerms(ctx, ruleRunnerExpressionIdx)
188     import resTerms._
189 
190     val inPlaceOffsets = runner.inPlaceArrayOffsets(ctx, resultRow, ruleRunnerExpressionIdx)
191 
192     val allExpr = realChildren.zipWithIndex.map { case (child, idx) =>
193       val eval = child.genCode(ctx)
194 
195       val converted =
196         code"""${eval.code}\n
197 
198             ${inPlaceOffsets.offsets(idx).apply(resultF(eval.value, idx))}
199              """
200 
201       (Trigger(child, idx, 0), converted)
202     }
203 
204     val funNames: Iterator[String] =
205       RuleRunnerUtils.generateFunctionGroups(ctx, allExpr,
206         variablesPerFunc, variableFuncGroup, paramsDef, paramsCall, extraConfig)._1
207 
208 
209     val resName = ctx.freshName("result")
210     val resNull = ctx.freshName("isNull")
211 
212     val exp = ExprCode(VariableValue(resName, ev.value.javaType), isNullVariable(resNull))
213 
214     val res = exp.copy(code =
215       code"""
216       // copy row
217       $resultRowCopy
218       ${funNames.map { f => s"$f($paramsCall);" }.mkString("\n")}
219 
220       InternalRow ${exp.value} = $resultRow;
221       boolean ${exp.isNull} = false;
222       """
223     )
224 
225     res
