68    * Used by codegen

70   def createDefaultRuleResult(): InternalRow =
71     InternalRow(packTheId(ruleSuite.id), defaultOverallResult,
72       ArrayBasedMapData(
73         ruleSuite.ruleSets.map{
74           ruleSet =>
75             packTheId(ruleSet.id) -> InternalRow(defaultOverallResult,

77                 ruleSet.rules.map( r => packTheId(r.id) -> defaultRuleResult).toMap
78               ))
79         }.toMap

82 
83   /**
84    * rolls the overalls up in place - must be genericarraydata / arraybasedmap data with a copy from createDefaultRuleResult
85    * used by codegen
86    */
87   def applyResult(level1: Int, level2: Int, result: InternalRow, ruleResult: Int): Unit = {
88     val sar = result.getMap(2).asInstanceOf[ArrayBasedMapData]
89     // update result directly
90     val sv = sar.valueArray.asInstanceOf[GenericArrayData]
91     val struct = sv.getStruct(level1, 2)
92     struct.getMap(1).valueArray.asInstanceOf[GenericArrayData].update(level2, ruleResult)
93 
94     // processOverall
95     val cur = struct.getInt(0)
96     val nr = defaultOverallProcessor(ruleResult, cur)
97     struct.update(0, nr)
98     result.update(1, defaultOverallProcessor(nr, result.getInt(1)))
99   }
100 
101   def inPlaceArrayOffsets(ctx: CodegenContext, resultRow: String, ruleRunnerExpressionIdx: Int): InPlaceOffsets = {
102     val className = this.getClass.getName
103     val runner = ctx.freshName("runner")
104     ctx.addImmutableStateIfNotExists(className, runner,
105       v => s"$v = (($className)references[$ruleRunnerExpressionIdx]);")
106 
107     InPlaceOffsets(ruleSuite.ruleSets.zipWithIndex.flatMap{
108       case (ruleSet, level1) =>
109         ruleSet.rules.zipWithIndex.map{
110           case (_, level2) =>
111             (result: String) =>
112               code"""
113                $runner.applyResult($level1, $level2, $resultRow, $result);
114                 """
115         }
116     }, code"", code"")
117   }
