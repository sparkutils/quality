80   /**
81    * Based on the Spark subexpressionElimination code.  Following the approach from Spark any statefuls are
82    * copied before codegen.
83    * This, alongside forcing initialise to be called, as part of #131 require the test cases to change.

85   def nonWholeStageSubexpressionElimination(ctx: CodegenContext, dirtyExpressions: Seq[Expression]): String = {
86     val expressions = ShimUtils.copyStateful(dirtyExpressions)
87     import ctx._
88     val equivalentExpressions: EquivalentExpressions = new EquivalentExpressions
89 
90     // Add each expression tree and compute the common subexpressions.
91     expressions.foreach(equivalentExpressions.addExprTree(_))
92     var subexprFunctions = ""
93     // Get all the expressions that appear at least twice and set up the state for subexpression
94     // elimination.
95     val commonExprs = QualityExprUtils.getAllEquivalentExprs(equivalentExpressions)
96     commonExprs.foreach { expr =>
97       val fnName = freshName("subExpr")
98       val isNull = addMutableState(JAVA_BOOLEAN, "subExprIsNull")
99       val value = addMutableState(javaType(expr.dataType), "subExprValue")
100 
101       // Generate the code for this expression tree and wrap it in a function.
102       val eval = expr.genCode(ctx)
103       val fn =
104         s"""
105            |private void $fnName(InternalRow $INPUT_ROW) {
106            |  ${eval.code}
107            |  $isNull = ${eval.isNull};
108            |  $value = ${eval.value};
109            |}
110            """.stripMargin
111 
112       // Add a state and a mapping of the common subexpressions that are associate with this
113       // state. Adding this expression to subExprEliminationExprMap means it will call `fn`
114       // when it is code generated. This decision should be a cost based one.
115       //
116       // The cost of doing subexpression elimination is:
117       //   1. Extra function call, although this is probably *good* as the JIT can decide to
118       //      inline or not.
119       // The benefit doing subexpression elimination is:
120       //   1. Running the expression logic. Even for a simple expression, it is likely more than 3
121       //      above.
122       //   2. Less code.
123       // Currently, we will do this for all non-leaf only expression trees (i.e. expr trees with
124       // at least two nodes) as the cost of doing it is expected to be low.
125 
126       val subExprCode = s"${addNewFunction(fnName, fn)}($INPUT_ROW);"
127       subexprFunctions += subExprCode
128       val state = QualityExprUtils.state(
129         ExprCode(code"$subExprCode",
130           JavaCode.isNullGlobal(isNull),
131           JavaCode.global(value, expr.dataType)))
132       QualityExprUtils.addSubExpr(ctx, expr, state)
133     }
134 
135     subexprFunctions
136   }
