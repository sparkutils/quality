23   // if they are we'll blow the stack, there isn't a valid reason for trying to call it as part of a rule
24   private def checkNestedCallsInSuite(ruleSuite: RuleSuite): Unit = {
25 
26     def checkRuleSuiteHasProcess(rule: String, ruleId: Id, typ: String): Unit =
27       if (rule.contains(process_if_attribute_missing_name)) {
28         qualityException(s"$process_if_attribute_missing_name should not be used in RuleSuite expressions but was found in $typ with Id $ruleId with text $rule")
29       }
30 
31     // unlike classic rulesuite calls these will still have rule text
32     ruleSuite.ruleSets.foreach(ruleSet => ruleSet.rules.map { rule =>
33       rule.expression match {
34         case h: HasRuleText =>
35           checkRuleSuiteHasProcess(h.rule, rule.id, "trigger rule")
36       }
37       if (rule.runOnPassProcessor ne NoOpRunOnPassProcessor.noOp) {
38         checkRuleSuiteHasProcess(rule.runOnPassProcessor.rule, rule.runOnPassProcessor.id, "output expression")
39       }
40     })
41     ruleSuite.lambdaFunctions.foreach{
42       l =>
43         checkRuleSuiteHasProcess(l.rule, l.id, "lambda function")
44     }
45   }

47   protected[quality] def registerProcessIfAttributeMissingForAgnostic(registerFunction: (String, Seq[Expression] => Expression) => Unit): Unit = {
48     // parse the rulesuite, call the functions with structs, set a new variable, probably needs to be direct
49     registerFunction(process_if_attribute_missing_name, {
50       case Seq(OfRuleSuite(ruleSuite), ddl, name) =>
51         val s = defaultParseTypes(getString(ddl, 1)).
52           collect {case s:StructType => s}.
53           getOrElse(qualityException(process_if_attribute_missing_name + " Did not get a struct dll for param 2"))
54 
55         checkNestedCallsInSuite(ruleSuite)
56 
57         // call for side effect
58         com.sparkutils.quality.validate(s, ruleSuite)
59 
60         // the resulting rulesuite will have the corrected unresolved expressions already available,
61         // the raw sql remains untouched
62         val r = com.sparkutils.quality.processIfAttributeMissing(ruleSuite, s)
63 
64         // adjust for name sensitivity settings
65         val aname = {
66           val tn = getString(name, 2)
67 
68           // unlike the recursive code, testing setup is not worth it
69           // $COVERAGE-OFF$
70           val nc =
71             if (SQLConf.get.caseSensitiveAnalysis)
72               tn
73             else

75           // $COVERAGE-ON$
76

78         }
