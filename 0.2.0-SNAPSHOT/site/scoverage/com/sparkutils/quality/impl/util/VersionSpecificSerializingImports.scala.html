197    * Returns a specific RuleSuite from available rule suites
198    * @param ds

200    * @return
201    */
202   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
203     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
204       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSuiteVersion) ).map{
205         p =>
206           RuleSet(p._1, p._2.map{ cr =>
207             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
208               runOnPassProcessor =
209                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
210                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
211                   getOrElse(NoOpRunOnPassProcessor.noOp)
212             )
213           })
214       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
215         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
216       }})
217   }
