65                             globalOutputExpressionSuites: Option[Dataset[Id]]): Option[org.apache.spark.sql.DataFrame] = {
66     val rname = uniqueName()

206   /**
207    * Returns a specific RuleSuite from available rule suites
208    * @param ds
209    * @param id
210    * @return
211    */
212   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
213     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
214       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSuiteVersion) ).map{
215         p =>
216           RuleSet(p._1, p._2.map{ cr =>
217             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
218               runOnPassProcessor =
219                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
220                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
221                   getOrElse(NoOpRunOnPassProcessor.noOp)
222             )
223           })
224       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
225         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
226       }})
227   }
