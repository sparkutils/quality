71     val rname = uniqueName()

74     val lfname = registerTempViewNameFromDS(lambdaFunctionRows)

217   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
218     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
219       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSuiteVersion) ).map{
220         p =>
221           RuleSet(p._1, p._2.map{ cr =>
222             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
223               runOnPassProcessor =
224                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
225                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
226                   getOrElse(NoOpRunOnPassProcessor.noOp)
227             )
228           })
229       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
230         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
231       }})
