160   /**
161    * Returns a specific RuleSuite from available rule suites
162    * @param ds
163    * @param id
164    * @return
165    */
166   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
167     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
168       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSuiteVersion) ).map{
169         p =>
170           RuleSet(p._1, p._2.map{ cr =>
171             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
172               runOnPassProcessor =
173                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
174                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
175                   getOrElse(NoOpRunOnPassProcessor.noOp)
176             )
177           })
178       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
179         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
180       }})
181   }
