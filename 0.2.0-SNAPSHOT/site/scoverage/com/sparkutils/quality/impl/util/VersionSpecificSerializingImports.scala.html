199    * Returns a specific RuleSuite from available rule suites
200    * @param ds
201    * @param id
202    * @return
203    */
204   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
205     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
206       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSuiteVersion) ).map{
207         p =>
208           RuleSet(p._1, p._2.map{ cr =>
209             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
210               runOnPassProcessor =
211                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
212                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
213                   getOrElse(NoOpRunOnPassProcessor.noOp)
214             )
215           })
216       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
217         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
218       }})
219   }
