97    * Same as functions without the wrapping Column

101   def expr(rule: String): Expression = {
102     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
103       newParser()
104     }
105     /*
106     attempt for a simple expression, then try a plan with exactly one output row.
107     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
108      */
109     val rawExpr =
110       try {
111         parser.parseExpression(rule)
112       } catch {
113         case ot: Throwable => // e.g. suitable for output expressions
114           try {
115             ScalarSubquery(parser.parsePlan(rule))
116           } catch {
117             case _: Throwable =>
118               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
119               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
120               val r = rule.split("->")
121               val wrapped =
122                 if (r.size == 2)
123                   s"${r(0)} -> ( ${r(1)} )"
124                 else
125                   s"( ${r(0)} )"
126 
127               try {
128                 parser.parseExpression(wrapped)
129               } catch {
130                 case _: Throwable => throw ot // if this didn't work return the original error
131               }
132           }
133       }
134 
135     val res =
136       rawExpr match {
137         case l: SparkLambdaFunction if hasSubQuery(l) =>
138           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
139           val names = l.arguments.map(a => a.name -> a).toMap
140           l.transformUp {
141             case s: SubqueryExpression =>
142               s.withNewPlan( s.plan.transform{
143               case snippet =>
144                 snippet.transformAllExpressions {
145                   case a: UnresolvedAttribute =>
146                     names.get(a.name).map(lamVar => lamVar).getOrElse(a)
147                 }
148               }
149             )
150           }
151         case _ => rawExpr
152       }
153 
154     res.transformUp {
155       case s: SubqueryExpression => SubQueryWrapper(Seq(s))
156     }
157   }
