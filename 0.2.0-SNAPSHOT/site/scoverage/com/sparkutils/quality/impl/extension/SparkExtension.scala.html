22 /**
23  * Registers Quality sql functions using the defaults for registerQualityFunctions, these can be overridden without having to subclass DriverPlugin.  Functions are registered via FunctionRegistry.builtIn making them available for view creation, if they should be registered via extensions only then use the quality_force_inject_function = true configuration.
24  *
25  * It also registers plan optimiser rule's such as AsUUIDFilter, which rewrites filters with variables backed by as_uuid.
26  *
27  * Optimiser rules can be disabled via the quality_disable_optimiser_rules system environment variable. "*" disables all rules, otherwise a comma separated list of fqn class names may be used.
28  */
29 class QualitySparkExtension extends ((SparkSessionExtensions) => Unit) with Logging {
30 
31   def parseTypes: String => Option[DataType] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultParseTypes
32   def zero: DataType => Option[Any] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultZero
33   def add: DataType => Option[(Expression, Expression) => Expression] = (dataType: DataType) => com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultAdd(dataType)
34   def mapCompare: DataType => Option[(Any, Any) => Int] = (dataType: DataType) => utils.defaultMapCompare(dataType)
35   def writer: String => Unit = println
36 
37   /**
38    * uses writer to write prefix: str
39    * @param str
40    * @param prefix defaults to "Quality SparkExtensions"
41    */
42   def dump(str: String, prefix: String = "Quality SparkExtensions") = writer(s"$prefix: $str")
43 
44   /**
45    * attempts to logInfo, typically doesn't work, but also then dumps the str.  Doesn't log when testing
46    * @param str
47    */
48   def attemptLogInfo(str: String) =
49     if (!Option(System.getProperty(testingConf)).contains("testing")) {
50       logInfo(str)
51       dump(str)
52     }
53 
54   /**
55    * Adds AsymmetricFilterExpressions for AsUUID
56    * Derived implementations should also call super if they wish to register the default rules (or specify both as extensions).
57    * These are registered after resolution is done.
58    * The first String should be the fqn classname / rule name for the rule
59    * @return
60    */
61   def optimiserRules: Seq[(String, SparkSession => Rule[LogicalPlan])] =
62     Seq((AsUUIDFilter.getClass.getName, _ => AsUUIDFilter), (IDBase64Filter.getClass.getName, _ => IDBase64Filter),
63       (FunNRewrite.getClass.getName, _ => FunNRewrite))
64 
65   override def apply(extensions: SparkSessionExtensions): Unit = {
66     val func =
67       if (com.sparkutils.quality.getConfig(forceInjectFunction, "false").toBoolean) {
68         attemptLogInfo("registering quality functions via injection - they are classed as temporary functions")
69         ShimUtils.registerFunctionViaExtension(extensions) _
70       } else {
71         attemptLogInfo("registering quality functions via builtin function registry - whilst you can use these in global views the extension must always be present")
72         ShimUtils.registerFunctionViaBuiltin _
73       }
74     com.sparkutils.quality.classicFunctions.registerQualityFunctions(parseTypes, zero, add, mapCompare, writer,
75       registerFunction = func, fromExtension = true
76     )
77     extensions.injectParser(QualityFunctionParser(_,_))
78     extensions.injectParser(ConnectCommandParsers(_,_))
79 
80     if (Testing.testing) {
81       ExtensionTesting.disableRuleResult = ""
82     }
83     val disableConf = com.sparkutils.quality.getConfig(disableRulesConf)
84     if (disableConf != "*") {
85       val disabledRules = disableConf.split(",").map(_.trim).toSet
86       val filteredRules = optimiserRules.filterNot(p => disabledRules.contains(p._1.trim))
87       val str = s"$disableRulesConf = $disabledRules leaving ${filteredRules.map(_._1)} remaining"
88       attemptLogInfo(str)
89       if (Testing.testing) {
90         ExtensionTesting.disableRuleResult = str
91       }
92       filteredRules.map(_._2).foreach(extensions.injectOptimizerRule _)
93     } else {
94       val str = s"All optimiser rules are disabled via $disableRulesConf"
95       attemptLogInfo(str)
96     }
97   }
98 
99 }
