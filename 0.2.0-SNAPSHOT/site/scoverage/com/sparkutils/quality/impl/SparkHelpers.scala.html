80   /**
81    * Assuming MapData entries are stable, as they are array backed, over a dataset.  When the position of 'what' is not in the cachedPositions a full search is performed with the new location being returned in addition.
82    *
83    * @param mapData the source mapdata to search through.
84    * @param cachedPositions already known positions to check for what
85    * @param what compared against the keyArray to obtain the position
86    * @param getValue obtain the value from the mapData, with any necessary conversions, against the index parameter
87    * @tparam T
88    * @return (result from getValue or null when not found, either cachedPositions or cachedPositions and a newly found position)
89    */
90   def getMapEntry[T <: AnyRef](mapData: MapData, cachedPositions: Seq[Int], what: Any)(getValue: Int => T): (T, Seq[Int]) = {
91     val keys = mapData.keyArray()
92     val n = keys.numElements()
93     val acc = (idx: Int) => keys.getLong(idx)
94 
95     def search() = {
96       var i = 0
97       var found = false
98       while (i < n && !found) {
99         if (acc(i) == what) {
100           found = true
101         } else {
102           i += 1
103         }
104       }
105       if (!found)
106         -1
107       else
108         i
109     }
110 
111     def withCached() =
112       cachedPositions.find { i =>
113         if (i < n)
114           acc(i) == what
115         else
116           false
117       }
118 
119     withCached().map(i => (getValue(i), cachedPositions))
120       .getOrElse {
121         val r = search()
122         if (r == -1)
123           (null.asInstanceOf[T], cachedPositions)
124         else
125           (getValue(r), cachedPositions :+ r)
126       }
127   }
