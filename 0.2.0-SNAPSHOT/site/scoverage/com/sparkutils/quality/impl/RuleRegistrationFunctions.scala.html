94     dataType match {
95       case _: MapType => Some((left, right) => MapMerge(Seq(left, right), (dataType: DataType) => defaultAdd(dataType, extension)))
96       case _: IntegerType | LongType | DoubleType =>
97         Some((left, right) => add(left, right, null))
98       case a: DecimalType =>
99         Some((left, right) => add(left, right, a))
100       case _ => extension(dataType)

137   def registerWithChecks(registerFunction: (String, Seq[Expression] => Expression) => Unit, name: String, argsf: Seq[Expression] => Expression, paramNumbers: Set[Int] = Set.empty, minimum: Int = -1) = {
138     val create =
139       if (paramNumbers.isEmpty && minimum == -1 )
140         argsf
141       else
142         (exps: Seq[Expression]) => {
143           if ((paramNumbers.nonEmpty && !paramNumbers.contains(exps.size)) || (minimum > exps.size)) {
144             val sizeerr =
145               if (paramNumbers.nonEmpty)
146                 s"Valid parameter counts are ${paramNumbers.mkString(", ")}"
147               else
148                 s"A minimum of $minimum parameters is required."
149             throw QualityException(s"Wrong number of arguments provided to Quality function $name, received ${exps.size}. $sizeerr")
150           }
151           argsf(exps)
152         }
153     registerFunction(name, create)
154     if (!mustKeepNames(name)) {
155       registerFunction(name.replaceAll("_",""), create)
156     }
157   }
