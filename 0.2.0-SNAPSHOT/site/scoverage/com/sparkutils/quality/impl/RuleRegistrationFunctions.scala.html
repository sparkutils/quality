96     dataType match {
97       case _: MapType => Some((left, right) => MapMerge(Seq(left, right), (dataType: DataType) => defaultAdd(dataType, extension)))
98       case _: IntegerType | LongType | DoubleType =>
99         Some((left, right) => add(left, right, null))
100       case a: DecimalType =>
101         Some((left, right) => add(left, right, a))
102       case _ => extension(dataType)
103     }
