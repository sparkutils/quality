95     dataType match {
96       case _: MapType => Some((left, right) => MapMerge(Seq(left, right), (dataType: DataType) => defaultAdd(dataType, extension)))
97       case _: IntegerType | LongType | DoubleType =>
98         Some((left, right) => add(left, right, null))
99       case a: DecimalType =>
100         Some((left, right) => add(left, right, a))
101       case _ => extension(dataType)
102     }

137     val create =
138       if (paramNumbers.isEmpty && minimum == -1 )
139         argsf
140       else
141         (exps: Seq[Expression]) => {
142           if ((paramNumbers.nonEmpty && !paramNumbers.contains(exps.size)) || (minimum > exps.size)) {
143             val sizeerr =
144               if (paramNumbers.nonEmpty)
145                 s"Valid parameter counts are ${paramNumbers.mkString(", ")}"
146               else
147                 s"A minimum of $minimum parameters is required."
148             throw QualityException(s"Wrong number of arguments provided to Quality function $name, received ${exps.size}. $sizeerr")
149           }
150           argsf(exps)
151         }
152     registerFunction(name, create)
153     if (!mustKeepNames(name)) {
154       registerFunction(name.replaceAll("_",""), create)
155     }
