22 object ExpressionRunner {
23   /**
24    * Runs the ruleSuite expressions saving results as a tuple of (ruleResult: yaml, resultType: String)
25    * Supplying a ddlType triggers the output type for the expression to be that ddl type, rather than using yaml conversion.
26    * @param renderOptions provides rendering options to the underlying snake yaml implementation
27    * @param ddlType optional DDL string, when present yaml output is disabled and the output expressions must all have the same type
28    * @param name the default column name "expressionResults"
29    */
30   def apply(ruleSuite: RuleSuite, name: String = "expressionResults", renderOptions: Map[String, String] = Map.empty, ddlType: String = "",
31             variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, compileEvals: Boolean = false): Column = {
32     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
33     val expressions = flattenExpressions(ruleSuite)
34     val collectExpressions =
35       if (ddlType.isEmpty)
36         expressions.map( i => YamlEncoderExpr(i, renderOptions))
37       else
38         expressions
39     val exprs =
40       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
41       if (compileEvals)
42         collectExpressions.map(PassThroughCompileEvals)
43       else
44         collectExpressions
45 
46     val ddl_type =
47       if (ddlType.isEmpty)
48         expressionResultTypeYaml
49       else
50         DataType.fromDDL(ddlType)
51 
52     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
53 
54     ShimUtils.column(
55       if (forceRunnerEval)
56         new ExpressionRunnerEval(cleaned, exprs,
57           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
58           compileEvals = compileEvals)
59       else
60         new ExpressionRunner(cleaned, exprs,
61           ddl_type, variablesPerFunc = variablesPerFunc, variableFuncGroup = variableFuncGroup,
62           compileEvals = compileEvals)
63     ).as(name)
64   }
65 }

95     val ruleSetRes = Array.ofDim[ArrayBasedMapData](ruleSuiteArrays.ruleSetIds.length)
