51   /**
52    * Attempts to load all the views present in the config.  If a view is already registered in that name it will be replaced.
53    * @param viewConfigs
54    * @return the names of views which have been replaced
55    */
56   def loadViews(viewConfigs: Seq[ViewConfig]): ViewLoadResults = {
57 
58     // assume some will not load, attempt count is used to stop cycles, 2x should be enough
59     var attemptCount = 0
60     var done = false
61     var leftToProcess = viewConfigs.map(t => t.name -> t).toMap
62 
63     val replaced = mutable.Set.empty[String]
64     // if we are missing a relation and the name is declared try to load that first
65     var mapOf = leftToProcess
66     var processed = mutable.Set.empty[String]
67 
68     def processView(viewPair: ViewConfig): Unit =
69       if (attemptCount < (viewConfigs.size * 2)) {
70         try {
71           val (name, config) = (viewPair.name, viewPair)
72           if (defaultViewLookup(name)) {
73             replaced += name
74           }
75 
76           config.source.fold(identity, SparkSession.active.sql)
77            .toDF().createOrReplaceTempView(name)
78 
79           // it worked, remove it
80           leftToProcess = leftToProcess - name
81           processed = processed + name
82         } catch {
83           // 3.5 introduced two different exceptions, ParserException and ExtendedAnalysisException
84           // AnalysisException no longer has the plan
85           case e: Exception =>
86             val res = ShimUtils.tableOrViewNotFound(e).getOrElse(throw e)
87             val sql = viewPair.source.right.getOrElse("")
88 
89             res.fold(a => throw ViewLoaderAnalysisException(a, s"AnalysisException for view ${viewPair.name}: $sql", viewPair.name, sql),
90               views => {
91                 val missingNames =
92                   views.flatMap { name =>
93                     val lookupName =
94                       if (mapOf.contains(name))
95                         name
96                       else
97                         s"`$name`"
98 
99                     if (mapOf.contains(lookupName)) { // quoted must also be in the view name if it's got minus' etc., on Spark < 3.2 this will be incorrectly unquoted
100                       attemptCount += 1
101                       processView(mapOf(lookupName))
102                       None
103                     } else // not one we can actually do anything about
104                       Some(name)
105                   }
106                 if (!missingNames.isEmpty) {
107                   throw MissingViewAnalysisException(e, s"Missing relations for view ${viewPair.name}: $missingNames used in sql $sql", viewPair.name, sql, missingNames)
108                 }
109               }
110             )
111         }
112       }
113 
114     while((attemptCount < (viewConfigs.size * 2)) && !done){
115       attemptCount += 1
116       leftToProcess.headOption.fold {
117         done = true;
118       } { p =>
119         processView(p._2)
120       }
121     }
122     ViewLoadResults(replaced.toSet, !done, leftToProcess.keySet)
123   }
