30    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
31    *
32    * @param ruleSuite The Qualty RuleSuite to evaluate
33    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
34    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
35    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
36    * @param variableFuncGroup Defaulting to 20
37    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
38    * @return A Column representing the Quality DQ expression built from this ruleSuite
39    */
40   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
41                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false): Column = {
42     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
43     val flattened = flattenExpressions(ruleSuite)
44     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
45     val input =
46       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
47       if (compileEvals)
48         flattened.map(PassThroughCompileEvals)
49       else
50         flattened
51 
52     val runner =
53       if (forceRunnerEval || resolveWith.isDefined)
54         new RuleRunnerEval(cleaned, input,
55           compileEvals, variablesPerFunc, variableFuncGroup)
56       else
57         new RuleRunner(cleaned, input,
58           compileEvals, variablesPerFunc, variableFuncGroup)
59 
60     column(
61       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
62         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
63 
64         resolved.withNewChildren(resolved.children.map{
65           // replace the expr
66           case PassThroughCompileEvals(child) => NonPassThrough(child)
67           case child => NonPassThrough(child)
68         })
69       } getOrElse runner
70     )
71   }

274  *
275  * The variablePerFunc (40) and variableFuncGroup (20) parameters are chosen due to 64k method limitation,

277  * requires it.
