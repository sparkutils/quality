48    *
49    * @param ruleSuite The Qualty RuleSuite to evaluate
50    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
51    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
52    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
53    * @param variableFuncGroup Defaulting to 20
54    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
55    * @return A Column representing the Quality DQ expression built from this ruleSuite
56    */
57   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
58                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false): Column = {
59     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
60     val flattened = flattenExpressions(ruleSuite)
61     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
62     val input =
63       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
64       if (compileEvals)
65         flattened.map(PassThroughCompileEvals)
66       else
67         flattened
68 
69     val runner =
70       if (forceRunnerEval || resolveWith.isDefined)
71         new RuleRunnerEval(cleaned, input,
72           compileEvals, variablesPerFunc, variableFuncGroup)
73       else
74         new RuleRunner(cleaned, input,
75           compileEvals, variablesPerFunc, variableFuncGroup)
76 
77     column(
78       QualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
79         val resolved = QualitySparkUtils.resolveExpression(df, runner)
80 
81         resolved.withNewChildren(resolved.children.map{
82           // replace the expr
83           case PassThroughCompileEvals(child) => NonPassThrough(child)
84           case child => NonPassThrough(child)
85         })
86       } getOrElse runner
87     )
88   }

298 /**
299  * Children will be rewritten by the plan, either by resolveWith, it's then re-incorporated into ruleSuite
300  *
301  * The variablePerFunc (40) and variableFuncGroup (20) parameters are chosen due to 64k method limitation,
302  * is null for all of a 1600 column row blasts the 64k jvm limit.  You may customise these values if a given rulesuite
303  * requires it.
304  *
305  * @param variablesPerFunc How many variables are in a function
306  * @param variableFuncGroup How many functions are then grouped into a new function
