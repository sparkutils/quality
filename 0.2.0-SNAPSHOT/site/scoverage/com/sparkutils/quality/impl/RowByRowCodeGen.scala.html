13 /**
14  * CODE is based on MutableProjection and generates a transformation between two encoders over a middle operation.
15  * The generated class itself can create new instances directly, unlike Spark projections that need to go through the
16  * codegen source generation cycle, only the compilation is cached.
17  *
18  * If the expression tree contains stateful expressions with codegenfallback the code must be regenerated against a
19  * fresh tree.
20  */

38   def projections(ctx: CodegenContext, expressions: Seq[Expression], mutableRow: String, useSubexprElimination: Boolean = false) = {
39     val validExpr = expressions.zipWithIndex.filter {
40       case (NoOp, _) => false
41       case _ => true
42     }
43     val exprVals = ctx.generateExpressions(validExpr.map(_._1), useSubexprElimination).toIndexedSeq
44 
45     // 4-tuples: (code for projection, isNull variable name, value variable name, column index)
46     val projectionCodes: Seq[(ExprCode, String, String)] = validExpr.zip(exprVals).map {
47       case ((e, i), ev) =>
48         val value = JavaCode.global(
49           ctx.addMutableState(CodeGenerator.javaType(e.dataType), "value"),
50           e.dataType)
51         val (code, isNull) = if (e.nullable) {
52           val isNull = ctx.addMutableState(CodeGenerator.JAVA_BOOLEAN, "isNull")
53           (s"""
54               |${ev.code}
55               |$isNull = ${ev.isNull};
56               |$value = ${ev.value};
57             """.stripMargin, JavaCode.isNullGlobal(isNull))
58         } else {
59           (s"""
60               |${ev.code}
61               |$value = ${ev.value};
62             """.stripMargin, FalseLiteral)
63         }
64         val expr = ExprCode(isNull, value)
65 
66         val update = CodeGenerator.updateColumn(
67           mutableRow,
68           e.dataType,
69           i,
70           expr,
71           e.nullable)
72         (expr, code, update)
73     }
74     projectionCodes
75   }
