18    * Creates a column that runs the folding RuleSuite.  This also forces registering the lambda functions used by that RuleSuite.
19    *
20    * FolderRunner runs all output expressions for matching rules in order of salience, the startingStruct is passed ot the first

22    *
23    * @param ruleSuite The ruleSuite with runOnPassProcessors
24    * @param startingStruct This struct is passed to the first matching rule, ideally you would use the spark dsl struct function to refer to existing columns
25    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
26    * @param debugMode When debugMode is enabled the resultDataType is wrapped in Array of (salience, result) pairs to ease debugging
27    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this.
28    * @param variablesPerFunc Defaulting to 40 allows, in combination with variableFuncGroup allows customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen
29    * @param variableFuncGroup Defaulting to 20
30    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
31    * @param forceTriggerEval Defaulting to false, passing true forces each trigger expression to be compiled (compileEvals) and used in place, false instead expands the trigger in-line giving possible performance boosts based on JIT
32    * @param useType In the case you must use select and can't use withColumn you may provide a type directly to stop the NPE
33    * @return A Column representing the QualityRules expression built from this ruleSuite
34    */

66   def ruleFolderRunnerClassic(ruleSuite: RuleSuite, startingStruct: Column, compileEvals: Boolean = false,
67                        debugMode: Boolean = false, resolveWith: Option[DataFrame] = None, variablesPerFunc: Int = 40,
68                        variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, useType: Option[StructType] = None,
69                        forceTriggerEval: Boolean = false): Column = {
70     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
71 
72     // needed to resolve variables -- this changes between invocation and stops the type checks.  In the test case it's subcode that is on one type nullable and the other not
73     val dataRef = new AtomicReference[DataType]()
74 
75     val realType = () => {
76       val starter =
77         useType.getOrElse(dataRef.get())
78       if (debugMode)
79       // wrap it in an array with the priority result
80         ArrayType(StructType(Seq(StructField("salience", IntegerType), StructField("result", starter))))
81       else
82         starter
83     }
84 
85     val lazyRef = RefExpressionLazyType(dataRef, true)
86 
87     val liftLambda = (e: Expression) => FunN(Seq(lazyRef), e, usedAsLambda = true)
88 
89     val (expressions, indexes) = flattenExpressions(ruleSuite, liftLambda)
90 
91     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
92     val starter = expression(startingStruct)
93     val exprs =
94       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
95       if (compileEvals)
96         expressions.map(PassThroughCompileEvals(_))
97       else
98         expressions
99 
100     val runner =
101       if (forceRunnerEval || resolveWith.isDefined)
102         new RuleFolderRunnerEval(cleaned, starter +: exprs,
103           realType, compileEvals = compileEvals,
104           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
105           expressionOffsets = indexes, dataRef, forceTriggerEval)
106       else
107         new RuleFolderRunner(cleaned, starter +: exprs,
108           realType, compileEvals = compileEvals,
109           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
110           expressionOffsets = indexes, dataRef, forceTriggerEval)
111 
112     column(
113       QualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
114         val resolved = QualitySparkUtils.resolveExpression(df, runner)
115 
116         resolved.withNewChildren(resolved.children.map{
117           // replace the expr
118           case PassThroughCompileEvals(child) => NonPassThrough(child)
119           case child => NonPassThrough(child)
120         })
121       } getOrElse runner
122     )
123   }
