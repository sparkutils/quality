259   protected[quality] case class SameOrNextVersionLower[T](map: Map[Id, Seq[T]]) extends Function1[Id, Option[Seq[T]]] {
260     implicit val ordering: Ordering[Id] =
261       (x: Id, y: Id) => {
262         if (x == y)
263           0
264         else
265           if (x.id != y.id)
266             x.id - y.id
267           else
268             x.version - y.version
269       }
270 
271     val sorted = TreeMap[Id, Seq[T]]() ++ map
272 
273     override def apply(id: Id): Option[Seq[T]] =
274       sorted.to(id).lastOption.map(_._2)
275   }
