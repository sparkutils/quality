61       """
62        select ruleExpr, ruleId, ruleVersion, ruleEngineSalience, ruleEngineId, ruleEngineVersion,
63        ruleSetId, ruleSetVersion, -- don't need to bump versions as they can coexist
64         suiteversions.ruleSuiteId, suiteversions.ruleSuiteVersion -- force the versions to be bumped to latest ruleSuite Versions
65         from
66          (select distinct ruleSuiteId, ruleSuiteVersion from rules) suiteversions join
67          rules l0 on l0.ruleSuiteId = suiteversions.ruleSuiteId and l0.ruleSuiteVersion <= suiteversions.ruleSuiteVersion

69           not exists (

72              and l1.ruleId = l0.ruleId
73              and l1.ruleSetId = l0.ruleSetId
74              and l1.ruleSuiteVersion <= suiteversions.ruleSuiteVersion
75              and l1.ruleVersion > l0.ruleVersion
76           )
77           and l0.ruleExpr != "DELETED"
78        """)

200   protected[quality] def lambdaOutputSQL(tableName: String, extra: String ="") =
201     s"""
202      select $extra ruleExpr, functionId, functionVersion,
203       versions.ruleSuiteId, versions.ruleSuiteVersion -- force the versions to be bumped to latest ruleSuiteVersions
204       from
205        (select distinct ruleSuiteId, ruleSuiteVersion from $tableName) versions join
206        $tableName l0 on l0.ruleSuiteId = versions.ruleSuiteId and l0.ruleSuiteVersion <= versions.ruleSuiteVersion
207        where
208         not exists (
209           select 0 from $tableName l1
210           where l1.ruleSuiteId = l0.ruleSuiteId and
211            l1.functionId = l0.functionId and
212            l1.ruleSuiteVersion <= versions.ruleSuiteVersion
213            and l1.functionVersion > l0.functionVersion
214         )
215         and l0.ruleExpr != "DELETED"

280   protected[quality] case class SameOrNextVersionLower[T](map: Map[Id, Seq[T]]) extends Function1[Id, Option[Seq[T]]] {
281     implicit val ordering: Ordering[Id] =
282       (x: Id, y: Id) => {
283         if (x == y)
284           0
285         else
286           if (x.id != y.id)
287             x.id - y.id
288           else
289             x.version - y.version
290       }
291 
292     val sorted = TreeMap[Id, Seq[T]]() ++ map
293 
294     override def apply(id: Id): Option[Seq[T]] =
295       sorted.to(id).lastOption.map(_._2)
296   }
