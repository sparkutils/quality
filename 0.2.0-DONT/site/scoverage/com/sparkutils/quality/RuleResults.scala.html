171   def failed: Long
172   def passed: Long
173   def softFailed: Long
174   def disabled: Long
175   def ignored: Long
176   def defaulted: Long
177   def probabilityPassed: Long
178   def probabilityFailed: Long
179 
180   def update(failed: Long = failed, passed: Long = passed, softFailed: Long = softFailed, disabled: Long = disabled,
181              ignored: Long = ignored, defaulted: Long = defaulted, probabilityPassed: Long = probabilityPassed,
182              probabilityFailed: Long = probabilityFailed): T
183 
184   def combineResults(other: T): T =
185     update(failed = failed + other.failed, passed = passed + other.passed, softFailed = softFailed + other.softFailed,
186       disabled = disabled + other.disabled, ignored = ignored + other.ignored, defaulted = defaulted + other.defaulted,
187       probabilityPassed = probabilityPassed + other.probabilityPassed,
188       probabilityFailed = probabilityFailed + other.probabilityFailed)
189 
190   def combine(other: T): T
191 
192   @tailrec
193   final def processResult(ruleResult: RuleResult, probabilityPass: Double = 0.8d): T =
194     ruleResult match {
195       case Failed => update(failed = failed + 1)
196       case Passed => update(passed = passed + 1)
197       case SoftFailed => update(softFailed = softFailed + 1)
198       case DisabledRule => update(disabled = disabled + 1)
199       case IgnoredRule => update(ignored = ignored + 1)
200       case DefaultRule => update(defaulted = defaulted + 1)
201       case Probability(percentage) if (percentage >= probabilityPass) => update(probabilityPassed = probabilityPassed + 1)
202       case Probability(_) => update(probabilityFailed = probabilityFailed + 1)
203       case RuleResultWithProcessor(ruleResult: RuleResult, _) =>
204         processResult(ruleResult, probabilityPass)
205     }
