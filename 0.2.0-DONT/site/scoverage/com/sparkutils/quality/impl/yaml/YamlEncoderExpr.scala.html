142   private def createMapNode(mt: MapType, map: MapData, keyType: Any => Node, valueType: Any => Node)(implicit renderOptions: Map[String, String]) =
143     if (map == null)
144       createNullNode
145     else {
146       val smap = MapUtils.toScalaMap(map, mt.keyType, mt.valueType)

157   private def createStructNode(st: StructType)(implicit renderOptions: Map[String, String]) = {
158     val converters = st.fields.map(f =>

161     (row: InternalRow) => {
162       if (row == null)

164       else {
165         val tuples = st.fields.zipWithIndex.map {

167             new NodeTuple(

171         }
172         new MappingNode(Tag.MAP, tuples.toSeq.asJava, DumperOptions.FlowStyle.FLOW)
173       }

254     }
