221    */
222   def addExpressionRunner(dataFrame: DataFrame, ruleSuite: RuleSuite, name: String = "expressionResults",
223                        renderOptions: Map[String, String] = Map.empty, ddlType: String = "",
224                        stripDDL: Boolean = false, postProcess: DataFrame => DataFrame = identity): DataFrame = {
225     import org.apache.spark.sql.functions.expr
226     val runner =
227       if (ddlType.isEmpty)
228         expressionRunner(ruleSuite, name = name, renderOptions = renderOptions)
229       else
230         typedExpressionRunner(ruleSuite, name = name, ddlType = ddlType)
231 
232     postProcess(
233       dataFrame.select(expr("*"),
234         if (stripDDL)
235           strip_result_ddl(runner).as(name)
236         else
237           runner
238       )
239     )
