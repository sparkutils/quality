240     val openb = v.toString().indexOf("[")

249     val ordered = a.flatMap {
250       case a: VariableValue => Some(a)
251       case _ => None
252     }
253

255       val (stripped, arrayInName) = stripBrackets(v)
256 
257       val (typ, array) =
258         if (v.javaType.isArray)

260         else if (v.javaType.isPrimitive)

262         else
263           (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
264

268         if (v.javaType.isArray && callsKeepArrays)
