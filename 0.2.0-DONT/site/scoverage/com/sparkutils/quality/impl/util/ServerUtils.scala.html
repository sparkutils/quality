236     val openb = v.toString().indexOf("[")

244     // filter out any top level arrays, the input is a set, so params need the same order
245     val ordered = a.flatMap {
246       case a: VariableValue => Some(a)
247       case _ => None
248     }
249 
250     (ordered.map { v =>
251       val (stripped, arrayInName) = stripBrackets(v)
252 
253       val (typ, array) =
254         if (v.javaType.isArray)

256         else if (v.javaType.isPrimitive)

258         else
259           (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
260

264         if (v.javaType.isArray && callsKeepArrays)
