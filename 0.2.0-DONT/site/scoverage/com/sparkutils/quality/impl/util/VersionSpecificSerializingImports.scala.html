174   /**
175    * Returns a specific RuleSuite from available rule suites
176    * @param ds
177    * @param id
178    * @return
179    */
180   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
181     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
182       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSetVersion) ).map{
183         p =>
184           RuleSet(p._1, p._2.map{ cr =>
185             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
186               runOnPassProcessor =
187                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
188                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
189                   getOrElse(NoOpRunOnPassProcessor.noOp)
190             )
191           })
192       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
193         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
194       }}, probablePass = row.probablePass.getOrElse(0.8),
195       defaultProcessor = row.defaultProcessor.map{o =>
196         DefaultProcessor(Id(o.functionId, o.functionVersion), OutputExpression(o.ruleExpr))}.
197         getOrElse(NoOpDefaultProcessor.noOp))
198   }
