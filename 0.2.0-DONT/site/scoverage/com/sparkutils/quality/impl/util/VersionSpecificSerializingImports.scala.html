175   /**
176    * Returns a specific RuleSuite from available rule suites
177    * @param ds
178    * @param id
179    * @return
180    */
181   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
182     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
183       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSetVersion) ).map{
184         p =>
185           RuleSet(p._1, p._2.map{ cr =>
186             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
187               runOnPassProcessor =
188                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
189                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
190                   getOrElse(NoOpRunOnPassProcessor.noOp)
191             )
192           })
193       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
194         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
195       }}, probablePass = row.probablePass.getOrElse(defaultProbablePass),
196       defaultProcessor = row.defaultProcessor.map{o =>
197         DefaultProcessor(Id(o.functionId, o.functionVersion), OutputExpression(o.ruleExpr))}.
198         getOrElse(NoOpDefaultProcessor.noOp))
199   }
