38    * @return A Column representing the Quality DQ expression built from this ruleSuite

40   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
41                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false): Column = {
42     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
43     val flattened = flattenExpressions(ruleSuite)
44     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
45     val input =
46       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
47       if (compileEvals)
48         flattened.map(PassThroughCompileEvals)
49       else
50         flattened
51 
52     val runner =
53       if (forceRunnerEval || resolveWith.isDefined)
54         new RuleRunnerEval(cleaned, input,
55           compileEvals, variablesPerFunc, variableFuncGroup)
56       else
57         new RuleRunner(cleaned, input,
58           compileEvals, variablesPerFunc, variableFuncGroup)
59 
60     column(
61       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
62         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
63 
64         resolved.withNewChildren(resolved.children.map{
65           // replace the expr
66           case PassThroughCompileEvals(child) => NonPassThrough(child)
67           case child => NonPassThrough(child)
68         })
69       } getOrElse runner
70     )

133     import ruleSuite._
134 
135     val ruleSetRes = Array.ofDim[InternalRow](ruleSuiteArrays.ruleSetIds.length)
136 
137     var rsOverall = startingResult
138 
139     var offset = 0
140 
141     for (rsi <- 0 until ruleSuiteArrays.ruleSetIds.length) {
142

144 
145       val ruleSetResults = results.slice(offset, offset + rulesetSize)
146       offset += rulesetSize

148       val overall = ruleSetResults.foldLeft(startingResult) {
149         (ov, res) =>
150           processOverall(res.asInstanceOf[Int], ov, probablePass) // convert needed for process
151       }

284 /**
285  * Children will be rewritten by the plan, either by resolveWith, it's then re-incorporated into ruleSuite
286  *
287  * The variablePerFunc (40) and variableFuncGroup (20) parameters are chosen due to 64k method limitation,
288  * is null for all of a 1600 column row blasts the 64k jvm limit.  You may customise these values if a given rulesuite
289  * requires it.
290  *
291  * @param variablesPerFunc How many variables are in a function
292  * @param variableFuncGroup How many functions are then grouped into a new function
293  */
