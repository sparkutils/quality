11 /**
12  * Rewrites FunN's when not used as LambdaFunctions (e.g. a parent HigherOrderFunction or HigherOrderFunctionLike).
13  * The application is a simple replacement of the resolved input expression with all occurrences of the
14  * matching LambdaVariable.
15  *
16  * This happens early enough to allow for user functions to be subexpr eliminated (as of Spark 4 not possible
17  * with either CodegenFallback or an expression containing any LambdaVariables).
18  *
19  * The use of USED_AS_LAMBDA and any Spark HoF with a registered LambdaCompilationHandler will keep the expression
20  * as a lambda.  The LambdaCompilationHandler case is a trade-off between overhead of tree copies/compilation in
21  * processors, the underlying hof eval vs compile and sub expression elimination.
22  *
23  * IMPORTANT: This only works on 3.2 and above as it introduced transformDownWithPruning
24  */

35 
36   def shouldBeDisabled: Boolean = {
37     val (all, disabledRules) = disabledOptimiserRules()
38     if (all)
39       true
40     else
41       disabledRules.contains(className)
42   }
43 
44   def disabled: Boolean
45 
46   def funNHandled(f: FunN): Boolean = f.name.isDefined && compilationHandlers.contains(f.name.get)
47 
48   override def apply(plan: LogicalPlan): LogicalPlan =
49     if (disabled)
50       plan
51     else
52       ClassicQualitySparkUtils.funNRewrite(plan, {
53         case f: FunN if !f.usedAsLambda &&
54           // if a direct child is a rewrite HoF then we shouldn't disable compilation by ripping it out (#83)
55           !f.children.exists( t => t.collect {
56             // full class type for HoF
57             case e if compilationHandlers.contains(e.getClass.getName) => true
58             // specific FunN name
59             case f: FunN if funNHandled(f) => true
60           }.nonEmpty )
61           &&
62           // if this FunN should be handled then we shouldn't rewrite
63           !(funNHandled(f)) =>
64           val pairs = f.elementVars.zip(f.arguments).toMap
65           val r =
66             f.function.asInstanceOf[LambdaFunction].function.transform{
67               case e: NamedLambdaVariable if pairs.contains(e) => pairs(e)
68             }
69           r
70       })
