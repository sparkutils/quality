31 /**
32  * Registers Quality sql functions using the defaults for registerQualityFunctions, these can be overridden without having to subclass DriverPlugin.  Functions are registered via FunctionRegistry.builtIn making them available for view creation, if they should be registered via extensions only then use the quality_force_inject_function = true configuration.
33  *
34  * It also registers plan optimiser rule's such as AsUUIDFilter, which rewrites filters with variables backed by as_uuid.
35  *
36  * Optimiser rules can be disabled via the quality_disable_optimiser_rules system environment variable. "*" disables all rules, otherwise a comma separated list of fqn class names may be used.
37  */
38 class QualitySparkExtension extends ((SparkSessionExtensions) => Unit) with Logging {
39 
40   def parseTypes: String => Option[DataType] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultParseTypes
41   def zero: DataType => Option[Any] = com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultZero
42   def add: DataType => Option[(Expression, Expression) => Expression] = (dataType: DataType) => com.sparkutils.quality.impl.RuleRegistrationFunctions.defaultAdd(dataType)
43   def mapCompare: DataType => Option[(Any, Any) => Int] = (dataType: DataType) => utils.defaultMapCompare(dataType)
44   def writer: String => Unit = println
45 
46   /**
47    * uses writer to write prefix: str

51   def dump(str: String, prefix: String = "Quality SparkExtensions") = writer(s"$prefix: $str")
52 
53   /**
54    * attempts to logInfo, typically doesn't work, but also then dumps the str.  Doesn't log when testing

56    */
57   def attemptLogInfo(str: String) =
58     if (!Option(System.getProperty(testingConf)).contains("testing")) {
59       logInfo(str)
60       dump(str)
61     }

63   /**
64    * Adds AsymmetricFilterExpressions for AsUUID
65    * Derived implementations should also call super if they wish to register the default rules (or specify both as extensions).
66    * These are registered after resolution is done.
67    * The first String should be the fqn classname / rule name for the rule
68    * @return
69    */
70   def optimiserRules: Seq[(String, SparkSession => Rule[LogicalPlan])] =
71     Seq((AsUUIDFilter.getClass.getName, _ => AsUUIDFilter), (IDBase64Filter.getClass.getName, _ => IDBase64Filter),
72       (FunNRewrite.getClass.getName, _ => FunNRewrite))
73 
74   override def apply(extensions: SparkSessionExtensions): Unit = {
75     val func =
76       if (com.sparkutils.quality.getConfig(forceInjectFunction, "false").toBoolean) {
77         attemptLogInfo("registering quality functions via injection - they are classed as temporary functions")
78         ShimUtils.registerFunctionViaExtension(extensions) _
79       } else {
80         attemptLogInfo("registering quality functions via builtin function registry - whilst you can use these in global views the extension must always be present")
81         ShimUtils.registerFunctionViaBuiltin _
82       }
83     com.sparkutils.quality.classicFunctions.registerQualityFunctions(parseTypes, zero, add, mapCompare, writer,
84       registerFunction = func, fromExtension = true
85     )
86     extensions.injectParser(QualityFunctionParser(_,_))
87     extensions.injectParser(ConnectCommandParsers(_,_))
88 
89     if (Testing.testing) {
90       ExtensionTesting.disableRuleResult = ""
91     }
92     val (all, disabledRules) = disabledOptimiserRules()
93     if (!all) {
94       val filteredRules = optimiserRules.filterNot(p => disabledRules.contains(p._1.trim))
95       val str = s"$disableRulesConf = $disabledRules leaving ${filteredRules.map(_._1)} remaining"
96       attemptLogInfo(str)
97       if (Testing.testing) {
98         ExtensionTesting.disableRuleResult = str
99       }
100       filteredRules.map(_._2).foreach(extensions.injectOptimizerRule _)
101     } else {
102       val str = s"All optimiser rules are disabled via $disableRulesConf"
103       attemptLogInfo(str)
104     }
105   }
106 
107 }
