41    * @param getValue obtain the value from the mapData, with any necessary conversions, against the index parameter

46     val keys = mapData.keyArray()

49 
50     def search() = {
51       var i = 0
52       var found = false
53       while (i < n && !found) {
54         if (acc(i) == what) {

56         } else {

59       }

64     }

66     def withCached() =

72       }
