98    * Same as functions without the wrapping Column
99    * @param rule
100    * @return
101    */
102   def expr(rule: String): Expression = {
103     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
104       newParser()
105     }
106     /*
107     attempt for a simple expression, then try a plan with exactly one output row.
108     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
109      */
110     val rawExpr =
111       try {
112         parser.parseExpression(rule)
113       } catch {
114         case ot: Throwable => // e.g. suitable for output expressions
115           try {
116             ScalarSubquery(parser.parsePlan(rule))
117           } catch {
118             case _: Throwable =>
119               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
120               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
121               val r = rule.split("->")
122               val wrapped =
123                 if (r.size == 2)
124                   s"${r(0)} -> ( ${r(1)} )"
125                 else
126                   s"( ${r(0)} )"
127 
128               try {
129                 parser.parseExpression(wrapped)
130               } catch {
131                 case _: Throwable => throw ot // if this didn't work return the original error
132               }
133           }
134       }
135 
136     val res =
137       rawExpr match {
138         case l: SparkLambdaFunction if hasSubQuery(l) =>
139           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
140           val names = l.arguments.map(a => a.name -> a).toMap
141           l.transformUp {
142             case s: SubqueryExpression =>
143               s.withNewPlan( s.plan.transform{
144               case snippet =>
145                 snippet.transformAllExpressions {
146                   case a: UnresolvedAttribute =>
147                     names.get(a.name).map(lamVar => lamVar).getOrElse(a)
148                 }
149               }
150             )
151           }
152         case _ => rawExpr
153       }
154 
155     res.transformUp {
156       case s: SubqueryExpression => SubQueryWrapper(Seq(s))
157     }
158   }
