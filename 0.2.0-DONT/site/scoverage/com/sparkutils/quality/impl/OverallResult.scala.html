29 protected[quality] object OverallResultHelper {
30   @tailrec
31   protected[quality] def inplace(ruleResult: RuleResult, currentResult: RuleResult, probablePass: Double): RuleResult =
32     ruleResult match {
33       case Passed | SoftFailed | DisabledRule | IgnoredRule | DefaultRule => currentResult
34       case RuleResultWithProcessor(ruleResult, _) => inplace(ruleResult, currentResult, probablePass)
35       case Failed => Failed
36       case Probability(x) =>
37         if (x < probablePass)
38           Failed
39         else
40           currentResult
41     }
42 
43   @tailrec
44   protected[quality] def inplaceForDefault(ruleResult: RuleResult, currentResult: RuleResult, probablePass: Double): RuleResult =
45     ruleResult match {
46       case Passed => Passed
47       case Probability(x) =>
48         if (x < probablePass)
49           currentResult
50         else
51           Passed
52       case RuleResultWithProcessor(ruleResult, _) => inplaceForDefault(ruleResult, currentResult, probablePass)
53       case _ => currentResult
54     }
55 
56   protected[quality] def inplaceInt(ruleResult: Int, currentResult: Int, probablePass: Double): Int =
57     ruleResult match {
58       case PassedInt | SoftFailedInt | DisabledRuleInt | IgnoredRuleInt | DefaultRuleInt => currentResult
59       case FailedInt => FailedInt
60       case x =>

63         else

65     }

67   protected[quality] def inplaceForDefaultInt(ruleResult: Int, currentResult: Int, probablePass: Double): Int =
68     ruleResult match {
