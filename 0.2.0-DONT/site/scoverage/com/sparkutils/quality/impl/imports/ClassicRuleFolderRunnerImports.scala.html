18    * Creates a column that runs the folding RuleSuite.  This also forces registering the lambda functions used by that RuleSuite.
19    *
20    * FolderRunner runs all output expressions for matching rules in order of salience, the startingStruct is passed ot the first
21    * matching, the result passed to the second etc.  In contrast to ruleEngineRunner OutputExpressions should be lambdas with one parameter, that of the structure
22    *
23    * NOTE if using this in a Connect session resolveWith, compileEvals, forceRunnerEval and forceTriggerEval are ignored and use defaults
24    *
25    * @param ruleSuite The ruleSuite with runOnPassProcessors
26    * @param startingStruct This struct is passed to the first matching rule, ideally you would use the spark dsl struct function to refer to existing columns
27    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
28    * @param debugMode When debugMode is enabled the resultDataType is wrapped in Array of (salience, result) pairs to ease debugging
29    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this.
30    * @param variablesPerFunc Defaulting to 40 allows, in combination with variableFuncGroup allows customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen
31    * @param variableFuncGroup Defaulting to 20
32    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
33    * @param forceTriggerEval Defaulting to false, passing true forces each trigger expression to be compiled (compileEvals) and used in place, false instead expands the trigger in-line giving possible performance boosts based on JIT
34    * @param useType In the case you must use select and can't use withColumn you may provide a type directly to stop the NPE
35    * @return A Column representing the QualityRules expression built from this ruleSuite
36    */

68   def ruleFolderRunnerClassic(ruleSuite: RuleSuite, startingStruct: Column, compileEvals: Boolean = false,
69                        debugMode: Boolean = false, resolveWith: Option[DataFrame] = None, variablesPerFunc: Int = 40,
70                        variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false, useType: Option[StructType] = None,
71                        forceTriggerEval: Boolean = false): Column = {
72     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
73 
74     // needed to resolve variables -- this changes between invocation and stops the type checks.  In the test case it's subcode that is on one type nullable and the other not
75     val dataRef = new AtomicReference[DataType]()
76 
77     val realType = () => {
78       val starter =
79         useType.getOrElse(dataRef.get())
80       if (debugMode)
81       // wrap it in an array with the priority result
82         ArrayType(StructType(Seq(StructField("salience", IntegerType), StructField("result", starter))))
83       else
84         starter
85     }
86 
87     val lazyRef = RefExpressionLazyType(dataRef, true)
88 
89     val liftLambda = (e: Expression) => FunN(Seq(lazyRef), e, usedAsLambda = true)
90 
91     val (expressions, indexes, triggerCount) = flattenExpressions(ruleSuite, liftLambda)
92 
93     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
94     val starter = expression(startingStruct)
95     val exprs =
96       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
97       if (compileEvals)
98         expressions.map(PassThroughCompileEvals(_))
99       else
100         expressions
101 
102     val runner =
103       if (forceRunnerEval || resolveWith.isDefined)
104         new RuleFolderRunnerEval(cleaned, starter +: exprs,
105           realType, compileEvals = compileEvals,
106           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
107           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount)
108       else
109         new RuleFolderRunner(cleaned, starter +: exprs,
110           realType, compileEvals = compileEvals,
111           debugMode = debugMode, variablesPerFunc, variableFuncGroup,
112           expressionOffsets = indexes, dataRef, forceTriggerEval, triggerCount = triggerCount)
113 
114     column(
115       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
116         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
117 
118         resolved.withNewChildren(resolved.children.map{
119           // replace the expr
120           case PassThroughCompileEvals(child) => NonPassThrough(child)
121           case child => NonPassThrough(child)
122         })
123       } getOrElse runner
124     )
125   }
