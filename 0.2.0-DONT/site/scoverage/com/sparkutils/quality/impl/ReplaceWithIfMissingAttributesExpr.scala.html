18   /**
19    * Processes a given RuleSuite to replace any coalesceIfMissingAttributes.  This may be called before validate / docs but
20    * *must* be called *before* adding the expression to a dataframe.
21    *
22    * @param ruleSuite
23    * @param schema The names to validate against, if empty no attempt to process coalesceIfAttributeMissing will be made
24    * @return
25    */
26   def processIfAttributeMissing(ruleSuite: RuleSuite, schema: StructType = StructType(Seq())): RuleSuite = {
27     val names = LookupIdFunctions.namesFromSchema(schema)
28 
29     val lambdas = ruleSuite.lambdaFunctions.map(lf => ProcessDisableIfMissing.processIfMissingLambdaCoalesce(lf.parsed, names))
30 
31     val wrs = ruleSuite.copy(lambdaFunctions = lambdas)
32 
33     mapRules(wrs) {
34       ProcessDisableIfMissing.processCoalesceIfAttributeMissing(_, names)
35     }
36   }

39   /**
40    * For a given schema evaluates all calls to coalesce in a lambda to swap out missing attributes
41    *
42    * Must be called before any planning - otherwise Spark will already have rejected it.
43    *
44    * @param rule
45    * @return
46    */
47   def processIfMissingLambdaCoalesce(rule: LambdaFunctionParsed, names: Set[String]): LambdaFunction =
48     rule.expr match {
49       case lambdaFunc: SparkLambdaFunction =>
50         lambdaFunc.function match {
51           case func: UnresolvedFunction =>
52             val rexp = processCoalesceIfAttributeMissing(func, names)
53             rule.copy(expr = lambdaFunc.copy(function = rexp))
54 
55           case _ => rule
56         }
57     }

69   /**
70    * Converts expression removing any coalesceIfAttributeMissing placeholding expressions using names to find each
71    * first expression with no missing names (or replacing with null should all of them be).
72    * @param expression
73    * @param names
74    * @return
75    */
76   def processCoalesceIfAttributeMissing(expression: Expression, names: Set[String]): Expression =
77     expression match {
78       case funcExpr: UnresolvedFunction if isReplaceCoalesceName(funcExpr) =>
79         processCoalesceIfAttributeMissing(expression.children, names, isCoalesceDisabled(funcExpr))
80       case e: Expression =>
81         e.withNewChildren(e.children.map(processCoalesceIfAttributeMissing(_, names)))
82     }
83 
84   /**
85    * Returns the first expression that has no missing names or null
86    * @param expressions
87    * @param names
88    * @return
89    */
90   protected def processCoalesceIfAttributeMissing(expressions: Seq[Expression], names: Set[String], isDisabled: Boolean): Expression = {
91     val res =
92       expressions.flatMap{ e =>
93         // make sure no nested children are coalesce's
94         val r = processCoalesceIfAttributeMissing(e, names)
95         // get the names from this expression
96         val explookup = fieldsFromExpression(r)
97         // lookup each of them in the names and if any are not true we can't use this expression
98         if (!explookup.attributesUsed.map(names).forall(identity))
99           None
100         else
101           Some(r)
102       }
103 
104     if (res.isEmpty) {
105       if (isDisabled) DisabledRuleExpr // special case as this is probably before the function registry gets called
106       else
107         Literal(null, NullType)
108     } else
109       res.head
110   }
111 
112   /**
113    * Converts both rule expressions and any output expressions removing any coalesceIfAttributeMissing placeholding expressions using names to find each
114    * first expression with no missing names (or replacing with null should all of them be).
115    * @param names
116    * @return
117    */
118   protected[quality] def processCoalesceIfAttributeMissing(rule: Rule, names: Set[String]): Rule =
119     rule match {
120       // rule and output
121       case Rule(id, e: quality.HasRuleText,
122         iorule @ quality.RunOnPassProcessor.RunOnPassProcessorImpl(_, _, _, o: quality.HasRuleText)) if o.rule.nonEmpty =>
123         Rule(id, ExpressionRuleExpr(e.rule, processCoalesceIfAttributeMissing(RuleLogicUtils.expr(e.rule), names)),
124           iorule.copy(returnIfPassed = OutputExpressionExpr(o.rule,
125             processCoalesceIfAttributeMissing(RuleLogicUtils.expr(o.rule), names))))
126 
127       // just a rule
128       case orule @ Rule(_, e: quality.HasRuleText, _) =>
129         orule.copy( expression = ExpressionRuleExpr(e.rule, processCoalesceIfAttributeMissing(RuleLogicUtils.expr(e.rule), names)))
130 
131       case _ =>
132         rule
133     }
