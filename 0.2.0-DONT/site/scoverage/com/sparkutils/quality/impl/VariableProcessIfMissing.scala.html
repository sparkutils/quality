16   private def checkNestedCallsInSuite(ruleSuite: RuleSuite): Unit = {
17 
18     def checkRuleSuiteHasProcess(rule: String, ruleId: Id, typ: String): Unit =
19       if (rule.toLowerCase.contains(process_if_attribute_missing_name)) {
20         qualityException(s"$process_if_attribute_missing_name should not be used in RuleSuite expressions but was found in $typ with Id $ruleId with text $rule")
21       }
22 
23     // unlike classic rulesuite calls these will still have rule text
24     ruleSuite.ruleSets.foreach(ruleSet => ruleSet.rules.map { rule =>
25       rule.expression match {
26         case h: quality.HasRuleText =>
27           checkRuleSuiteHasProcess(h.rule, rule.id, "trigger rule")
28       }
29       if (rule.runOnPassProcessor != NoOpRunOnPassProcessor.noOp) {
30         checkRuleSuiteHasProcess(rule.runOnPassProcessor.rule, rule.runOnPassProcessor.id, "output expression")
31       }
32     })
33     ruleSuite.lambdaFunctions.foreach{
34       l =>
35         checkRuleSuiteHasProcess(l.rule, l.id, "lambda function")
36     }
37   }

39   protected[quality] def registerProcessIfAttributeMissingForAgnostic(registerFunction: (String, Seq[Expression] => Expression) => Unit): Unit = {
40     // parse the rulesuite, call the functions with structs, set a new variable, probably needs to be direct
41     registerFunction(process_if_attribute_missing_name, {
42       case Seq(OfRuleSuite(ruleSuite), ddl, name) =>

49         // call for side effect

51 
52         // the resulting rulesuite will have the corrected unresolved expressions already available,
53         // the raw sql remains untouched
54         val r = com.sparkutils.quality.classicFunctions.processIfAttributeMissing(ruleSuite, s)
55 
56         ShimUtils.createVariable(getString(name, 2), Literal.create(RuleSuiteHelpers.serialize(r), BinaryType), true)
57     })
58   }
