143   def allMergedIn(base: MapData, newRows: Seq[(InternalRow, Boolean)]): Boolean = {
144     val inplace = base.valueArray().asInstanceOf[GenericArrayData].array
145 
146     val bN = base.numElements()
147     val bK = base.keyArray()
148 
149     var notIn = false
150     var i = 0
151     while( i < newRows.length && !notIn) {
152       val id = newRows(i)._1.getLong(0)
153       var j = 0
154       var in = false
155       while( j < bN && !in) {
156         if (bK.getLong(j) == id) {
157           in = true

159         }
160         j += 1
161       }

163       i += 1
164     }

206               else
207                 // it's the rule level result

209               , rs, stats, depth + 1) // this should be the correct row, so we should be inside already...
210 
211       // if any new rows were added below then we'll have to re-create all above
212       val createdSubMap = newRows.exists(_._2)
213       def newRow: InternalRow = {
214         // if rs is GenericArrayData map based we may be able to in-place swap entries, scanning will be quicker than map changes

223             i <- 0 until nextSM.numElements()

226           }
227           for {

229           } {

233           val values = new GenericArrayData(newMap.values.toArray)

235         }
236       }
237 
238       (if (depth == stats.length || !(createdNewMap || createdSubMap))
239         rs
240       else
241         newRow,
242         createdNewMap || createdSubMap)
243     }
244

246     if (res._2) {

249       import head._
