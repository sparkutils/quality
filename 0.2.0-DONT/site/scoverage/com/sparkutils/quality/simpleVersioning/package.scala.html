249   protected[quality] case class SameOrNextVersionLower[T](map: Map[Id, Seq[T]]) extends Function1[Id, Option[Seq[T]]] {
250     implicit val ordering: Ordering[Id] =
251       (x: Id, y: Id) => {
252         if (x == y)
253           0
254         else
255           if (x.id != y.id)
256             x.id - y.id
257           else
258             x.version - y.version
259       }
260 
261     val sorted = TreeMap[Id, Seq[T]]() ++ map
262 
263     override def apply(id: Id): Option[Seq[T]] =
264       sorted.to(id).lastOption.map(_._2)
265   }
