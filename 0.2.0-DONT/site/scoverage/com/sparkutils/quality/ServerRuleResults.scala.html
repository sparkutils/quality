47       disabled = t.disabled + other.disabled, ignored = t.ignored + other.ignored, defaulted = t.defaulted + other.defaulted,

53   @tailrec
54   final def processResult(t: T)(ruleResult: RuleResult, probabilityPass: Double = defaultProbablePass): T =
55     ruleResult match {
56       case Failed => update(t)(failed = t.failed + 1)
57       case Passed => update(t)(passed = t.passed + 1)
58       case SoftFailed => update(t)(softFailed = t.softFailed + 1)
59       case DisabledRule => update(t)(disabled = t.disabled + 1)
60       case IgnoredRule => update(t)(ignored = t.ignored + 1)
61       case DefaultRule => update(t)(defaulted = t.defaulted + 1)
62       case Probability(percentage) if (percentage >= probabilityPass) => update(t)(probabilityPassed = t.probabilityPassed + 1)
63       case Probability(_) => update(t)(probabilityFailed = t.probabilityFailed + 1)
64       case RuleResultWithProcessor(ruleResult: RuleResult, _) =>
65         processResult(t)(ruleResult, probabilityPass)
66     }
