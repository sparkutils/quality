51   @tailrec
52   final def processResult(t: T)(ruleResult: RuleResult, probabilityPass: Double = 0.8d): T =
53     ruleResult match {
54       case Failed => update(t)(failed = t.failed + 1)
55       case Passed => update(t)(passed = t.passed + 1)
56       case SoftFailed => update(t)(softFailed = t.softFailed + 1)
57       case DisabledRule => update(t)(disabled = t.disabled + 1)
58       case IgnoredRule => update(t)(ignored = t.ignored + 1)
59       case DefaultRule => update(t)(defaulted = t.defaulted + 1)
60       case Probability(percentage) if (percentage >= probabilityPass) => update(t)(probabilityPassed = t.probabilityPassed + 1)
61       case Probability(_) => update(t)(probabilityFailed = t.probabilityFailed + 1)
62       case RuleResultWithProcessor(ruleResult: RuleResult, _) =>
63         processResult(t)(ruleResult, probabilityPass)
64     }
