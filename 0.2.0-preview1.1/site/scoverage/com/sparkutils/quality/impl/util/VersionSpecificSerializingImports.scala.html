164    * Returns a specific RuleSuite from available rule suites
165    * @param ds
166    * @param id
167    * @return
168    */
169   def rule_suite(row: CombinedRuleSuiteRows): RuleSuite = {
170     RuleSuite(Id(row.ruleSuiteId, row.ruleSuiteVersion),
171       row.ruleRows.groupBy(r => Id(r.ruleRow.ruleSetId, r.ruleRow.ruleSetVersion) ).map{
172         p =>
173           RuleSet(p._1, p._2.map{ cr =>
174             Rule(Id(cr.ruleRow.ruleId, cr.ruleRow.ruleVersion), ExpressionRule(cr.ruleRow.ruleExpr),
175               runOnPassProcessor =
176                 cr.outputExpressionRow.map(or => RunOnPassProcessor(cr.ruleRow.ruleEngineSalience,
177                     Id(or.functionId, or.functionVersion), OutputExpression(or.ruleExpr))).
178                   getOrElse(NoOpRunOnPassProcessor.noOp)
179             )
180           })
181       }.toSeq, lambdaFunctions = row.lambdaFunctions.fold(Seq.empty[LambdaFunction]){_.map { lr =>
182         LambdaFunction(lr.name, lr.ruleExpr, Id(lr.functionId, lr.functionVersion))
183       }}, probablePass = row.probablePass.getOrElse(defaultProbablePass),
184       defaultProcessor = row.defaultProcessor.map{o =>
185         DefaultProcessor(Id(o.functionId, o.functionVersion), OutputExpression(o.ruleExpr))}.
186         getOrElse(NoOpDefaultProcessor.noOp))
187   }
