58    * Same as functions)without the wrapping Column
59    * @param rule
60    * @return
61    */
62   def expr(rule: String) = {
63     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
64       newParser()
65     }
66     /*
67     attempt for a simple expression, then try a plan with exactly one output row.
68     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
69      */
70     val rawExpr =
71       try {
72         parser.parseExpression(rule)
73       } catch {
74         case ot: Throwable => // e.g. suitable for output expressions
75           try {
76             ScalarSubquery(parser.parsePlan(rule))
77           } catch {
78             case _: Throwable =>
79               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
80               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
81               val r = rule.split("->")
82               val wrapped =
83                 if (r.size == 2)
84                   s"${r(0)} -> ( ${r(1)} )"
85                 else
86                   s"( ${r(0)} )"
87 
88               try {
89                 parser.parseExpression(wrapped)
90               } catch {
91                 case _: Throwable => throw ot // if this didn't work return the original error
92               }
93           }
94       }
95 
96     val res =
97       rawExpr match {
98         case l: SparkLambdaFunction if hasSubQuery(l) =>
99           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
100           val names = l.arguments.map(a => a.name -> a).toMap
101           l.transformUp {
102             case s: SubqueryExpression =>
103               s.withNewPlan( s.plan.transform{
104               case snippet =>
105                 snippet.transformAllExpressions {
106                   case a: UnresolvedAttribute =>
107                     names.get(a.name).map(lamVar => lamVar).getOrElse(a)
108                 }
109               }
110             )
111           }
112         case _ => rawExpr
113       }
114 
115     res.transformUp {
116       case s: SubqueryExpression => SubQueryWrapper(Seq(s))
117     }
118   }
