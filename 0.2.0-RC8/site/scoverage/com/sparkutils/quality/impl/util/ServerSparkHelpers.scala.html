13   /**
14    * UnsafeArrayData doesn't allow calling .array, foreach when needed and for others use array
15    *
16    * @param array
17    * @param dataType
18    * @param f
19    * @return
20    */
21   def mapArray[T: ClassTag](array: ArrayData, dataType: DataType, f: Any => T, by: Int = 0): Array[T] =
22     array match {
23       case _: UnsafeArrayData =>
24         val res = Array.ofDim[T](array.numElements() + by)
25         array.foreach(dataType, (i, v) => res.update(i, f(v)))
26         res
27       case _ => {
28         val r = array.array.map(f)
29         if (by == 0)
30           r
31         else {
32           val res = Array.ofDim[T](array.numElements() + by)
33           r.copyToArray(res)
34           res
35         }
36 
37       }
38     }

42    * @param array

47     array match {

139    *
140    * @param mapData the source mapdata to search through.

144    * @tparam T

148     val keys = mapData.keyArray()
149     val n = keys.numElements()
150 
151     def withCached() =
152       cachedPositions.find { i =>
153         if (i < n)

227     var index = -1
228     var i = 0
229     val keys = map.keyArray()
230     while(index == -1 && i < map.numElements()) {

233       }
234       i += 1
235     }
