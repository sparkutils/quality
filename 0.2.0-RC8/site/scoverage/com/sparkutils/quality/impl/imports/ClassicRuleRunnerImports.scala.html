11   /**
12    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
13    *
14    * NOTE if using this in a Connect session resolveWith, compileEvals and forceRunnerEval are ignored and use defaults
15    *
16    * @param ruleSuite The Qualty RuleSuite to evaluate
17    * @param compileEvals Should the rules be compiled out to interim objects - by default false
18    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
19    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
20    * @param variableFuncGroup Defaulting to 20
21    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
22    * @return A Column representing the Quality DQ expression built from this ruleSuite
23    */
24   def ruleRunner(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
25                  variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
26                  extraConfig: Map[String, String] = Map.empty): Column =
27     if (checkResolveMakesSenseOrClassic(resolveWith))
28       RuleRunnerImpl.ruleRunnerImplClassic(ruleSuite, compileEvals, resolveWith,
29         variablesPerFunc, variableFuncGroup, forceRunnerEval, extraConfig)
30     else
31       ShimUtils.callFunction("dq_rule_runner", lit(RuleSuiteHelpers.serialize(ruleSuite)),
32         lit(variablesPerFunc), lit(variableFuncGroup), typedLit(extraConfig))
