390             code"""
391               boolean ${exprRunner.isNull} = false;
392               ${funNames.map { f => s"$f(${params.paramsCall});" }.mkString("\n")}

396               }.mkString(";\n")}
397               """
398         ), gr.extraClasses)

401     val eval = group.groupFilter.genCode(ctx)
402 
403     // if ruleEngine is used salience may need comparison, if it's expression or dq any comparison is meaningless

411         }
412         """
