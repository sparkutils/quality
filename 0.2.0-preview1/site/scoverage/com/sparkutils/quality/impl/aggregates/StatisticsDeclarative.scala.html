184       var i = 0

188         }

212               if (nextInputRowSize > 0)
213                 resultRows.valueArray().getStruct(i, nextInputRowSize)
214               else
215                 // it's the rule level result

217               , rs, stats, depth + 1) // this should be the correct row, so we should be inside already...

219       // if any new rows were added below then we'll have to re-create all above
220       val createdSubMap = newRows.exists(_._2)

231             i <- 0 until nextSM.numElements()

234           }

237           } {

239           }

241           val values = new GenericArrayData(newMap.values.toArray)
242           nextStatsBuildWhenNewMap(rs, new ArrayBasedMapData(keys, values))
243         }
244       }
245 
246       (if (depth == stats.length || !(createdNewMap || createdSubMap))
247         rs
248       else
249         newRow,
250         createdNewMap || createdSubMap)
251     }
252

254     if (res._2) {

256       val head = stats.head
257       import head._
258       // either we need to re-integrate it or it was brand new
259       val m = curGroup.getMap(statsMapOffset)
260       var rs_i = -1
261       var i = 0
