95     dataType match {
96       case _: MapType => Some((left, right) => MapMerge(Seq(left, right), (dataType: DataType) => defaultAdd(dataType, extension)))
97       case _: IntegerType | LongType | DoubleType =>
98         Some((left, right) => add(left, right, null))
99       case a: DecimalType =>
100         Some((left, right) => add(left, right, a))
101       case _ => extension(dataType)
102     }
