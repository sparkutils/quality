35    * Creates a column that runs the RuleSuite.  This also forces registering the lambda functions used by that RuleSuite
36    *
37    * @param ruleSuite The Qualty RuleSuite to evaluate
38    * @param compileEvals Should the rules be compiled out to interim objects - by default false, allowing optimisations
39    * @param resolveWith This experimental parameter can take the DataFrame these rules will be added to and pre-resolve and optimise the sql expressions, see the documentation for details on when to and not to use this. RuleRunner does not currently do wholestagecodegen when resolveWith is used.
40    * @param variablesPerFunc Defaulting to 40, it allows, in combination with variableFuncGroup customisation of handling the 64k jvm method size limitation when performing WholeStageCodeGen.  You _shouldn't_ need it but it's there just in case.
41    * @param variableFuncGroup Defaulting to 20
42    * @param forceRunnerEval Defaulting to false, passing true forces a simplified partially interpreted evaluation (compileEvals must be false to get fully interpreted)
43    * @return A Column representing the Quality DQ expression built from this ruleSuite
44    */
45   def ruleRunnerImplClassic(ruleSuite: RuleSuite, compileEvals: Boolean = false, resolveWith: Option[DataFrame] = None,
46                      variablesPerFunc: Int = 40, variableFuncGroup: Int = 20, forceRunnerEval: Boolean = false,
47                      extraConfig: Map[String, String] = Map.empty): Column = {
48     com.sparkutils.quality.registerLambdaFunctions( ruleSuite.lambdaFunctions )
49     val flattened = flattenExpressions(ruleSuite)
50     val cleaned = RuleLogicUtils.cleanExprs(ruleSuite)
51     val input =
52       // ExpressionProxy and SubExprEvaluationRuntime cannot be used with compileEvals
53       if (compileEvals)
54         flattened.map(PassThroughCompileEvals)
55       else
56         flattened
57 
58     val runner =
59       if (forceRunnerEval || resolveWith.isDefined)
60         new RuleRunnerEval(cleaned, input,
61           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
62       else
63         new RuleRunner(cleaned, input,
64           compileEvals, variablesPerFunc, variableFuncGroup, extraConfig)
65 
66     column(
67       ClassicQualitySparkUtils.resolveWithOverride(resolveWith).map { df =>
68         val resolved = ClassicQualitySparkUtils.resolveExpression(df, runner)
69 
70         resolved.withNewChildren(resolved.children.map{
71           // replace the expr
72           case PassThroughCompileEvals(child) => NonPassThrough(child)
73           case child => NonPassThrough(child)
74         })
75       } getOrElse ZeroCodeGenWrap.wrap(runner)
76     )
77   }

182   def nonOutputRuleGen[T: ClassTag](ctx: CodegenContext, runner: Runner, ev: ExprCode, utilsName: String,
183                        realChildren: Seq[Expression], variablesPerFunc: Int, variableFuncGroup: Int,
184                        resultF: (ExprValue, Int) => String, extraConfig: Map[String, String],
185                        ruleRunnerExpressionIdx: Int
186                       ): ExprCode = {

188     import paramInfo._
189 
190     val resTerms = resultRowTerms(ctx, ruleRunnerExpressionIdx)
191     import resTerms._
192 
193     val inPlaceOffsets = runner.inPlaceArrayOffsets(ctx, resultRow, ruleRunnerExpressionIdx)
194 
195     val allExpr = realChildren.zipWithIndex.map { case (child, idx) =>
196       val generate =
197         (ctx: CodegenContext, p: ParameterInformation) => {
198           val eval = child.genCode(ctx)
199 
200           code"""${eval.code}\n
201 
202             ${inPlaceOffsets.offsets(idx).apply(resultF(eval.value, idx))}
203              """
204         }
205       (Trigger(child, idx, 0), generate)
206     }
207 
208     val resName = ctx.freshName("result")
209     val resNull = ctx.freshName("isNull")
210 
211     val funNames: Iterator[String] =
212       RuleRunnerUtils.generateFunctionGroups(ctx, runner, paramInfo, resultRow,
213         Seq(VariableValue(resultRow, classOf[InternalRow])), allExpr,
214         variablesPerFunc, variableFuncGroup, extraConfig)._1
215 
216     val exp = ExprCode(VariableValue(resName, ev.value.javaType), isNullVariable(resNull))
217 
218     val res = exp.copy(code =
219       code"""
220       // copy row
221       $resultRowCopy
222       ${funNames.map { f => s"$f($paramsCall);" }.mkString("\n")}
223 
224       InternalRow ${exp.value} = $resultRow;
225       boolean ${exp.isNull} = false;
226       """
227     )
228 
229     res
230   }

248  * Children will be rewritten by the plan, either by resolveWith, it's then re-incorporated into ruleSuite
249  *
250  * The variablePerFunc (40) and variableFuncGroup (20) parameters are chosen due to 64k method limitation,
251  * is null for all of a 1600 column row blasts the 64k jvm limit.  You may customise these values if a given rulesuite
252  * requires it.

254  * @param variablesPerFunc How many variables are in a function
255  * @param variableFuncGroup How many functions are then grouped into a new function

289   protected def doGenCodeI(outerCtx: CodegenContext, ev: ExprCode): ExprCode = {
290 
291     val (clazz, fres) = SeparateCompilation.withSubExpressions(this, realChildren, outerCtx, ev, ruleSuite.id) {
292       (ctx, ruleRunnerExpressionIdx, _) =>
293 
294         // must be called before the rule gen runs
295         val params = genParams(ctx, this)
296 
297         // bind the rules
298         val utilsName = "com.sparkutils.quality.impl.RuleRunnerUtils"
299 
300         val res =
301           nonOutputRuleGen[T](ctx, this, ev, utilsName, realChildren, variablesPerFunc, variableFuncGroup,
302             (code: ExprValue, idx: Int) => s"com.sparkutils.quality.impl.RuleLogicUtils.anyToRuleResultInt($code)",
303             extraConfig, ruleRunnerExpressionIdx
304           )

306       ((params, classOf[RuleRunnerBase[T]].getName), res, Seq.empty)
307     }
308     generatorClassSource = clazz
309     fres
310   }
