43   private[quality] def attempt(bin: Array[Byte]): Option[T] =
44     try {

46     } catch {

48     }

52   private[quality] val enc = implicitly[Encoder[E]]
53   private[quality] val valueClass = enc.schema.fields.size == 1 && enc.schema.fields(0).name == "value"
54   private[quality] val combinedRowType: DataType = {
55     if (valueClass)
56       enc.schema.fields(0).dataType
57     else

59   }

62   def unapply(expression: Any): Option[T] = {
63     if (encoder eq null) {
64       encoder = ShimUtils.expressionEncoder(enc).resolveAndBind()
65     }
66 
67     val sameType =
68       equalsIgnoreCaseAndNullability(expression.asInstanceOf[Expression].dataType, combinedRowType)
69 
70     expression match {
71       case e: Literal if e.dataType == oType =>
72         Option(e.value).map(_.asInstanceOf[T])
73       case e: Literal if e.dataType == BinaryType =>
74         attempt(getBinary(e, 0))
75       case e: VariableReference if e.dataType == BinaryType =>
76         attempt(e.eval().asInstanceOf[Array[Byte]])
77       case e: VariableReference if sameType =>

83         )))
84       case _ => None
85     }
