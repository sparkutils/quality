107    * Same as functions without the wrapping Column
108    * @param rule
109    * @return
110    */
111   def expr(rule: String): Expression = {
112     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
113       newParser()
114     }
115     /*
116     attempt for a simple expression, then try a plan with exactly one output row.
117     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
118      */
119     val rawExpr =
120       try {
121         parser.parseExpression(rule)
122       } catch {
123         case ot: Throwable => // e.g. suitable for output expressions
124           try {
125             ScalarSubquery(parser.parsePlan(rule))
126           } catch {
127             case _: Throwable =>
128               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
129               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
130               val r = rule.split("->")
131               val wrapped =
132                 if (r.size == 2)
133                   s"${r(0)} -> ( ${r(1)} )"
134                 else
135                   s"( ${r(0)} )"
136 
137               try {
138                 parser.parseExpression(wrapped)
139               } catch {
140                 case _: Throwable => throw ot // if this didn't work return the original error
141               }
142           }
143       }
144 
145     val res =
146       rawExpr match {
147         case l: SparkLambdaFunction if hasSubQuery(l) =>
148           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
149           val names = l.arguments.map(a => a.name -> a).toMap
150           l.transformUp {
151             case s: SubqueryExpression =>
152               s.withNewPlan( s.plan.transform{
153               case snippet =>
154                 snippet.transformAllExpressions {
155                   case a: UnresolvedAttribute =>
156                     names.get(a.name).map(lamVar => lamVar).getOrElse(a)
157                 }
158               }
159             )
160           }
161         case _ => rawExpr
162       }
163 
164     res.transformUp {
165       case s: SubqueryExpression => SubQueryWrapper(Seq(s))
166     }
