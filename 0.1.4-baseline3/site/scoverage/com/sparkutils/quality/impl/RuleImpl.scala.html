68   def expr(rule: String) = {
69     val parser = SparkSession.getActiveSession.map(_.sessionState.sqlParser).getOrElse {
70       newParser()
71     }
72     /*
73     attempt for a simple expression, then try a plan with exactly one output row.
74     "In" will handle ListQuery, "Exists" similarly exists, everything else will likely fail as a parser error
75      */
76     val rawExpr =
77       try {
78         parser.parseExpression(rule)
79       } catch {
80         case ot: Throwable => // e.g. suitable for output expressions
81           try {
82             ScalarSubquery(parser.parsePlan(rule))
83           } catch {
84             case _: Throwable =>
85               // quite possibly a lambda using a subquery so try and force it via ( sub ) trick,
86               // but allow us to replace later after resolution.  DBR > 12 requires () on simple expressions as well
87               val r = rule.split("->")
88               val wrapped =
89                 if (r.size == 2)
90                   s"${r(0)} -> ( ${r(1)} )"
91                 else
92                   s"( ${r(0)} )"
93 
94               try {
95                 parser.parseExpression(wrapped)
96               } catch {
97                 case _: Throwable => throw ot // if this didn't work return the original error
98               }
99           }
100       }
101 
102     val res =
103       rawExpr match {
104         case l: SparkLambdaFunction if hasSubQuery(l) =>
105           // The lambda's will be parsed as UnresolvedAttributes and not the needed lambdas
106           val names = l.arguments.map(a => a.name -> a).toMap
107           l.transformUp {
108             case s: SubqueryExpression =>
109               s.withNewPlan( s.plan.transform{
110               case snippet =>
111                 snippet.transformAllExpressions {
112                   case a: UnresolvedAttribute =>
113                     names.get(a.name).map(lamVar => lamVar).getOrElse(a)
114                 }
115               }
116             )
117           }
118         case _ => rawExpr
119       }
120 
121     res.transformUp {
122       case s: SubqueryExpression => SubQueryWrapper(Seq(s))
123     }
124   }
