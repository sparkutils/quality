52  *
53  * Law: Each Rule generated must have a stable Id for the same column, the version used is the same as the RuleSet

56  *

69                           ruleExpr: String) {
70   val arg = ruleExpr.split("->")(0).trim
71   val rulePart = ruleExpr.split("->")(1).trim
72   //val columnFilterExp = RuleLogicUtils.expr(columnFilter)

74   /**
75    * @param stablePosition - should always be the same for this column, evaluated with columnName
76    *
77    */
78   protected[quality] case class MetaRuleGenerator( stablePosition: String => Int) {
79     /**
80      * Create a new rule - this must obey the law stated above
81      *
82      * @param columnName     - will be lowercase fed by filterColumns
83      * @return
84      */
85     def withNameAndOrd(columnName: String) =
86       Rule(Id(ruleSetId + stablePosition(columnName), ruleSetVersion),
87         ExpressionRule(rulePart.replaceAll(s"\\b$arg\\b", columnName)))
88   }

90   /**

92    * @param dataFrame
93    * @return a lower case set of column names
94    */
95   protected[quality] def filterColumns(dataFrame: DataFrame, transform: DataFrame => DataFrame = identity): Set[String] = {
96     import dataFrame.sparkSession
97     import sparkSession.implicits._
98 
99     val origds = sparkSession.createDataset( dataFrame.schema.fields.map(f=>
100       SimpleField(f.name, f.dataType.sql, f.nullable)) ).toDF()
101     val ds = transform(origds)
102       .filter(columnFilter)
103     ds.collect().map(_.getAs[String]("name")).map(_.toLowerCase).toSet
104   }

113   /**
114    * Generates rules over a given dataframe using the stablePosition function
115    * @param dataFrame
116    * @param stablePosition - should always be the same for this column, evaluated with columnName
117    * @param transform - allows enriching of the column, e.g. joining another table for extra filtering information.
118    * @return
119    */
120   def generateRuleSet(dataFrame: DataFrame, stablePosition: String => Int, transform: DataFrame => DataFrame = identity): RuleSet = RuleSet(Id(ruleSetId, ruleSetVersion),{
121     val cols = filterColumns(dataFrame, transform).toSeq // remove dupes, but stay as seq
122     val gen = createGenerator(stablePosition)
123     import gen.withNameAndOrd
124     cols.map(withNameAndOrd)
125   })
