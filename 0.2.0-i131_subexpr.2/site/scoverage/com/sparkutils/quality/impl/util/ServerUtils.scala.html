317   def stripBrackets(v: VariableValue): (String, String) = {
318     val openb = v.toString().indexOf("[")
319     if (openb == -1)
320       (v.variableName, "")
321     else
322       (v.variableName.dropRight(v.length - openb), v.variableName.drop(openb))
323   }
324 
325   def formatParams(ctx: CodegenContext, a: Seq[ExprValue], callsKeepArrays: Boolean = false): ParameterInformation = {
326     // filter out any top level arrays, the input is a set, so params need the same order
327     val ordered = a.flatMap {
328       case a: VariableValue => Some(a)
329       case _ => None
330     }
331 
332     val pairs = ordered.map { v =>
333       val (stripped, arrayInName) = stripBrackets(v)
334 
335       val (typ, array) =
336         if (v.javaType.isArray)
337           (s"${v.javaType.getComponentType.getName}", "[]")
338         else if (v.javaType.isPrimitive)
339           (v.javaType.toString, arrayInName.replaceAll("[^\\[\\]]",""))
340         else
341           (v.javaType.getName, arrayInName.replaceAll("[^\\[\\]]",""))
342 
343       (s"$typ$array", stripped, v.javaType)
344     }
345 
346     ParameterInformation(pairs.map {
347       case (typ, stripped, _) =>
348 
349         s"$typ $stripped"
350       }.mkString(", ")
351       , ordered.map(v =>
352         if (v.javaType.isArray && callsKeepArrays)
353           v.variableName
354         else
355           stripBrackets(v)._1
356       ).mkString(", "), ordered.size, pairs)
357   }
